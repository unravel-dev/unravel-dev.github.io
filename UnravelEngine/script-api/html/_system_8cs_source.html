189 // The core logic for iteration in priority order, plus deferred add & remove cleanup
190 private void InvokeInternal(Action<ScriptComponent> action)
191 {
192 isInvoking = true;
193 // Don't clear removedDuringInvoke - it's already empty or will be cleared at end
194
195 // Determine which method we're calling
196 bool isUpdate = ReferenceEquals(action, updateAction);
197 bool isFixedUpdate = ReferenceEquals(action, fixedUpdateAction);
198 bool isLateUpdate = ReferenceEquals(action, lateUpdateAction);
199
200 try
201 {
202 // Iterate through type buckets in priority order
203 for (int i = 0; i < sortedBuckets.Count; i++)
204 {
205 var bucket = sortedBuckets[i];
206
207 // Skip bucket if it doesn't override the method we're calling
208 if (isUpdate && !bucket.overrides.hasUpdate) continue;
209 if (isFixedUpdate && !bucket.overrides.hasFixedUpdate) continue;
210 if (isLateUpdate && !bucket.overrides.hasLateUpdate) continue;
211
212 var slots = bucket.slots;
213
214 // Iterate through slots (including nulls/gaps from removals)
215 for (int j = 0; j < slots.Count; j++)
216 {
217 var comp = slots[j];
218 // Fast path: skip HashSet lookup if no components were removed during invoke
219 if (!ReferenceEquals(comp, null) && (removedDuringInvoke.Count == 0 || !removedDuringInvoke.Contains(comp)))
220 {
221 action(comp);
222 }
223 }
224 }
225 }
226 finally
227 {
228 isInvoking = false;
229 }
230
231 // Process pending operations
232 if (pendingOps.Count > 0)
233 {
234 for (int i = 0; i < pendingOps.Count; i++)
235 {
236 var op = pendingOps[i];
237 // Skip cancelled operations (Comp is null when operations were collapsed)
238 if (ReferenceEquals(op.Comp, null)) continue;
239
240 if (op.add)
241 {
242 InsertComponent(op.Comp);
243 }
244 else
245 {
246 RemoveComponent(op.Comp);
247 }
248 }
249 pendingOps.Clear();
250 pendingOpIndexByComponent.Clear();
251 }
252
253 // Clear removed tracking AFTER processing pending operations (reuse the HashSet)
254 if (removedDuringInvoke.Count > 0)
255 {
256 removedDuringInvoke.Clear();
257 }
258 }

276 // Inserts component into appropriate type bucket, only if it overrides at least one update method
277 private void InsertComponent(ScriptComponent comp)
278 {
279 Type t = comp.GetType();
280
281 // Check which methods this type overrides (cached per type)
282 MethodOverrides overrides = GetMethodOverrides(t);
283
284 // Skip components that don't override any update methods
285 if (!overrides.hasUpdate && !overrides.hasFixedUpdate && !overrides.hasLateUpdate)
286 {
287 return;
288 }
289
290 int priority = GetPriorityFor(t); // Pass Type directly to avoid redundant GetType() call
291
292 // Get or create bucket for this type
293 if (!bucketsByType.TryGetValue(t, out var bucket))
294 {
295 bucket = new TypeBucket { type = t, priority = priority, overrides = overrides };
296 bucketsByType[t] = bucket;
297
298 // Insert bucket in sorted position (binary search)
299 InsertBucketSorted(bucket);
300 }
301
302 // Try to reuse a free slot first (from previous removals)
303 int slotIndex;
304 if (bucket.freeSlots.Count > 0)
305 {
306 // Reuse last free slot (O(1) removal from end of list)
307 slotIndex = bucket.freeSlots[bucket.freeSlots.Count - 1];
308 bucket.freeSlots.RemoveAt(bucket.freeSlots.Count - 1);
309 bucket.slots[slotIndex] = comp;
310 }
311 else
312 {
313 // No free slots, append new slot
314 slotIndex = bucket.slots.Count;
315 bucket.slots.Add(comp);
316 }
317
318 // Cache location for O(1) removal
319 locationByComponent[comp] = new ComponentLocation { bucket = bucket, slotIndex = slotIndex };
320 }

322 // Gets or caches which methods a type overrides
323 private MethodOverrides GetMethodOverrides(Type t)
324 {
325 if (methodOverrideCache.TryGetValue(t, out var cached))
326 {
327 return cached;
328 }
329
330 // Check if type overrides each method (check once per type, then cache)
331 var overrides = new MethodOverrides
332 {
333 hasUpdate = IsMethodOverridden(t, "OnUpdate"),
334 hasFixedUpdate = IsMethodOverridden(t, "OnFixedUpdate"),
335 hasLateUpdate = IsMethodOverridden(t, "OnLateUpdate")
336 };
337
338 methodOverrideCache[t] = overrides;
339 return overrides;
340 }

342 // Checks if a type overrides a specific method from ScriptComponent
343 private static bool IsMethodOverridden(Type derivedType, string methodName)
344 {
345 var method = derivedType.GetMethod(methodName,
346 System.Reflection.BindingFlags.Public |
347 System.Reflection.BindingFlags.Instance |
348 System.Reflection.BindingFlags.DeclaredOnly);
349
350 // If method is declared in derived type, it's overridden
351 if (method != null && method.DeclaringType == derivedType)
352 {
353 return true;
354 }
355
356 // Check base types up to (but not including) ScriptComponent
357 Type currentType = derivedType.BaseType;
358 while (currentType != null && currentType != scriptComponentBaseType)
359 {
360 method = currentType.GetMethod(methodName,
361 System.Reflection.BindingFlags.Public |
362 System.Reflection.BindingFlags.Instance |
363 System.Reflection.BindingFlags.DeclaredOnly);
364
365 if (method != null && method.DeclaringType == currentType)
366 {
367 return true;
368 }
369
370 currentType = currentType.BaseType;
371 }
372
373 return false;
374 }

376 // O(1) removal using cached location lookup
377 private void RemoveComponent(ScriptComponent comp)
378 {
379 // O(1) lookup instead of linear search
380 if (!locationByComponent.TryGetValue(comp, out var location))
381 {
382 // Component not found - this can happen if:
383 // 1. Component was never added (didn't override any methods)
384 // 2. Component was already removed
385 // 3. Component was added during invocation and immediately removed (operations collapsed)
386 // In all cases, we should ensure it's not in removedDuringInvoke for next frame
387 removedDuringInvoke.Remove(comp);
388 return;
389 }
390
391 // Mark slot as free (null) instead of removing - no element shifting needed
392 location.bucket.slots[location.slotIndex] = null;
393 location.bucket.freeSlots.Add(location.slotIndex);
394
395 locationByComponent.Remove(comp);
396
397 // Compact bucket if it has too many gaps (tunable threshold)
398 // This prevents unbounded memory growth while keeping removals fast
399 if (location.bucket.freeSlots.Count > 16 &&
400 location.bucket.freeSlots.Count > location.bucket.slots.Count / 2)
401 {
402 CompactBucket(location.bucket);
403 }
404 }

406 // Compacts a bucket by removing null slots and rebuilding the free list
407 private void CompactBucket(TypeBucket bucket)
408 {
409 int writeIndex = 0;
410 for (int readIndex = 0; readIndex < bucket.slots.Count; readIndex++)
411 {
412 var comp = bucket.slots[readIndex];
413 if (!ReferenceEquals(comp, null))
414 {
415 if (writeIndex != readIndex)
416 {
417 bucket.slots[writeIndex] = comp;
418 // Update cached location - must write back to dictionary since ComponentLocation is a struct
419 locationByComponent[comp] = new ComponentLocation { bucket = bucket, slotIndex = writeIndex };
420 }
421 writeIndex++;
422 }
423 }
424
425 // Trim excess slots
426 if (writeIndex < bucket.slots.Count)
427 {
428 bucket.slots.RemoveRange(writeIndex, bucket.slots.Count - writeIndex);
429 }
430
431 bucket.freeSlots.Clear();
432 }

434 // Binary search insertion to keep buckets sorted by priority
435 private void InsertBucketSorted(TypeBucket bucket)
436 {
437 int left = 0;
438 int right = sortedBuckets.Count;
439
440 while (left < right)
441 {
442 int mid = (left + right) / 2;
443 if (sortedBuckets[mid].priority <= bucket.priority)
444 {
445 left = mid + 1;
446 }
447 else
448 {
449 right = mid;
450 }
451 }
452
453 sortedBuckets.Insert(left, bucket);
454 }

476 private void CollapseOperations(ScriptComponent comp, bool isAdd)
477 {
478 // O(1) lookup for last operation index
479 if (!pendingOpIndexByComponent.TryGetValue(comp, out int lastOpIndex))
480 {
481 // No previous operations for this component, just add the new one
482 int newIndex = pendingOps.Count;
483 pendingOps.Add(new PendingOp { add = isAdd, Comp = comp });
484 pendingOpIndexByComponent[comp] = newIndex;
485
486 // Track removal for fast lookup during invoke
487 if (!isAdd)
488 {
489 removedDuringInvoke.Add(comp);
490 }
491 }
492 else
493 {
494 var lastOp = pendingOps[lastOpIndex];
495
496 if (lastOp.add && !isAdd)
497 {
498 // Last was add, this is remove -> cancel both operations
499 // Mark the operation as cancelled (set Comp to null) instead of removing from list
500 // to avoid invalidating indices in pendingOpIndexByComponent
501 pendingOps[lastOpIndex] = new PendingOp { add = false, Comp = null };
502 pendingOpIndexByComponent.Remove(comp);
503 removedDuringInvoke.Add(comp);
504 }
505 else if (!lastOp.add && isAdd)
506 {
507 // Last was remove, this is add -> cancel both operations
508 pendingOps[lastOpIndex] = new PendingOp { add = false, Comp = null };
509 pendingOpIndexByComponent.Remove(comp);
510 removedDuringInvoke.Remove(comp);
511 }
512 else if (lastOp.add && isAdd)
513 {
514 // Last was add, this is add -> do nothing (already added)
515 // No need to add another add operation
516 }
517 else if (!lastOp.add && !isAdd)
518 {
519 // Last was remove, this is remove -> do nothing (already removed)
520 // No need to add another remove operation
521 }
522 }

589 {
590 int gen0 = GC.CollectionCount(0);
591 int gen1 = GC.CollectionCount(1);
592 int gen2 = GC.CollectionCount(2);
593 long memory = GC.GetTotalMemory(false);
594
595 int gen0Delta = gen0 - lastGen0Count;
596 int gen1Delta = gen1 - lastGen1Count;
597 int gen2Delta = gen2 - lastGen2Count;

600 // Log every time (even if no change) for debugging
601 if (gen0Delta > 0 || gen1Delta > 0 || gen2Delta > 0 || Math.Abs(memDelta) > 1024)

603 // Get Mono heap size (native memory used by Mono runtime)
604 long monoHeap = internal_m2n_get_dotnet_heap_size();
605 long monoUsed = internal_m2n_get_dotnet_used_size();
606
607 Log.Info($"[GC] {context} - Collections: Gen0={gen0Delta}, Gen1={gen1Delta}, Gen2={gen2Delta} | Managed: {memDelta / 1024.0:F2} KB (Total: {memory / 1024.0:F2} KB) | Mono Heap: {monoHeap / (1024.0 * 1024.0):F2} MB (Used: {monoUsed / (1024.0 * 1024.0):F2} MB)");
608 }
609
610 lastGen0Count = gen0;
611 lastGen1Count = gen1;
612 lastGen2Count = gen2;
613 lastMemory = memory;
