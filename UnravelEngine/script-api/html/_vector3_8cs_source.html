252 public static Vector3 Slerp(Vector3 a, Vector3 b, float t)
253 {
254 // get cosine of angle between vectors (-1 -> 1)
255 float CosAlpha = Vector3.Dot(a, b);
256 // get angle (0 -> pi)
257 float Alpha = Mathf.Acos(CosAlpha);
258 // get sine of angle between vectors (0 -> 1)
259 float SinAlpha = Mathf.Sin(Alpha);
260 // this breaks down when SinAlpha = 0, i.e. Alpha = 0 or pi
261 float t1 = Mathf.Sin((1.0f - t) * Alpha) / SinAlpha;
262 float t2 = Mathf.Sin(t * Alpha) / SinAlpha;
263
264 // interpolate src vectors
265 return a * t1 + b * t2;
266 }
