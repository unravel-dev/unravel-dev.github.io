257 // helper to check one axis
258 bool Slab(float o, float d, float mn, float mx, ref float t0, ref float t1)
259 {
260 if (MathF.Abs(d) > float.Epsilon)
261 {
262 float inv = 1f / d;
263 float t1n = (mn - o) * inv;
264 float t2n = (mx - o) * inv;
265 if (t1n > t2n) (t1n, t2n) = (t2n, t1n);
266 t0 = MathF.Max(t0, t1n);
267 t1 = MathF.Min(t1, t2n);
268 return t0 <= t1;
269 }
270 else
271 {
272 return o >= mn && o <= mx;
273 }
274 }

284 public bool Intersect(Vector3 origin, Vector3 dir, out float t, bool restrictRange = true)
