124 public static Quaternion SlerpUnclamped(Quaternion x, Quaternion y, float a)
125 {
126 Quaternion z = y;
127
128 float cosTheta = Quaternion.Dot(x, y);
129
130 // If cosTheta < 0, the interpolation will take the long way around the sphere.
131 // To fix this, one quat must be negated.
132 if(cosTheta < 0.0f)
133 {
134 z = -y;
135 cosTheta = -cosTheta;
136 }
137
138 // Perform a linear interpolation when cosTheta is close to 1 to avoid side effect of sin(angle) becoming a zero denominator
139 if(cosTheta > 1.0f - Mathf.kEpsilon)
140 {
141 // Linear interpolation
142 Quaternion result;
143 result.x = Mathf.LerpUnclamped(x.x, z.x, a);
144 result.y = Mathf.LerpUnclamped(x.y, z.y, a);
145 result.z = Mathf.LerpUnclamped(x.z, z.z, a);
146 result.w = Mathf.LerpUnclamped(x.w, z.w, a);
147 return result;
148 }
149 else
150 {
151 // Essential Mathematics, page 467
152 float angle = Mathf.Acos(cosTheta);
153 return (Mathf.Sin((1.0f - a) * angle) * x + Mathf.Sin(a * angle) * z) / Mathf.Sin(angle);
154 }
155 }

335 public static Vector3 operator *(Quaternion rotation, Vector3 point)
336 {
337 float num = rotation.x * 2f;
338 float num2 = rotation.y * 2f;
339 float num3 = rotation.z * 2f;
340 float num4 = rotation.x * num;
341 float num5 = rotation.y * num2;
342 float num6 = rotation.z * num3;
343 float num7 = rotation.x * num2;
344 float num8 = rotation.x * num3;
345 float num9 = rotation.y * num3;
346 float num10 = rotation.w * num;
347 float num11 = rotation.w * num2;
348 float num12 = rotation.w * num3;
349 Vector3 result = default(Vector3);
350 result.x = (1f - (num5 + num6)) * point.x + (num7 - num12) * point.y + (num8 + num11) * point.z;
351 result.y = (num7 + num12) * point.x + (1f - (num4 + num6)) * point.y + (num9 - num10) * point.z;
352 result.z = (num8 - num11) * point.x + (num9 + num10) * point.y + (1f - (num4 + num5)) * point.z;
353 return result;
354 }
