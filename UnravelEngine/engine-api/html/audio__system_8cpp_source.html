212void audio_system::on_play_end(rtti::context& ctx)
213{
214 APPLOG_TRACE("{}::{}", hpp::type_name_str(*this), __func__);
215
216 auto& ec = ctx.get_cached<ecs>();
217 auto& scn = ec.get_scene();
218 auto& registry = *scn.registry;
219
220 registry.view<audio_source_component, active_component>().each(
221 [&](auto e, auto&& comp, auto&& active)
222 {
223 comp.on_play_end();
224 });
225
226 // One-shots and their clip limits belong to the session that started them; bus volumes keep
227 // whatever a fade reached.
228 voices_.stop_all();
229 for(auto& fade : bus_fades_)
230 {
231 fade.is_active = false;
232 }
233
234 registry.on_construct<active_component>().disconnect<&on_create_active_component>();
235 registry.on_destroy<active_component>().disconnect<&on_destroy_active_component>();
236
237 registry.on_construct<audio_source_component>().disconnect<&on_create_component>();
238 registry.on_destroy<audio_source_component>().disconnect<&on_destroy_component>();
239}

269void audio_system::update_fades(rtti::context& ctx, float seconds)
270{
271 for(std::size_t i = 0; i < audio_bus_count; ++i)
272 {
273 if(bus_fades_[i].is_active)
274 {
275 apply_bus_volume(static_cast<audio_bus>(i), bus_fades_[i].advance(seconds));
276 }
277 }

279 registry.view<audio_source_component, active_component>().each(
280 [&](auto e, auto&& comp, auto&& active)
281 {
282 comp.update_fade(seconds);
283 });
284 voices_.update(seconds);

287void audio_system::on_frame_update(rtti::context& ctx, delta_t dt)
288{
289 APP_SCOPE_PERF("Audio/System Update");
290 auto& ev = ctx.get_cached<events>();
291
292 auto& ec = ctx.get_cached<ecs>();
293 auto& scn = ec.get_scene();
294 auto& registry = *scn.registry;
295
296 // update auidio spatial properties from transform
297 registry.view<transform_component, audio_listener_component, active_component>().each(
298 [&](auto e, auto&& transform, auto&& comp, auto&& active)
299 {
300 comp.update(transform.get_transform_global(), dt);
301 });
302
303 registry.view<transform_component, audio_source_component, active_component>().each(
304 [&](auto e, auto&& transform, auto&& comp, auto&& active)
305 {
306 comp.update(transform.get_transform_global());
307 });
308
309 update_fades(ctx, get_unscaled_seconds(ctx, dt));
310}
