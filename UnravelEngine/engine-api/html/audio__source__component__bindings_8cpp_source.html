346auto internal_m2n_audio_play_one_shot(const hpp::uuid& clip,
347 float volume,
348 float pitch,
349 std::uint8_t bus,
350 int priority,
351 bool spatial,
352 const math::vec3& position) -> uint64_t
353{
354 auto system = find_audio_system();
355 if(!system || clip.is_nil())
356 {
357 return 0;
358 }
359 auto& am = engine::context().get_cached<asset_manager>();
360 one_shot_params params;
361 params.volume = volume;
362 params.pitch = pitch;
363 params.bus = bus < audio_bus_count ? static_cast<audio_bus>(bus) : audio_bus::sfx;
364 params.priority = priority;
365 params.spatial = spatial;
366 params.position = position;
367 return system->play_one_shot(am.get_asset<audio_clip>(clip), params);
368}
