24struct animation_state
25{
26 using seconds_t = animation_clip::seconds_t;
27
29 auto get_progress() const -> float
30 {
31 if(blend_space)
32 {
33 return phase;
34 }
35 if(pending_progress)
36 {
37 return *pending_progress;
38 }
39 // Resolved by the time step; a state not stepped yet looks the clip up.
40 const auto clp = loaded_clip ? loaded_clip : clip.try_get();
41 if(clp && clp->duration > animation_clip::seconds_t{0})
42 {
43 return elapsed / clp->duration;
44 }
45 return 0.0f;
46 }
47 void set_progress(float progress)
48 {
49 progress = std::isfinite(progress) ? math::clamp(progress, 0.0f, 1.0f) : 0.0f;
50 if(blend_space)
51 {
52 phase = progress;
53 begin_step();
54 return;
55 }
56 // Never waits: a clip still loading takes the progress once the time step resolves it.
57 const auto clp = loaded_clip ? loaded_clip : clip.try_get();
58 if(clp && clp->duration > animation_clip::seconds_t{0})
59 {
60 elapsed = clp->duration * progress;
61 pending_progress.reset();
62 begin_step();
63 return;
64 }
65 pending_progress = progress;
66 }
67
69 void begin_step()
70 {
71 step_begin_elapsed = elapsed;
72 step_begin_phase = phase;
73 step_begin_loop_count = loop_count;
74 }
75
77 auto get_cycle_position() const -> double
78 {
79 return double(loop_count) + double(get_progress());
80 }
81
84 void reset_event_window(bool inclusive = true)
85 {
86 event_cursor = get_cycle_position();
87 event_cursor_inclusive = inclusive;
88 event_window_valid = false;
89 }

139struct blend_over_time
140{
141 animation_clip::seconds_t duration{};
142 animation_clip::seconds_t elapsed{};
143
144 auto get_progress() const -> float
145 {
146 // A degenerate blend duration completes immediately (avoids 0/0 -> NaN,
147 // which would never satisfy the >= 1 completion check).
148 if(duration.count() <= 0.0f)
149 {
150 return 1.0f;
151 }
152 // Compute the normalized blending time (clamped between 0 and 1)
153 auto normalized_blend_time = static_cast<float>(elapsed.count() / duration.count());
154 normalized_blend_time = std::clamp(normalized_blend_time, 0.0f, 1.0f);
155 return normalized_blend_time;
156 }
