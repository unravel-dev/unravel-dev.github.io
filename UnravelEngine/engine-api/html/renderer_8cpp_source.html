138auto renderer::create_window_for_display(int index, const std::string& title, uint32_t flags)
139 -> const std::unique_ptr<render_window>&
140{
141 auto mode = os::display::get_desktop_mode(index);
142 auto bounds = os::display::get_usable_bounds(index);
143
144 if(flags & os::window::resizable)
145 {
146 auto frame_size = init_window_->get_frame_size();
147 bounds.y += static_cast<int>(static_cast<float>(frame_size.top) / mode.display_scale);
148 bounds.h -= static_cast<int>(static_cast<float>(frame_size.top + frame_size.bottom) / mode.display_scale);
149 bounds.x += static_cast<int>(static_cast<float>(frame_size.left) / mode.display_scale);
150 bounds.w -= static_cast<int>(static_cast<float>(frame_size.left + frame_size.right) / mode.display_scale);
151 }
152
153 os::window window(title, bounds.x, bounds.y, bounds.w * mode.display_scale, bounds.h * mode.display_scale, flags);
154
155 set_main_window(std::move(window));
156 return render_window_;
157}

170auto renderer::init_backend(const cmd_line::parser& parser) -> bool
171{
172 init_window_.reset();
173 init_window_ =
174 std::make_unique<os::window>("INIT", os::window::centered, os::window::centered, 64, 64, os::window::hidden);
175 const auto sz = init_window_->get_size();
176
177 gfx::init_type init_data;
178 init_data.type = get_renderer_type(parser);
179 init_data.resolution.width = sz.w;
180 init_data.resolution.height = sz.h;
181 init_data.resolution.reset = get_reset_flags(parser);
182 init_data.platformData.ndt = init_window_->get_native_display();

188 }
189 reset_flags_ = init_data.resolution.reset;
190
191 init_data.limits.numDrawCalls = 65536;
192 init_data.limits.numDrawCallPeakFrames = 0;
193
194 if(!gfx::init(init_data))
195 {
196 APPLOG_ERROR("Could not initialize rendering backend!");
197 return false;
198 }
199 APPLOG_TRACE("Using {0} rendering backend.", gfx::get_renderer_name(gfx::get_renderer_type()));

201 APPLOG_TRACE("DebugDraw Init.");
202 ddInit();

204 const bgfx::Caps* caps = bgfx::getCaps();
205 if (0 != (caps->supported & BGFX_CAPS_GRAPHICS_DEBUGGER) )

248auto renderer::get_renderer_type(const cmd_line::parser& parser) const -> gfx::renderer_type
249{
250 // auto detect
251 auto preferred_renderer_type = gfx::renderer_type::Count;
252
253 std::string preferred_renderer;
254 if(parser.try_get("renderer", preferred_renderer))
255 {
256 if(preferred_renderer == "opengl")
257 {
258 preferred_renderer_type = gfx::renderer_type::OpenGL;

260 else if(preferred_renderer == "vulkan")

262 preferred_renderer_type = gfx::renderer_type::Vulkan;

271 }

351void renderer::frame_begin(rtti::context& ctx, delta_t /*dt*/)
352{
353 // Drive GPU eviction at the start of the frame, before this frame's resource creates are
354 // submitted, so freed memory precedes new allocations in bgfx's command stream (lowers peak
355 // usage). Runs on the render thread, independently of the editor, using the policy persisted in
356 // the project's graphics settings (a no-op until a settings instance exists).
357 if(ctx.has<settings>())
358 {
359 update_eviction(ctx.get<settings>().graphics.eviction);
360 }
361
362 auto window = get_main_window();
363 if(window)
364 {
365 auto& pass = window->begin_present_pass();
366 pass.clear();
367 }
368}

370void renderer::frame_end(rtti::context& /*ctx*/, delta_t /*dt*/)
371{
372 APP_SCOPE_PERF("Graphics Frame Submit");
373
374 gfx::render_pass pass(gfx::render_pass::get_max_pass_id(), "Backbuffer/Update Pass");
375 pass.bind();
376
377 gfx::frame();
378
379 if(!request_screenshot_.empty())
380 {
381 gfx::request_screen_shot(get_main_window()->get_surface()->native_handle(), request_screenshot_.c_str());
382 request_screenshot_ = {};
383 }
384
385 gfx::render_pass::reset();
386}
