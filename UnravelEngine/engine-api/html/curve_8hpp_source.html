42template<typename T>
43void curve<T>::set_linked(bool linked)
44{
45 linked_ = linked;
46 if(!linked)
47 {
48 return;
49 }
50 // Every channel takes a key wherever any channel has one, holding the value it sampled there,
51 // so linking keeps the curve's shape instead of discarding the keys of the other channels.
52 const auto positions = collect_key_positions();
53 channels_t linked_channels;
54 for(size_t channel = 0; channel < channel_count; ++channel)
55 {
56 const auto& source = channels_[channel];
57 if(source.is_empty())
58 {
59 continue;
60 }
61 curve_channel::keys_t keys;
62 keys.reserve(positions.size());
63 for(float position : positions)
64 {
65 curve_key key = source.get_key(0);
66 const auto& source_keys = source.get_keys();
67 const auto existing = std::find_if(source_keys.begin(),
68 source_keys.end(),
69 [position](const curve_key& candidate)
70 { return candidate.position == position; });
71 if(existing != source_keys.end())
72 {
73 key = *existing;
74 }
75 else
76 {
77 key.position = position;
78 key.value = source.sample(position);
79 }
80 keys.push_back(key);
81 }
82 linked_channels[channel].set_keys(std::move(keys));
83 linked_channels[channel].refresh_automatic_tangents();
84 }
85 channels_ = linked_channels;
86 mark_bake_dirty();
87}

132template<typename T>
133auto curve<T>::sample(float position) const -> T
134{
135 if(bake_table_.empty())
136 {
137 return sample_direct(position);
138 }
139
140 const float width = bake_domain_.max - bake_domain_.min;
141 if(width <= 0.0f)
142 {
143 return bake_table_.front();
144 }
145
146 const float alpha = (position - bake_domain_.min) / width;
147 if(alpha <= 0.0f)
148 {
149 return bake_table_.front();
150 }
151 if(alpha >= 1.0f)
152 {
153 return bake_table_.back();
154 }
155
156 const float entry = alpha * float(bake_table_.size() - 1);
157 const size_t index = size_t(entry);
158 if(index + 1 >= bake_table_.size())
159 {
160 return bake_table_.back();
161 }
162 if(!bake_interpolates_)
163 {
164 return bake_table_[index];
165 }
166
167 const float fraction = entry - float(index);
168 T result{};
169 for(size_t channel = 0; channel < channel_count; ++channel)
170 {
171 const float low = traits::get(bake_table_[index], channel);
172 const float high = traits::get(bake_table_[index + 1], channel);
173 traits::set(result, channel, low + (high - low) * fraction);
174 }
175 return result;
176}

178template<typename T>
179void curve<T>::bake(size_t resolution)
180{
181 // A table of one entry cannot span anything, and the lookup divides by its last index.
182 if(!is_valid() || resolution < 2)
183 {
184 clear_bake();
185 return;
186 }
187
188 bake_domain_ = get_domain();
189 bake_interpolates_ = true;
190 for(const auto& channel : channels_)
191 {
192 if(channel.has_constant_key())
193 {
194 bake_interpolates_ = false;
195 break;
196 }
197 }
198
199 const float width = bake_domain_.max - bake_domain_.min;
200 bake_table_.resize(resolution);
201 for(size_t entry = 0; entry < resolution; ++entry)
202 {
203 const float alpha = float(entry) / float(resolution - 1);
204 bake_table_[entry] = sample_direct(bake_domain_.min + width * alpha);
205 }
206}

330template<typename T>
331void curve<T>::set_points_keeping_shape(const points_t& points)
332{
333 constexpr float same_position = 1.0e-5f;
334 const channels_t previous = channels_;
335 set_points(points);
336 for(size_t channel = 0; channel < channel_count; ++channel)
337 {
338 curve_channel::keys_t keys = channels_[channel].get_keys();
339 const auto& before = previous[channel].get_keys();
340 for(curve_key& key : keys)
341 {
342 const auto match = std::find_if(before.begin(),
343 before.end(),
344 [&](const curve_key& old)
345 {
346 return std::abs(old.position - key.position) <= same_position;
347 });
348 if(match == before.end())
349 {
350 continue;
351 }
352 key.interpolation = match->interpolation;
353 key.tangent_mode = match->tangent_mode;
354 key.in_tangent = match->in_tangent;
355 key.out_tangent = match->out_tangent;
356 }
357 channels_[channel].set_keys(std::move(keys));
358 channels_[channel].refresh_automatic_tangents();
359 }
360 mark_bake_dirty();
