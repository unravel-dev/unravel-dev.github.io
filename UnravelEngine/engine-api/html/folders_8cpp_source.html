34auto is_android_package_folder(const fs::path& folder) -> bool
35{
36 fs::error_code ec;
37 bool has_package = false;
38 bool has_earlier_data = false;
39 for(const auto& entry : fs::directory_iterator(folder, ec))
40 {
41 const auto& path = entry.path();
42 if(entry.is_regular_file(ec) && string_utils::to_lower(path.extension().generic_string()) == ".apk")
43 {
44 has_package = true;
45 }
46 else if(entry.is_directory(ec) && path.filename() == DATA_FOLDER && is_previous_deploy(folder))
47 {
48 has_earlier_data = true;
49 }
50 else
51 {
52 return false;
53 }
54 }
55 // data/ alone is a desktop build without its player, not an Android deploy's.
56 return !ec && (has_package || !has_earlier_data);
57}

59auto is_inside(const fs::path& path, const fs::path& folder) -> bool
60{
61 fs::error_code ec;
62 const auto canonical_folder = fs::weakly_canonical(folder, ec);
63 const auto canonical_path = fs::weakly_canonical(path, ec);
64 if(canonical_folder.empty() || canonical_path.empty())
65 {
66 return false;
67 }
68 const auto folder_key = fs::make_path_key(canonical_folder.generic_string());
69 const auto path_key = fs::make_path_key(canonical_path.generic_string());
70 return path_key.compare(0, folder_key.size(), folder_key) == 0 &&
71 (path_key.size() == folder_key.size() || path_key[folder_key.size()] == '/' || folder_key.back() == '/');
72}
