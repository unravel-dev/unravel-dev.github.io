16auto evaluate_segment(const curve_key& start, const curve_key& end, float position) -> float
17{
18 const float width = end.position - start.position;
19 if(width <= CURVE_MIN_SEGMENT_WIDTH)
20 {
21 return end.value;
22 }
23 const float alpha = (position - start.position) / width;
24 switch(start.interpolation)
25 {
26 case curve_interpolation::constant:
27 return start.value;
28 case curve_interpolation::linear:
29 return start.value + (end.value - start.value) * alpha;
30 case curve_interpolation::smooth:
31 {
32 // Cubic Hermite. The tangents are slopes in value per unit of position, so they are
33 // scaled by the segment's width to become the basis' tangent vectors.
34 const float alpha2 = alpha * alpha;
35 const float alpha3 = alpha2 * alpha;
36 const float h00 = 2.0f * alpha3 - 3.0f * alpha2 + 1.0f;
37 const float h10 = alpha3 - 2.0f * alpha2 + alpha;
38 const float h01 = -2.0f * alpha3 + 3.0f * alpha2;
39 const float h11 = alpha3 - alpha2;
40 return h00 * start.value + h10 * width * start.out_tangent + h01 * end.value +
41 h11 * width * end.in_tangent;
42 }
43 }
44 return start.value;
45}

103auto curve_channel::add_key(float position, float value) -> size_t
104{
105 curve_key key;
106 key.position = position;
107 key.value = value;
108 const auto at = std::upper_bound(keys_.begin(), keys_.end(), key);
109 const auto index = size_t(std::distance(keys_.begin(), at));
110 // A key dropped into a curve takes the shape of the segment it lands in, so adding one to a
111 // smooth curve does not leave a linear notch in it.
112 if(!keys_.empty())
113 {
114 const curve_key& neighbour = index > 0 ? keys_[index - 1] : keys_.front();
115 key.interpolation = neighbour.interpolation;
116 key.tangent_mode = neighbour.tangent_mode;
117 }
118 keys_.insert(at, key);
119 return index;

194auto curve_channel::sample(float position) const -> float
195{
196 if(keys_.empty())
197 {
198 return 0.0f;
199 }
200 // Past either end the end value holds rather than extrapolating, so a curve authored over a
201 // lifetime stays bounded outside it.
202 if(position <= keys_.front().position)
203 {
204 return keys_.front().value;
205 }
206 if(position >= keys_.back().position)
207 {
208 return keys_.back().value;
209 }
210
211 const curve_key probe{position, 0.0f, 0.0f, 0.0f, curve_interpolation::linear, curve_tangent_mode::automatic};
212 const auto upper = std::upper_bound(keys_.begin(), keys_.end(), probe);
213 const auto lower = std::prev(upper);
214 return evaluate_segment(*lower, *upper, position);
215}

217void curve_channel::refresh_automatic_tangents()
218{
219 const size_t count = keys_.size();
220 for(size_t index = 0; index < count; ++index)
221 {
222 curve_key& key = keys_[index];
223 if(key.tangent_mode == curve_tangent_mode::free)
224 {
225 continue;
226 }
227 if(key.tangent_mode == curve_tangent_mode::flat)
228 {
229 key.in_tangent = 0.0f;
230 key.out_tangent = 0.0f;
231 continue;
232 }
233 // A key with a neighbour either side takes the slope through them, the Catmull-Rom
234 // tangent, so the curve flows through it instead of easing at every key.
235 //
236 // The ends go level. Taking the slope of their one segment there would reproduce that
237 // segment exactly, so two smooth keys would draw the straight line they were asked to
238 // round off; level ends give the eased shape that is the point of asking for smooth, and
239 // they keep the curve from overshooting past the outermost key.
240 const bool is_end = index == 0 || index + 1 == count;
241 float slope = 0.0f;
242 if(!is_end)
243 {
244 const curve_key& previous = keys_[index - 1];
245 const curve_key& next = keys_[index + 1];
246 const float width = next.position - previous.position;
247 slope = width > CURVE_MIN_SEGMENT_WIDTH ? (next.value - previous.value) / width : 0.0f;
248 }
249 key.in_tangent = slope;
250 key.out_tangent = slope;
251 }
252}

254void curve_channel::reverse(float domain_min, float domain_max)
255{
256 const float sum = domain_min + domain_max;
257 for(auto& key : keys_)
258 {
259 key.position = sum - key.position;
260 // Mirroring the shape mirrors the slopes too, and swaps which side each one is on.
261 const float in_tangent = key.in_tangent;
262 key.in_tangent = -key.out_tangent;
263 key.out_tangent = -in_tangent;
264 }
265 // Mirroring the positions put the keys in descending order; this restores ascending order
266 // without re-sorting, which would have to break ties arbitrarily.
267 std::reverse(keys_.begin(), keys_.end());
268 // A key's interpolation governs the segment leaving it. Reversed, the segment that used to
269 // leave a key now arrives at it, so each key takes the interpolation of the one after it.
270 for(size_t index = 0; index + 1 < keys_.size(); ++index)
271 {
272 keys_[index].interpolation = keys_[index + 1].interpolation;
273 }
274}
