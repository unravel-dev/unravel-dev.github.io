39// material → up to six texture maps.
40//
41// Only `pbr_material` carries texture slots today; the base `material` class
42// is abstract / has no maps. We dynamic-cast rather than make this virtual to
43// keep the dep-graph concern out of the material interface.
44//------------------------------------------------------------------------------
45auto get_referenced_uids(const material& m) -> std::vector<hpp::uuid>
46{
47 std::vector<hpp::uuid> result;
48 if(m.is<pbr_material>())
49 {
50 const auto& pbr = static_cast<const pbr_material&>(m);
51
52 result.reserve(6);
53 push_if_set(result, pbr.get_color_map());
54 push_if_set(result, pbr.get_normal_map());
55 push_if_set(result, pbr.get_roughness_map());
56 push_if_set(result, pbr.get_metalness_map());
57 push_if_set(result, pbr.get_ao_map());
58 push_if_set(result, pbr.get_emissive_map());

88//------------------------------------------------------------------------------
89// Depth-first walk of a parsed prefab buffer, collecting `prefab_component`'s
90// `source.uid` wherever one appears.
91//
92// Matches on the component's serialized name rather than on the shape of the
93// value, because several components serialize a `source`/`uid` pair and only
94// this one denotes a nested prefab instance.
95//------------------------------------------------------------------------------
96void collect_prefab_sources(const simdjson::dom::element& value, std::vector<hpp::uuid>& out)
97{
98 if(value.is_array())
99 {
100 for(auto element : value.get_array())
101 {
102 collect_prefab_sources(element, out);
103 }
104 return;
105 }
106
107 if(!value.is_object())
108 {
109 return;
110 }
111
112 for(auto field : value.get_object())
113 {
114 if(field.key == "prefab_component")
115 {
116 hpp::string_view uid_str;
117 if(!field.value["source"]["uid"].get_string().get(uid_str))
118 {
119 const auto uid = hpp::uuid::from_string(uid_str);
120 if(uid.has_value() && !uid->is_nil())
121 {
122 out.push_back(*uid);
123 }
124 }
125 }
126
127 // Descended into regardless: a prefab_component's own subtree holds nothing
128 // further, but instances nest, so the rest of the document still has to be walked.
129 collect_prefab_sources(field.value, out);
130 }
131}

135//------------------------------------------------------------------------------
136// prefab -> the prefabs instanced inside it. See the header for why this reads
137// the buffer rather than deserializing it.
138//------------------------------------------------------------------------------
139auto get_referenced_uids(const prefab& p) -> std::vector<hpp::uuid>
140{
141 std::vector<hpp::uuid> result;
142
143 const auto& buffer = p.buffer.data;
144 if(buffer.empty())
145 {
146 return result;
147 }
148
149 simdjson::dom::parser parser;
150 simdjson::dom::element document;
151 const auto error = parser.parse(buffer.data(), buffer.size()).get(document);
152 if(error)
153 {
154 // A buffer that will not parse cannot be loaded either; whatever surfaces that
155 // is the right place to report it, not a dependency query.
156 return result;
157 }
158
159 collect_prefab_sources(document, result);
160
161 // Document order, first occurrence wins. One prefab may be instanced many times.
162 std::vector<hpp::uuid> unique;
163 unique.reserve(result.size());
164 for(const auto& uid : result)
165 {
166 if(std::find(unique.begin(), unique.end(), uid) == unique.end())
167 {
168 unique.push_back(uid);
169 }
170 }
171 return unique;
172}

174//------------------------------------------------------------------------------
175// Walks the loaded set of `T` and appends the UUID of every asset whose
176// reference list contains `target`. Skips assets that aren't currently
177// resident (`peek()` returns nullptr) — they have no in-memory thumbnail to
178// invalidate, and will be re-rendered fresh on the next request.
179//
180// Lives at file scope (not as a lambda) so we can inline-template it cleanly
181// over multiple types without copy-pasting the body.
182//------------------------------------------------------------------------------
183template<typename T>
184void collect_dependents(asset_manager& am,
185 const hpp::uuid& target,
186 std::vector<hpp::uuid>& out)
187{
188 am.for_each_asset<T>(
189 [&](const auto& kvp) -> void
190 {
191 const auto& handle = kvp.second;
192 if(handle.uid().is_nil())
193 {
194 return;
195 }
196
197 auto loaded = handle.peek();
198 if(!loaded)
199 {
200 return;
201 }
202
203 const auto deps = get_referenced_uids(*loaded);
204 if(std::find(deps.begin(), deps.end(), target) != deps.end())
205 {
206 out.push_back(handle.uid());
207 }
208 });
209}

226auto find_transitive_loaded_dependents(asset_manager& am, const hpp::uuid& root)
227 -> std::vector<hpp::uuid>
228{
229 std::vector<hpp::uuid> result;
230 if(root.is_nil())
231 {
232 return result;
233 }
234
235 std::unordered_set<hpp::uuid> visited;
236 std::deque<hpp::uuid> queue;
237 visited.insert(root);
238 queue.push_back(root);
239
240 while(!queue.empty())
241 {
242 const auto current = queue.front();
243 queue.pop_front();
244
245 for(const auto& dep : find_loaded_dependents(am, current))
246 {
247 if(visited.insert(dep).second)
248 {
249 result.push_back(dep);
250 queue.push_back(dep);
251 }
252 }
253 }
254
255 return result;
256}

258//------------------------------------------------------------------------------
259// Kahn's algorithm. Chosen over a DFS post-order because what is left over when
260// it stalls *is* the answer for cycles - no separate colouring pass, and the
261// leftovers naturally include assets that merely depend on a cycle, which are
262// just as unbuildable as its members.
263//------------------------------------------------------------------------------
264auto compute_build_order(const std::vector<hpp::uuid>& roots,
265 const dependency_resolver& resolve_dependencies) -> build_order
266{
267 build_order result;
268 if(!resolve_dependencies)
269 {
270 return result;
271 }
272
273 // Discovery. Kept in a vector as well as a set so the output is a function of
274 // input order rather than of hash iteration order.
275 std::vector<hpp::uuid> nodes;
276 std::unordered_set<hpp::uuid> known;
277 std::unordered_map<hpp::uuid, std::vector<hpp::uuid>> dependencies;
278
279 std::deque<hpp::uuid> pending_discovery(roots.begin(), roots.end());
280 while(!pending_discovery.empty())
281 {
282 const auto uid = pending_discovery.front();
283 pending_discovery.pop_front();
284
285 if(uid.is_nil() || !known.insert(uid).second)
286 {
287 continue;
288 }
289 nodes.push_back(uid);
290
291 // Deduplicated here rather than trusted from the resolver: a repeated
292 // dependency would otherwise be counted twice and the node would never reach
293 // zero, reporting a cycle that does not exist.
294 std::vector<hpp::uuid> deps;
295 for(const auto& dep : resolve_dependencies(uid))
296 {
297 if(dep.is_nil() || std::find(deps.begin(), deps.end(), dep) != deps.end())
298 {
299 continue;
300 }
301 deps.push_back(dep);
302 pending_discovery.push_back(dep);
303 }
304 dependencies.emplace(uid, std::move(deps));
305 }
306
307 // Unresolved-dependency counts, and the reverse edges used to decrement them.
308 std::unordered_map<hpp::uuid, size_t> unresolved;
309 std::unordered_map<hpp::uuid, std::vector<hpp::uuid>> dependents;
310 for(const auto& uid : nodes)
311 {
312 const auto& deps = dependencies[uid];
313 unresolved[uid] = deps.size();
314 for(const auto& dep : deps)
315 {
316 dependents[dep].push_back(uid);
317 }
318 }
319
320 std::vector<hpp::uuid> ready;
321 for(const auto& uid : nodes)
322 {
323 if(unresolved[uid] == 0)
324 {
325 ready.push_back(uid);
326 }
327 }
328
329 std::unordered_set<hpp::uuid> emitted;
330 for(size_t head = 0; head < ready.size(); ++head)
331 {
332 const auto uid = ready[head];
333 result.ordered.push_back(uid);
334 emitted.insert(uid);
335
336 const auto it = dependents.find(uid);
337 if(it == dependents.end())
338 {
339 continue;
340 }
341 for(const auto& dependent : it->second)
342 {
343 if(--unresolved[dependent] == 0)
344 {
345 ready.push_back(dependent);
346 }
347 }
348 }
349
350 // Whatever never reached zero is in a cycle or downstream of one.
351 for(const auto& uid : nodes)
352 {
353 if(emitted.count(uid) == 0)
354 {
355 result.cyclic.push_back(uid);
356 }
357 }
358
359 return result;
360}

383// Reads one asset's nested-instance references, forcing a load. Returns false if
384// the uid does not name an asset of this type, so the caller can try the other.
385//------------------------------------------------------------------------------
386template<typename T>
387auto try_read_references(asset_manager& am, const hpp::uuid& uid, std::vector<hpp::uuid>& out) -> bool
388{
389 auto handle = am.get_asset<T>(uid);
390 if(!handle.is_valid())
391 {
392 return false;
393 }
394
395 auto asset = handle.get();
396 if(!asset || asset->buffer.data.empty())
397 {
398 return false;
399 }
400
401 out = get_referenced_uids(*asset);
402 return true;
403}
