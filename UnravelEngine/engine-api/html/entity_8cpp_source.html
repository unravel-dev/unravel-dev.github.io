185auto instance_path_between(entt::handle outer_root, entt::handle inner_root, std::vector<hpp::uuid>& out) -> bool
186{
187 out.clear();
188 if(!outer_root || !inner_root)
189 {
190 return false;
191 }
192 if(outer_root == inner_root)
193 {
194 return true;
195 }
196
197 std::vector<hpp::uuid> reversed;
198 auto current = inner_root;
199 while(current && current != outer_root)
200 {
201 if(const auto* prefab_comp = current.try_get<prefab_component>())
202 {
203 if(prefab_comp->instance_id.is_nil())
204 {
205 return false;
206 }
207 reversed.push_back(prefab_comp->instance_id);
208 }
209 const auto* trans_comp = current.try_get<transform_component>();
210 current = trans_comp != nullptr ? trans_comp->get_parent() : entt::handle{};
211 }
212 if(current != outer_root)
213 {
214 return false;
215 }
216
217 out.assign(reversed.rbegin(), reversed.rend());
218 return true;
219}

456auto parse_script_position(hpp::string_view component_path, size_t& position, size_t& rest) -> bool
457{
458 if(!component_path.starts_with(POSITIONAL_SCRIPT_PREFIX))
459 {
460 return false;
461 }
462 const auto digits_begin = component_path.data() + POSITIONAL_SCRIPT_PREFIX.size();
463 const auto digits_end = component_path.data() + component_path.size();
464 const auto [last, error] = std::from_chars(digits_begin, digits_end, position);
465 if(error != std::errc{} || last == digits_begin || last == digits_end || *last != ']')
466 {
467 return false;
468 }
469 rest = static_cast<size_t>(last - component_path.data()) + 1;
470 return true;
471}

738void cleanup_uid_mapping()
739{
740 // Unconsumed load stubs. They were never visible to gameplay, so they must not
741 // produce pre-destroy notifications on the way out.
742 scene::scoped_destroy_suppression no_pre_destroy;
743
744 auto& load_ctx = get_load_context();
745 auto* frame = load_ctx.current_instance();
746 if(frame == nullptr)
747 {
748 return;
749 }
750
751 for(auto& [uid, mapping] : frame->mapping_by_prefab_uid)
752 {
753 if(mapping.consumed() || !mapping.handle)
754 {
755 continue;
756 }
757
758 // Unmentioned is a deletion only for this document's own content. An entity another
759 // document introduced here - an outer one adding under this instance - carries that
760 // document's name, and one the user made carries none; neither is this document's to
761 // remove.
762 const auto* id_comp = mapping.handle.try_get<prefab_id_component>();
763 if(id_comp != nullptr && id_comp->document != load_ctx.document_uid)
764 {
765 continue;
766 }
767
768 scene::destroy_entity(mapping.handle);
769 }
770
771 // Named by the document: decided exactly, one instance at a time. No record addressed it
772 // means the author removed that instance. Ordered by path, so an outer instance goes before
773 // anything nested inside it and the inner handles are already invalid by the time they come
774 // up.
775 for(auto& [path, scope] : frame->nested_scopes)
776 {
777 if(scope.consumed || !scope.root)
778 {
779 continue;
780 }
781
782 // Unmentioned is only a deletion for a slot this document placed. An instance an outer
783 // document added here is named by that document, one the user added is not named at
784 // all - and neither is this document's to remove. The slot says whose it is.
785 const auto* nested_prefab = scope.root.try_get<prefab_component>();
786 if(nested_prefab == nullptr || nested_prefab->instance_document != load_ctx.document_uid)
787 {
788 continue;
789 }
790
791 scene::destroy_entity(scope.root);
792 }

862 return;
863 }
864
865 auto* save_ctx = try_get_save_context();
866 const hpp::uuid document_uid = save_ctx != nullptr ? save_ctx->document_uid : hpp::uuid{};
867
868 const auto settle_entity_identity = [&](entt::handle entity)
869 {
870 auto& id_comp = entity.get_or_emplace<prefab_id_component>();
871 id_comp.generate_if_nil();
872 const bool supplied_by_nested =
873 std::find(document_chain.begin(), document_chain.end(), id_comp.document) != document_chain.end();
874 if(!supplied_by_nested && !document_uid.is_nil())
875 {
876 id_comp.document = document_uid;
877 }
878 };
879
880 if(is_subtree_root)
881 {
882 // The document's own root, whether a plain entity or the instance being re-saved as
883 // its own document.
884 settle_entity_identity(obj);
885 }
886 else if(auto* prefab_comp = obj.try_get<prefab_component>())
887 {
888 // A slot is named by the document that placed it, and an unnamed instance anywhere
889 // under the save root was placed by this document: the nested assets name their own
890 // slots at their own save, so anything still unnamed here is this document's addition.
891 if(prefab_comp->instance_id.is_nil())
892 {
893 prefab_comp->instance_id = generate_uuid();
894 prefab_comp->instance_document = document_uid;
895 }
896
897 enclosing_instance = obj;
898 document_chain.push_back(prefab_comp->source.uid());
899 }
900 else
901 {
902 settle_entity_identity(obj);
903 }
904
905 if(auto* trans_comp = obj.try_get<transform_component>())
906 {
907 for(auto child : trans_comp->get_children())
908 {
909 prepare_prefab_identity(child, enclosing_instance, false, document_chain);
910 }
911 }
912}

914auto is_parent(entt::const_handle potential_parent, entt::const_handle child) -> bool
915{
916 if(!potential_parent)
917 {
918 return false;
919 }
920 // Traverse up the hierarchy from `child`
921 while(true)
922 {
923 // Access the transform component once per entity
924 const auto* transform = child.try_get<transform_component>();
925 if(!transform)
926 {
927 return false; // Reached the root without finding `potential_parent`
928 }
929 auto parent = transform->get_parent();
930 if(!parent)
931 {
932 return false;
933 }
934
935 if(parent == potential_parent)
936 {
937 return true; // Found the parent relationship
938 }
939
940 child = parent; // Move up the hierarchy
941 }
942}
943auto find_root(entt::const_handle e) -> entt::const_handle
944{
945 // Loop to find the root entity
946 while(true)
947 {
948 // Access the `transform_component` once
949 const auto* transform = e.try_get<transform_component>();
950 if(!transform || !transform->get_parent())
951 {
952 break; // If no parent, we are at the root
953 }
954 e = transform->get_parent(); // Move to the parent entity
955 }
956 return e; // Root entity
957}
958
959auto are_related(entt::const_handle lhs, entt::const_handle rhs) -> bool
960{
961 return find_root(lhs) == find_root(rhs);
962}

1238template<typename Archive>
1239auto load_entity_from_prefab_uid(Archive& ar, entt::handle& obj, entity_flags flags) -> bool
1240{
1241 hpp::uuid uid;
1242 try_load(ar, ser20::make_nvp("prefab_uid", uid));
1243
1244
1245 // Resolved against the innermost instance only. A prefab uid identifies an entity
1246 // within one prefab asset, so it is meaningful relative to the instance being loaded
1247 // and nothing wider.
1248 auto& load_ctx = get_load_context();
1249 auto* frame = load_ctx.current_instance();
1250 if(frame == nullptr)
1251 {
1252 return false;
1253 }
1254
1255 // Nothing nested here, so nothing can be scoped to an instance: the two lookups below
1256 // would miss on every record. Deleted nested instances count as nested state even though
1257 // nothing of them is live - their records still carry instance ids, and skipping the
1258 // instance-id reads here would skip the removed-path check with them, recreating an
1259 // instance whose only remaining trace is the removal entry.
1260 if(frame->nested_scopes.empty() && frame->removed_instance_paths.empty())
1261 {
1262 auto plain_it = frame->mapping_by_prefab_uid.find(uid);
1263 if(plain_it == frame->mapping_by_prefab_uid.end())
1264 {
1265 return false;
1266 }
1267 return resolve_plain_prefab_uid(load_ctx, plain_it->second, obj);
1268 }
1269
1270 // A nested instance names itself, so it can be matched exactly - which prefab uids
1271 // cannot do, since a duplicate or a copy carries the same one.
1272 hpp::uuid instance_uid;
1273 try_load(ar, ser20::make_nvp("instance_uid", instance_uid));
1274
1275 // And an entity inside one says which, so the document can reach into a specific
1276 // instance instead of every instance of that prefab at once.
1277 std::vector<hpp::uuid> instance_path;
1278 try_load(ar, ser20::make_nvp("instance_path", instance_path));
1279
1280 if(!instance_uid.is_nil())
1281 {
1282 auto scope_path = instance_path;
1283 scope_path.push_back(instance_uid);
1284
1285 if(is_removed_instance_path(*frame, scope_path))
1286 {
1287 // Deleted here. A hit with an empty handle, so the record is skipped instead of
1288 // creating the instance again.
1289 obj = {};
1290 return true;
1291 }
1292
1293 auto scope_it = frame->nested_scopes.find(scope_path);
1294 if(scope_it == frame->nested_scopes.end())
1295 {
1296 // The document has a nested instance this one does not: the author added it,
1297 // most often by duplicating. Refuse the match so the caller falls through to the
1298 // paths that create it.
1299 return false;
1300 }
1301
1302 if(load_ctx.resolving_record)
1303 {
1304 scope_it->second.consumed = true;
1305 }
1306
1307 obj = scope_it->second.root;
1308
1309 // Filtered against its own authoring, like its contents: what the document says
1310 // about this instance is its override set and its link, not the placement or name
1311 // the instance has been given here. Its prefab_component is let through regardless -
1312 // it is that override set, and it loads before anything it has to filter.
1313 load_ctx.pending_nested_owner = scope_it->second.root;
1314 return true;
1315 }
1316
1317 if(!instance_path.empty())
1318 {
1319 if(is_removed_instance_path(*frame, instance_path))
1320 {
1321 obj = {};
1322 return true;
1323 }
1324
1325 auto scope_it = frame->nested_scopes.find(instance_path);
1326 if(scope_it == frame->nested_scopes.end())
1327 {
1328 // Content of an instance that is not here yet - it arrives with the instance.
1329 return false;
1330 }
1331
1332 auto entity_it = scope_it->second.by_prefab_uid.find(uid);
1333 if(entity_it == scope_it->second.by_prefab_uid.end())
1334 {
1335 // The document knows an entity inside this instance that the instance does not:
1336 // one added under it. Created like any other addition.
1337 return false;
1338 }
1339
1340 // May be null, for an entity the instance removed - a hit with an empty handle, so
1341 // the record is skipped rather than bringing it back.
1342 obj = entity_it->second;
1343
1344 // Content this document introduced inside the nested instance - an entity the
1345 // containing document added under it - is the document's own, not a statement about
1346 // the nested asset's content. The filter treats its record as content.
1347 if(obj && load_ctx.resolving_record)
1348 {
1349 const auto* id_comp = obj.try_get<prefab_id_component>();
1350 load_ctx.pending_record_is_own_content = id_comp != nullptr && !id_comp->document.is_nil() &&
1351 id_comp->document == load_ctx.document_uid;
1352 }
1353
1354 // Which instance's authoring this record belongs to, so the filter can let through
1355 // what that instance overrides and hold back the rest of the snapshot.
1356 load_ctx.pending_nested_owner = scope_it->second.root;
1357 return true;
1358 }
1359
1360 auto it = frame->mapping_by_prefab_uid.find(uid);
1361 if(it != frame->mapping_by_prefab_uid.end())
1362 {
1363 return resolve_plain_prefab_uid(load_ctx, it->second, obj);
1364 }
1365
1366 return false;
1367}

1437template<typename Component>
1438auto should_save_component(const entt::const_handle& obj) -> bool
1439{
1440 if constexpr(std::is_same_v<Component, prefab_component>)
1441 {
1442 auto& save_ctx = get_save_context();
1443
1444 // Dropped for the entity this file *defines*. Its prefab_component says it is an
1445 // instance of some other asset, and baking that into the asset being written would
1446 // declare this file a variant of that one - a different feature, and not what a
1447 // "save as prefab" means.
1448 //
1449 // Kept for every descendant. A prefab instance nested inside the one being saved is
1450 // a link to another asset, and dropping it is what flattened nested prefabs into
1451 // loose entities: the inner asset's edits stopped reaching them, permanently and
1452 // silently (audit B4).
1453 if(save_ctx.is_saving_to_prefab() && obj == save_ctx.save_source)
1454 {
1455 return false;
1456 }
1457 }
1458 else if constexpr(std::is_same_v<Component, prefab_id_component>)
1459 {
1460 // if we are cloning
1461 // we need to generate a new id for the entity
1462 auto& save_ctx = get_save_context();
1463 if(save_ctx.is_cloning())
1464 {
1465 if(save_ctx.get_clone_mode() != clone_mode_t::cloning_prefab_instance)
1466 {
1467 return false;
1468 }
1469 // A clone keeps prefab ids only at and below an instance root inside it. A plain
1470 // branch that merely contains an instance is content of whatever container the
1471 // branch sits in; a copy of that part is the container's addition, not its content
1472 // twice over - and two entities under one id would leave the container matching
1473 // whichever it met last.
1474 auto current = obj;
1475 bool inside_instance = false;
1476 while(current)
1477 {
1478 if(current.all_of<prefab_component>())
1479 {
1480 inside_instance = true;
1481 break;
1482 }
1483 if(current == save_ctx.save_source)
1484 {
1485 break;
1486 }
1487 const auto* trans_comp = current.try_get<transform_component>();
1488 current = trans_comp != nullptr ? entt::const_handle(trans_comp->get_parent()) : entt::const_handle{};
1489 }
1490 return inside_instance;
1491 }
1492 }
1493 return true;
1494}

1921

1923 try_load(ar, ser20::make_nvp("version", version));

2746auto sync_prefab_instance(entt::handle instance) -> bool
2747{

2749 {

2751 }
2752
2753 auto* prefab_comp = instance.try_get<prefab_component>();
2754 auto* trans_comp = instance.try_get<transform_component>();
2755 if(prefab_comp == nullptr || trans_comp == nullptr || !prefab_comp->source)
2756 {
2757 return false;
2758 }
2759
2760 const auto source = prefab_comp->source;
2761
2762 auto parent = trans_comp->get_parent();

2765
2766 // Copied, not referenced. prefab_component sets in_place_delete to false, so its pool
2767 // swaps elements on removal - and the load below both adds and removes prefab_components
2768 // while this callback is live.
2769 // What the replay must leave alone, relative to this instance and at every depth below it:
2770 // what was stated here (this root's local list, the local lists of the named instances
2771 // nested in it, an authoring root's adopted list above) and what the documents above state.
2772 // What it lets through of the nested content is what *this* document states - read from the
2773 // document as it loads, or from the live list for a file written before documents carried
2774 // one. Nothing is attributed afterwards; each list has one author.
2775 const replay_statements protect = collect_replay_statements(instance);
2776 std::vector<entt::handle> nested_roots;
2777 collect_nested_instance_roots(instance, nested_roots, true);
2778 const bool needs_override_tracking = !nested_roots.empty() || !protect.local.empty() || !protect.stated.empty();
2779
2780 // Slot chains from this instance to each nested owner the filter meets, computed once.
2781 std::map<entt::entity, std::vector<hpp::uuid>> owner_paths;
2782
2783 // Statements the replay found stale: entity uid and component path, dropped after the load.
2784 std::vector<std::pair<hpp::uuid, std::string>> stale_statements;
2785
2786 // The components a statement keeps whole in this instance's own content - all a stale check
2787 // can concern. Usually none, and then the check is not installed: it would otherwise run for
2788 // every component an entity does not have.
2789 std::set<std::string, std::less<>> kept_whole_components;
2790 for(const auto* statements : {&protect.local, &protect.stated})
2791 {
2792 for(const auto& override_data : statements->overrides)
2793 {
2794 if(override_data.instance_path.empty() && override_data.component_path.find('/') == std::string::npos)
2795 {
2796 kept_whole_components.insert(override_data.component_path);
2797 }
2798 }
2799 }
2800
2801 serialization::path_context path_ctx;
2802 serialization::path_context* old_ctx = serialization::get_path_context();
2803 if(needs_override_tracking)
2804 {
2805 // The slot chain from this instance to a nested one, worked out once per replay.
2806 const auto owner_path_of = [&owner_paths, instance](entt::handle owner) -> const std::vector<hpp::uuid>*
2807 {
2808 if(const auto cached = owner_paths.find(owner.entity()); cached != owner_paths.end())
2809 {
2810 return &cached->second;
2811 }
2812 std::vector<hpp::uuid> chain;
2813 if(!instance_path_between(instance, owner, chain))
2814 {
2815 return nullptr;
2816 }
2817 return &owner_paths.emplace(owner.entity(), std::move(chain)).first->second;
2818 };
2819
2820 // What the replayed document states about the content it nests.
2821 const auto document_statements_of = [instance]() -> const prefab_statements*
2822 {
2823 auto& load_ctx = get_load_context();
2824 if(load_ctx.has_document_statements)
2825 {
2826 return &load_ctx.document_statements;
2827 }
2828 const auto* own = instance.try_get<prefab_component>();

2830 };

2833 [&protect, owner_path_of, document_statements_of](const std::string& property_path)
2834 {
2835 const auto parts = split_serialization_path(property_path);
2836 if(!parts.valid)
2837 {
2838 return true;
2839 }
2840 const auto component_path = replay_component_path(parts);
2841

2844 if(!owner)
2845 {
2846 // This instance's own content: what is stated here or above stays; the rest
2847 // is the document's to restate. Protection is on-or-above: an override on one
2848 // field must not shield its siblings, so the level above passes and each field
2849 // decides for itself.

2851 !protect.stated.has_override_on_or_above({}, parts.entity_uuid, component_path);
2852 }

2855 // prefab_component is let through regardless: link, slot and the document's copy
2856 // of that instance's own list, which loads before anything it has to filter.
2857 if(is_prefab_bookkeeping_component(component_path))
2858 {

2862 {

2866 if(owner_path == nullptr)

2869 }

2874 {

2876 !protect.stated.has_override_on_or_above(*owner_path, parts.entity_uuid, component_path);
2877 }
2878

2880 // On-or-above, for the same reason as above: a local position override on a nested
2881 // root must not keep the document's scale out of the same transform.
2882 if(protect.local.has_override_on_or_above(*owner_path, parts.entity_uuid, component_path) ||
2883 protect.stated.has_override_on_or_above(*owner_path, parts.entity_uuid, component_path))
2884 {

2889 const auto* owner_id = owner.try_get<prefab_id_component>();
2890 if(owner_id != nullptr && owner_id->id == parts.entity_uuid &&
2891 is_nested_root_placement_path(component_path))
2892 {
2893 return true;
2894 }
2895
2896 const auto* document = document_statements_of();
2897 return document != nullptr &&
2898 document->has_override_touching(*owner_path, parts.entity_uuid, component_path);
2899 };
2900

2902 // that keeps a whole component of it - the check for one gone stale.
2903 if(!kept_whole_components.empty())
2904 {

2911 if(component.starts_with("has_"))
2912 {
2913 component.remove_prefix(4);
2914 }

2916 {
2917 return false;
2918 }

2920 if(!parts.valid || get_load_context().current_nested_owner)
2921 {
2922 return false;
2923 }

2926 protect.stated.has_override({}, parts.entity_uuid, component_path);

2948 return false;

3517{

3522
3523 // APPLOG_INFO_PERF(std::chrono::microseconds);
3524
3525 // Every root goes through a throwaway buffer on its way across. Indenting those is
3526 // pure cost: this is the edit-scene -> play-scene path, so it runs on every play press.
3527 serialization::scoped_output_format compact(serialization::output_format::compact);
3528
3529 // One load context for the whole scene, not one per root.
3530 //
3531 // The roots are serialized separately, but an entity link that crosses from one root
3532 // into another is written as the target's uid and resolved through the context's uid
3533 // map. With a context per root that map only ever held the root being loaded, so a
3534 // cross-root link found nothing, fell through to "create a new entity", and the link
3535 // ended up aimed at an empty entity nothing else referenced - silently, on every play
3536 // press.

3538 // Sharing the context turns the same path into a forward reference: the first mention
3539 // creates the entity and records its uid, and the record that actually defines it -
3540 // loaded with whichever root owns it - resolves to that same entity and fills it in.
3541 // That is already how a parent's "children" list works within a single root; it just
3542 // never spanned roots before.

3544 // load_from_archive_start still pushes per root, but push_load_context returns false
3545 // when one is already active and the matching pop is then a no-op, so the inner calls

3553 [&](auto e, auto&& comp1, auto&& comp2)
3554 {
3555 std::stringstream ss;
3556 save_to_stream(ss, src_scene.create_handle(e));
