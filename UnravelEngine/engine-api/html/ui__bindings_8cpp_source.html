166template<typename T>
167void dispatch_ui_event_to_manager(const T& event_data)
168{
169 try
170 {
171 const auto& ctx = engine::context();
172 const auto& script_cache = ctx.get_cached<script_system>().get_cache();
173
174 if(!script_cache.ui_dispatch_event_method.valid())
175 {
176 APPLOG_ERROR("UIEventManager.InternalDispatchEvent method not found");
177 return;
178 }
179
180 auto method_invoker =
181 dotnet::make_method_invoker<void(const T&)>(script_cache.ui_dispatch_event_method, false);
182 method_invoker(event_data);
183 }
184 catch (const dotnet::exception& e)
185 {
186 APPLOG_ERROR("C# exception dispatching UI event: {}", e.what());
187 }
188 catch (const std::exception& e)
189 {
190 APPLOG_ERROR("Error dispatching UI event: {}", e.what());
191 }
192}

196// UI Event Type Classification
197enum class ui_event_type
198{
199 unknown,
200 key,
201 textinput,
202 pointer,
203 change,
204 value
205};
206
207// Determine UI event type for efficient dispatch
208auto get_ui_event_type(const Rml::Event& event) -> ui_event_type
209{
210 const auto event_id = event.GetId();
211
212 // Check key events first (most common check)
213 if (event_id == Rml::EventId::Keydown || event_id == Rml::EventId::Keyup)
214 {
215 return ui_event_type::key;
216 }
217
218 // Check text input events
219 if (event_id == Rml::EventId::Textinput)
220 {
221 return ui_event_type::textinput;
222 }
223
224 // Check pointer events
225 if (event_id == Rml::EventId::Click || event_id == Rml::EventId::Mousedown || event_id == Rml::EventId::Mouseup ||
226 event_id == Rml::EventId::Mousemove || event_id == Rml::EventId::Mouseover || event_id == Rml::EventId::Mouseout ||
227 event_id == Rml::EventId::Mousescroll || event_id == Rml::EventId::Dblclick || event_id == Rml::EventId::Drag ||
228 event_id == Rml::EventId::Dragstart || event_id == Rml::EventId::Dragover || event_id == Rml::EventId::Dragdrop)
229 {
230 return ui_event_type::pointer;
231 }
232
233 // Check change events (need to examine the event more closely)
234 if (event_id == Rml::EventId::Change)
235 {
236 auto value_str = event.GetParameter<std::string>("value", "");
237 if (!value_str.empty())
238 {
239 // If the element doesn't have min/max attributes, it's likely a text input or similar (change event)
240 if (auto* element = event.GetCurrentElement())
241 {
242 if (!element->HasAttribute("min") && !element->HasAttribute("max"))
243 {
244 return ui_event_type::change;
245 }
246 else
247 {
248 // Has min/max attributes, likely a slider (value event)
249 return ui_event_type::value;
250 }
251 }
252 }
253 }
254
255 return ui_event_type::unknown;
256}

391// Global event listener that dispatches all UI events to C# UIEventManager
392class ui_global_event_listener : public Rml::EventListener
393{
394 Rml::Event* current_event_ = nullptr;
395public:
396 void ProcessEvent(Rml::Event& event) override
397 {
398 current_event_ = &event;
399 try
400 {
401 // Get event information
402 auto* target_element = event.GetTargetElement();
403 if (!target_element)
404 {
405 return;
406 }
407
408 auto* current_element = event.GetCurrentElement();
409 if (!current_element)
410 {
411 return;
412 }
413
414 // Determine event type and dispatch accordingly using efficient switch
415 const auto event_type = get_ui_event_type(event);
416
417 switch (event_type)
418 {
419 case ui_event_type::key:
420 dispatch_key_event_to_manager(event, target_element, current_element);
421 break;
422
423 case ui_event_type::textinput:
424 dispatch_textinput_event_to_manager(event, target_element, current_element);
425 break;
426
427 case ui_event_type::pointer:
428 dispatch_pointer_event_to_manager(event, target_element, current_element);
429 break;
430
431 case ui_event_type::change:
432 dispatch_change_event_to_manager(event, target_element, current_element);
433 break;
434
435 case ui_event_type::value:
436 dispatch_value_event_to_manager(event, target_element, current_element);
437 break;
438
439 case ui_event_type::unknown:
440 default:
441 // Fallback to base event for unknown types
442 dispatch_base_event_to_manager(event, target_element, current_element);
443 break;
444 }
445 }
446 catch (const std::exception& e)
447 {
448 APPLOG_ERROR("Error processing UI event: {}", e.what());
449 }
450 current_event_ = nullptr;
451 }
452
453 // Allow access to current event for propagation control
454 auto get_current_event() const -> Rml::Event*
455 {
456 return current_event_;
457 }
458};

551{

559 // Check if this element exists in the owner entity's UI document
