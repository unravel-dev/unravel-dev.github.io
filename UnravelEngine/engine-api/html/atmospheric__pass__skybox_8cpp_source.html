93

102
103 float sky_brightness_vec[4] = {params.sky_brightness, 0.0f, 0.0f, 0.0f};
104 gfx::set_uniform(program_.u_sky_brightness, sky_brightness_vec);
105 params.pre_exposure.bind();
106
107 // 4) The critical part: Use DEPTH_TEST_EQUAL, so we only fill the background
108 uint64_t state =
109 BGFX_STATE_WRITE_RGB | BGFX_STATE_WRITE_A |
110 BGFX_STATE_DEPTH_TEST_LEQUAL; // or CULL_NONE, depending on vertex winding
111
112 bgfx::setState(state);
113
114 // 5) Submit the draw
115 bgfx::setVertexBuffer(0, vb_->native_handle());
