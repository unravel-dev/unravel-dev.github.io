28auto extract_href_from_link_tag(hpp::string_view link_tag) -> hpp::string_view
29{
30 size_t href_pos = link_tag.find("href=");
31 if(href_pos == std::string::npos)
32 {
33 return {};
34 }
35 href_pos += 5;
36 char quote_char = link_tag[href_pos];
37 if(quote_char != '"' && quote_char != '\'')
38 {
39 return {};
40 }
41 href_pos++;
42 size_t href_end = link_tag.find(quote_char, href_pos);
43 if(href_end == std::string::npos)

45 return {};
46 }

105void resolve_shader_dependencies(const fs::path& file_path,
106 std::vector<fs::path>& processed_files,
107 std::unordered_set<std::string>& visited,
108 bool is_root)
109{
110 if(!visited.insert(visit_key_for_path(file_path)).second)
111 {
112 return;
113 }
114
115 // Root source is watched separately; only emit external deps here.
116 if(is_root)
117 {
118 append_shader_varying_dependency(file_path, processed_files);
119 }
120 else
121 {
122 processed_files.push_back(file_path);
123 }
124
125 std::ifstream file(file_path);
126 if(!file.is_open())
127 {
128 return;
129 }
130
131 const std::string include_keyword = "#include";
132 std::string line;
133 while(std::getline(file, line))
134 {
135 line.erase(0, line.find_first_not_of(" \t"));
136 if(line.compare(0, include_keyword.length(), include_keyword) != 0)
137 {
138 continue;
139 }
140 size_t start = line.find_first_of("\"<") + 1;
141 size_t end = line.find_last_of("\">");
142 if(start == std::string::npos || end == std::string::npos || start >= end)
143 {
144 continue;
145 }
146 if(line[start - 1] == '<' && line[end] == '>')
147 {
148 continue;
149 }
150 const std::string include_path = normalize_dependency_path_string(line.substr(start, end - start));
151 // Resolved against the including file's directory AND against the shared shader include
152 // root, because shaderc searches both. Relative-only resolution silently drops every
153 // root-relative include -- `#include "gi/foo.sh"` from inside `shaders/gi/` becomes
154 // `shaders/gi/gi/foo.sh`, which does not exist.
155 //
156 // A dropped dependency is not a build error, it is a STALE BINARY: editing the shared
157 // header leaves every shader that includes it compiled from the previous version, so
158 // writers and readers of a shared layout silently disagree. That fails as corrupt data
159 // at runtime, with nothing pointing back at the build.
160 fs::path resolved_path = fs::absolute(file_path.parent_path() / fs::path(include_path));
161 fs::error_code exists_err;
162 if(!fs::exists(resolved_path, exists_err) || exists_err)
163 {
164 const fs::path root_relative = fs::absolute(find_shader_include_root(file_path) / fs::path(include_path));
165 if(fs::exists(root_relative, exists_err) && !exists_err)
166 {
167 resolved_path = root_relative;
168 }
169 }
170 resolve_shader_dependencies(resolved_path, processed_files, visited, false);
171 }
