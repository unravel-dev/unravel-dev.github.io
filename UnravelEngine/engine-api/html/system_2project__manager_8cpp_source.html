145 {
146 // Load or create the project signature file.
147 //
148 // * No file on disk: legacy project (or freshly created). Stamp the
149 // current engine version in both `created` and `opened` and mint a
150 // fresh GUID. This is the backward-compatibility path - opening an
151 // older project simply writes a new `project.cfg` on the spot.
152 //
153 // * File present & engine_version_opened >= current engine: the
154 // project is up to date (or from a newer engine entirely, which we
155 // don't guard against in this path). Just update `opened` to the
156 // running engine version.

163 // Headless/CLI opens intentionally proceed so automation is not
164 // blocked by this.
165 const bool had_info_file = load_project_info();
166 const auto current = version::get_current();

168 if(!had_info_file)

170 project_info_.engine_version_created = current;
171 project_info_.project_guid = hpp::to_string(generate_uuid());

175 project_info_.project_guid);
176 }

179 APPLOG_WARNING("Project was last opened with an older engine ({} < {}). "
180 "Proceeding - data loss is possible if the on-disk "
181 "format has diverged.",
182 project_info_.engine_version_opened.to_string(),
183 current.to_string());
184 }

186 project_info_.engine_version_opened = current;
187 save_project_info();
188 }

192 save_project_settings(ctx);
193
194 ls.begin_module("Deploy Settings");

290auto project_manager::inspect_project(const fs::path& project_path) const -> project_compat_report
291{
292 project_compat_report report;
293
294 // We cannot rely on `app:/` here because the inspected project may not be
295 // the one currently open; resolve the on-disk path directly instead.
296 const fs::path info_path = project_path / "project.cfg";
297
298 fs::error_code ec;
299 if(!fs::exists(info_path, ec) || ec)
300 {
301 report.status = project_compat::no_info_file;
302 return report;
303 }
304
305 if(!load_from_file(info_path.string(), report.on_disk))
306 {
307 // The file exists but failed to parse. Treat as legacy so we'll
308 // rewrite it on open, but log - this is unusual and worth surfacing.
309 APPLOG_WARNING("project.cfg at {} exists but could not be parsed. "
310 "It will be regenerated on open.",
311 info_path.string());
312 report.status = project_compat::no_info_file;
313 report.on_disk = {};
314 return report;
315 }
316
317 if(is_from_older_engine(report.on_disk.engine_version_opened, version::get_current()))
318 {
319 report.status = project_compat::engine_older;
320 }
321 else
322 {
323 report.status = project_compat::ok;
324 }
325 return report;
326}

350void project_manager::fixup_editor_settings_on_save()
351{
352 // fixup recent_projects
353 if(has_open_project())
354 {
355 auto& rp = editor_settings_.projects.recent_projects;
356 auto project_path = fs::resolve_protocol("app:/");
357 if(std::find_if(std::begin(rp),
358 std::end(rp),
359 [&](const auto& prj)
360 {
361 return project_path.generic_string() == prj;
362 }) == std::end(rp))
363 {
364 rp.emplace_back(std::move(project_path));
365 }
366
367 std::sort(std::begin(rp),
368 std::end(rp),
369 [](const auto& lhs_path, const auto& rhs_path)
370 {
371 fs::error_code ec;
372 auto lhs_time = fs::last_write_time(lhs_path / app_deploy_file, ec);
373 auto rhs_time = fs::last_write_time(rhs_path / app_deploy_file, ec);
374
375 return lhs_time > rhs_time;
376 });
377 }
