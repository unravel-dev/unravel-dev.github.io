1/*
2 * The atmospheric pass renders a dynamic sky with the Perez sky model.
3 *
4 * Typically, the sky is rendered using cubemaps or other environment maps.
5 * This approach can provide a high-quality sky, but the downside is that the
6 * image is static. To achieve daytime changes in sky appearance, there is a need
7 * in a dynamic model.
8 *
9 * The Perez sky model is a simple, but good enough model which is, in essence,
10 * a function that interpolates a sky color. As input, it requires several
11 * turbidity coefficients, a color at zenith and direction to the sun. The
12 * turbidity coefficients are fitted to more complex physically based models.
13 * Color at zenith depends on daytime and can vary depending on many factors.
14 *
15 * Two tables (atmospheric_pass_perez.cpp) hold the sun's and the zenith sky's
16 * luminance by hour of the day. Luminance in those tables represents actual
17 * scale of light energy that comes from sun compared to the sky.
18 */

40// Sun position computation based on Earth's orbital elements: the sun's mean
41// longitude and the obliquity of the ecliptic (23.4 degrees) give its declination.

84private:
85 void calculate_sun_orbit()
86 {
87 const float day = 30.0f * float(month_) + 15.0f;
88 float lambda = 280.46f + 0.9856474f * day;
89 lambda = bx::toRad(lambda);
90 delta_ = bx::asin(bx::sin(ecliptic_obliquity_) * bx::sin(lambda));
91 }
92
93 void update_sun_position(float _hour)
94 {
95 const float latitude = bx::toRad(latitude_);
96 const float hh = _hour * bx::kPi / 12.0f;
97 const float azimuth =
98 bx::atan2(bx::sin(hh), bx::cos(hh) * bx::sin(latitude) - bx::tan(delta_) * bx::cos(latitude));
99
100 const float altitude =
101 bx::asin(bx::sin(latitude) * bx::sin(delta_) + bx::cos(latitude) * bx::cos(delta_) * bx::cos(hh));
102
103 const bx::Quaternion rot0 = bx::fromAxisAngle(up_dir_, -azimuth);
104 const bx::Vec3 dir = bx::mul(north_dir_, rot0);
105 const bx::Vec3 uxd = bx::cross(up_dir_, dir);
106
107 const bx::Quaternion rot1 = bx::fromAxisAngle(uxd, altitude);
108 sun_dir_ = bx::mul(dir, rot1);
109 }
110
111 float ecliptic_obliquity_;
112 float delta_;
113};
