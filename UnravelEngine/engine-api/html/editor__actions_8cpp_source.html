275void remove_extensions(std::vector<std::vector<std::string>>& resourceExtensions,
276 const std::vector<std::string>& extsToRemove)
277{
278 // Convert extsToRemove to a set of lowercase strings
279 std::unordered_set<std::string> extsToRemoveSet;
280 for(const auto& ext : extsToRemove)
281 {
282 extsToRemoveSet.insert(string_utils::to_lower(ext));
283 }
284
285 for(auto outerIt = resourceExtensions.begin(); outerIt != resourceExtensions.end();)
286 {
287 std::vector<std::string>& innerVec = *outerIt;
288
289 innerVec.erase(std::remove_if(innerVec.begin(),
290 innerVec.end(),
291 [&extsToRemoveSet](const std::string& ext)
292 {
293 return extsToRemoveSet.find(string_utils::to_lower(ext)) !=
294 extsToRemoveSet.end();
295 }),
296 innerVec.end());
297
298 if(innerVec.empty())
299 {
300 outerIt = resourceExtensions.erase(outerIt);
301 }
302 else
303 {
304 ++outerIt;
305 }

409auto collect_csharp_sources(const fs::path& source_directory, std::vector<fs::path>& out_sources) -> bool
410{
411 fs::error_code ec;
412 const fs::recursive_directory_iterator end;
413 fs::recursive_directory_iterator it(source_directory, ec);
414 if(ec)
415 {
416 APPLOG_ERROR("Failed to iterate source directory {}: {}", source_directory.string(), ec.message());
417 return false;
418 }
419 while(it != end)
420 {
421 const fs::path current_path = it->path();
422 const bool is_regular = it->is_regular_file(ec);
423 if(ec)
424 {
425 APPLOG_ERROR("Failed to query {}: {}", current_path.string(), ec.message());
426 return false;
427 }
428 if(is_regular && current_path.extension() == ".cs")
429 {
430 out_sources.push_back(current_path);
431 }
432 it.increment(ec);
433 if(ec)
434 {
435 APPLOG_ERROR("Failed while iterating {}: {}", source_directory.string(), ec.message());
436 return false;
437 }
438 }
439 return true;
440}

487auto build_external_dll_references(const std::vector<fs::path>& external_dll_paths) -> std::string
488{
489 std::string external_dll_references;

492 {
493 const std::string dll_name = dll_path.filename().string();
494 const fs::path dll_absolute_path = fs::absolute(dll_path, ec);
495 if(ec)
496 {
497 APPLOG_ERROR("Failed to resolve absolute path for {}: {}", dll_path.string(), ec.message());
498 return {};
499 }
500 external_dll_references += " <Reference Include=\"" + dll_name + "\">\n";
501 external_dll_references += " <HintPath>" + dll_absolute_path.generic_string() + "</HintPath>\n";
502 external_dll_references += " <Private>False</Private>\n";
503 external_dll_references += " </Reference>\n";
504 }
505 return external_dll_references;

1165auto editor_actions::save_entity_as_prefab(rtti::context& ctx,
1166 entt::handle entity,
1167 const fs::path& prefab_path,
1168 bool attach) -> asset_handle<prefab>
1169{
1170 animation_entity_preview::stop_all();
1171 if(!asset_writer::atomic_save_to_file(prefab_path.string(), entity))
1172 {
1173 return {};
1174 }
1175 auto& am = ctx.get_cached<asset_manager>();
1176 auto prefab_asset = am.get_asset<prefab>(fs::convert_to_protocol(prefab_path).generic_string());
1177 if(!attach)
1178 {
1179 return prefab_asset;
1180 }
1181 // The entity may already be an instance of a different prefab. Its old bookkeeping is keyed by
1182 // uids that also exist in the file just written (prefab uids survive the save), so keeping it
1183 // would suppress those properties from the new asset on every resync - and a stale instance_id
1184 // would claim a slot in a document that never wrote this instance. Becoming the source of a new
1185 // prefab leaves nothing overridden by definition.
1186 //
1187 // Reset field by field, not replaced: emplace_or_replace on an existing component takes entt's
1188 // patch path, which assigns a default-constructed component over the live one - owner handle
1189 // included - and fires only on_update, so the owner never gets re-stamped. That null owner
1190 // surfaced as a crash in the inspector's Apply All. And not removed-and-re-added either: the
1191 // on_destroy hook strips prefab ids from the whole subtree.
1192 // A nested instance saved as its own prefab is no longer its container's slot: the container's
1193 // document would put the old instance back there on its next replay. The slot is stated removed
1194 // on the container, and the new instance is the user's.
1195 if(const auto* old_prefab = entity.try_get<prefab_component>();
1196 old_prefab != nullptr && !old_prefab->instance_id.is_nil())
1197 {
1198 const auto* trans = entity.try_get<transform_component>();
1199 auto container =
1200 trans != nullptr ? prefab_override_context::find_prefab_root_entity(trans->get_parent()) : entt::handle{};
1201 if(auto* container_prefab = container ? container.try_get<prefab_component>() : nullptr)
1202 {
1203 container_prefab->remove_instance(old_prefab->instance_id);
1204 container_prefab->changed = true;
1205 }
1206 }
1207 auto& prefab_comp = entity.get_or_emplace<prefab_component>();
1208 prefab_comp.clear_overrides();
1209 prefab_comp.instance_id = {};
1210 prefab_comp.instance_document = {};
1211 prefab_comp.source = prefab_asset;
1212 return prefab_asset;
1213}

1232auto editor_actions::migrate_prefabs(rtti::context& ctx, size_t* total_count) -> size_t
1233{
1234 if(total_count != nullptr)
1235 {
1236 *total_count = 0;
1237 }
1238 auto& play = ctx.get_cached<play_mode>();
1239 if(play.is_active())
1240 {
1241 APPLOG_ERROR("Migrate: refusing to run during play mode.");
1242 return 0;
1243 }
1244 auto& em = ctx.get_cached<editing_manager>();
1245 if(em.is_prefab_mode())
1246 {
1247 APPLOG_ERROR("Migrate: leave prefab mode first.");
1248 return 0;
1249 }
1250 const auto report = validate_prefab_graph(ctx);
1251 if(!report.is_valid())
1252 {
1253 APPLOG_ERROR("Migrate: aborted, the prefab graph contains a cycle. Nothing was written.");
1254 return 0;
1255 }
1256
1257 // Nested prefabs before the prefabs that contain them, scenes last - the same order the
1258 // bake uses, and for the same reason: a container re-saved after the assets it nests finds
1259 // their slots named by them, and names only its own additions.
1260 auto& am = ctx.get_cached<asset_manager>();
1261 const auto order = asset_deps::compute_build_order(asset_deps::collect_prefab_asset_uids(am),
1262 asset_deps::make_prefab_dependency_resolver(am));
1263 scene scratch("prefab_migrate");
1264 size_t written = 0;
1265 for(const auto& uid : order.ordered)
1266 {
1267 if(bake_prefab_asset(ctx, scratch, uid, {}, true))
1268 {
1269 ++written;
1270 }
1271 }
1272 scratch.unload();
1273 if(total_count != nullptr)
1274 {
1275 *total_count = order.ordered.size();
1276 }
1277 APPLOG_INFO("Migrate: re-saved {} of {} prefab(s) and scene(s).", written, order.ordered.size());
1278 return written;
1279}

1281auto editor_actions::migrate_texture_color_spaces(const std::string& group) -> size_t
1282{
1283 auto& ctx = engine::context();
1284 auto& am = ctx.get_cached<asset_manager>();
1285
1286 auto tag = [&](const asset_handle<gfx::texture>& tex, texture_importer_meta::color_space colorspace) -> bool
1287 {
1288 if(!tex)
1289 {
1290 return false;
1291 }
1292 const auto meta_path = asset_writer::resolve_meta_file(tex);
1293 asset_meta meta;
1294 if(!load_from_file(meta_path.string(), meta))
1295 {
1296 // No meta yet (texture not scanned): the watcher will create a default
1297 // one; re-running the migration afterwards will pick it up.
1298 return false;
1299 }
1300 auto importer = std::dynamic_pointer_cast<texture_importer_meta>(meta.importer);
1301 if(!importer)
1302 {
1303 return false;
1304 }
1305 if(importer->colorspace != texture_importer_meta::color_space::automatic)
1306 {
1307 // Explicit user choice wins over the migration.
1308 return false;
1309 }
1310 importer->colorspace = colorspace;
1311 fs::error_code err;
1312 asset_writer::atomic_write_file(meta_path,
1313 [&](const fs::path& temp)
1314 {
1315 save_to_file(temp.string(), meta);
1316 },
1317 err);
1318 if(err)
1319 {
1320 APPLOG_WARNING("Color space migration: failed to write meta for {}: {}", tex.id(), err.message());
1321 return false;
1322 }
1323 // Touch the source so the watcher recompiles it into a tagged container.
1324 fs::error_code ec;
1325 auto source = fs::absolute(fs::resolve_protocol(tex.id()).string(), ec);
1326 if(!ec)
1327 {
1328 fs::watcher::touch(source, false);
1329 }
1330 return true;
1331 };
1332
1333 size_t tagged = 0;
1334 auto materials = am.get_assets<material>(group);
1335 for(const auto& asset : materials)
1336 {
1337 auto mat = asset.get();
1338 if(!mat || !mat->is<pbr_material>())
1339 {
1340 continue;
1341 }
1342 const auto& pbr = static_cast<const pbr_material&>(*mat);
1343 tagged += size_t(tag(pbr.get_color_map(), texture_importer_meta::color_space::srgb));
1344 tagged += size_t(tag(pbr.get_emissive_map(), texture_importer_meta::color_space::srgb));
1345 tagged += size_t(tag(pbr.get_normal_map(), texture_importer_meta::color_space::linear));
1346 tagged += size_t(tag(pbr.get_roughness_map(), texture_importer_meta::color_space::linear));
1347 tagged += size_t(tag(pbr.get_metalness_map(), texture_importer_meta::color_space::linear));
1348 tagged += size_t(tag(pbr.get_ao_map(), texture_importer_meta::color_space::linear));
1349 }
1350 APPLOG_INFO("Color space migration: tagged {} texture(s) from {} material(s).", tagged, materials.size());
1351 return tagged;
1352}
