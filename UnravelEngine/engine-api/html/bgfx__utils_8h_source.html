104inline uint32_t encodeNormalRgba8(float _x, float _y = 0.0f, float _z = 0.0f, float _w = 0.0f)
105{
106 const float src[] = {
107 _x * 0.5f + 0.5f,
108 _y * 0.5f + 0.5f,
109 _z * 0.5f + 0.5f,
110 _w * 0.5f + 0.5f,
111 };
112 uint32_t dst;
113 bx::packRgba8(&dst, src);
114 return dst;
115}
