65auto format_duration(float seconds) -> std::string
66{
67 if(seconds < float(SECONDS_PER_MINUTE))
68 {
69 return fmt::format("{:.1f} s", seconds);
70 }
71 const int whole_seconds = int(seconds);

81auto find_running_detail(const std::vector<tpp::running_job>& running, const std::string& kind) -> std::string
82{

84 running.end(),

88 });
89 if(job == running.end())

91 return {};

198void footer_panel::track_jobs(const std::map<std::string, size_t>& counts)
199{
200 const auto now = std::chrono::steady_clock::now();
201 if(!job_batch_.is_active)
202 {
203 job_batch_ = {};
204 job_batch_.is_active = true;
205 job_batch_.started_at = now;
206 }
207 job_batch_.last_busy_at = now;
208 for(auto& [name, kind] : job_batch_.kinds)
209 {
210 if(counts.find(name) == counts.end())
211 {
212 kind.pending = 0;
213 }
214 }
215 for(const auto& [name, count] : counts)
216 {
217 job_kind_progress& kind = job_batch_.kinds[name];
218 // Every rise between two looks is jobs added; jobs added and finished in between go uncounted.
219 if(count > kind.pending)
220 {
221 kind.total += count - kind.pending;
222 }
223 kind.pending = count;
224 }
225}

227void footer_panel::update_job_status(rtti::context& ctx)
228{
229 auto& thr = ctx.get_cached<threader>();
230 const auto counts = thr.pool->get_jobs_count_detailed();
231 const auto now = std::chrono::steady_clock::now();
232 if(!counts.empty())
233 {
234 track_jobs(counts);
235 const auto running = thr.pool->get_running_jobs();
236 job_status::job_card card;
237 card.elapsed_seconds = std::chrono::duration<float>(now - job_batch_.started_at).count();
238 // The kinds with jobs left first, the busiest at the top.
239 std::vector<std::pair<std::string, job_kind_progress>> kinds(job_batch_.kinds.begin(), job_batch_.kinds.end());
240 std::stable_sort(kinds.begin(),
241 kinds.end(),
242 [](const auto& lhs, const auto& rhs)
243 {
244 return lhs.second.pending > rhs.second.pending;
245 });
246 for(const auto& [name, kind] : kinds)
247 {
248 card.pending += kind.pending;
249 if(kind.pending == 0)
250 {
251 continue;
252 }
253 if(card.rows.size() == job_status::JOB_CARD_MAX_KINDS)
254 {
255 ++card.hidden_kind_count;
256 continue;
257 }
258 card.rows.push_back({name.empty() ? job_status::JOB_UNNAMED_KIND : name,
259 kind.total - kind.pending,
260 kind.total,
261 job_status::find_running_detail(running, name)});
262 }
263 auto callback = [card = std::move(card)](const ImGuiToast&, float opacity, const ImVec4&)
264 {
265 job_status::draw_job_card(card, opacity);
266 };
267 ImGui::PushNotification(job_status::JOB_STATUS_TOAST_ID, ImGuiToast(ImGuiToastType_None, callback, job_status::JOB_CARD_DISMISS_MILLISECONDS));
268 return;
269 }
270 if(!job_batch_.is_active || now - job_batch_.last_busy_at < job_status::JOB_BATCH_IDLE_TIME)
271 {
272 return;
273 }
274 size_t total = 0;
275 for(const auto& [name, kind] : job_batch_.kinds)
276 {
277 total += kind.total;
278 }
279 const float seconds = std::chrono::duration<float>(job_batch_.last_busy_at - job_batch_.started_at).count();
280 ImGuiToast toast(ImGuiToastType_Success, job_status::JOB_SUMMARY_DISMISS_MILLISECONDS);
281 toast.set_title("%s", "Jobs finished");
282 toast.set_content("%s", fmt::format("{} in {}", job_status::format_job_count(total), job_status::format_duration(seconds)).c_str());
283 ImGui::PushNotification(job_status::JOB_STATUS_TOAST_ID, toast);
284 job_batch_ = {};
285}
