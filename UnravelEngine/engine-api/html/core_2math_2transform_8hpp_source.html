821template<typename T, precision Q>
822auto inverse(transform_t<T, Q> const& t) noexcept -> transform_t<T, Q>
823{
824 // Uniform TRS: exact component inverse. No 4x4 inverse, and the result's
825 // components are already valid, so no decompose is deferred onto the caller.
826 if(t.can_use_simplified_calculations())
827 {
828 transform_t<T, Q> result;
829 const T inv_scale = T(1) / detail::scale_fix(t.scale_.x);
830 result.rotation_ = glm::conjugate(t.rotation_);
831 result.scale_ = typename transform_t<T, Q>::vec3_t(inv_scale, inv_scale, inv_scale);
832 result.position_ = -(result.rotation_ * (t.position_ * inv_scale));
833 result.matrix_needs_recompute_ = true;
834 return result;
835 }
836
837 const auto& m = t.get_matrix();
838 if(!t.components_need_recompute_ && t.is_perspective_identity_cached_)
839 {
840 // Affine: cheaper and better conditioned than the general inverse.
841 return glm::affineInverse(m);
842 }
843 return glm::inverse(m);
844}

924TRANSFORM_INLINE auto transform_t<T, Q>::sync_euler_degrees_to_hint(vec3_t angles, const vec3_t& hint) noexcept
925 -> vec3_t
926{
927 auto repeat_working = [](T t, T length) -> T
928 {
929 return (t - (floor(t / length) * length));
930 };
931 angles.x = repeat_working(angles.x - hint.x + T(180), T(360)) + hint.x - T(180);
932 angles.y = repeat_working(angles.y - hint.y + T(180), T(360)) + hint.y - T(180);
933 angles.z = repeat_working(angles.z - hint.z + T(180), T(360)) + hint.z - T(180);
934 return angles;
935}

1102TRANSFORM_INLINE void transform_t<T, Q>::set_rotation(const quat_t& rotation) noexcept
1103{
1104 update_components();
1105 const quat_t normalized = glm::normalize(rotation);
1106 // Undo/property actions re-apply the same orientation via set_rotation after
1107 // set_rotation_euler*. Keep the typed Euler hint only while it still maps to the
1108 // NEW orientation. Comparing new-vs-previous instead would let many small
1109 // incremental rotations (each within tolerance of the last) drift arbitrarily
1110 // far from a hint that is never invalidated.
1111 if(!euler_hint_dirty_)
1112 {
1113 const quat_t from_hint = glm::normalize(quat_t(radians(euler_angles_hint_)));
1114 if(!rotations_equivalent(from_hint, normalized))
1115 {
1116 make_euler_hint_dirty();
1117 }
1118 }
1119 rotation_ = normalized;
1120 make_matrix_dirty();
1121}

1411 typename transform_t<T, Q>::vec3_t
1412{
1413 if(can_use_simplified_calculations_without_uniform_scale())
1414 {
1415 // Direct transformation using components
1416 return get_position() + (get_rotation() * (get_scale() * v));
1417 }
1418
1419 // Use matrix multiplication
1420 vec4_t result = get_matrix() * vec4_t(v, T(1));
1421 if(!components_need_recompute_ && is_perspective_identity_cached_)
1422 {
1423 // Affine matrix: w is exactly 1, skip the divide.
1424 return vec3_t(result);
1425 }
1426 if(abs(result.w) <= epsilon<T>())
1427 {
1428 // Point on the eye plane of a perspective transform: no finite projection.
1429 return vec3_t(result);
1430 }

1434template<typename T, precision Q>

1436 typename transform_t<T, Q>::vec3_t
1437{
1438 if(can_use_simplified_calculations_without_uniform_scale())
1439 {
1440 auto scale = detail::scale_fix(get_scale());
1441 // Compute inverse scale
1442 vec3_t inv_scale = T(1) / scale;
1443
1444 // Compute inverse rotation
1445 // Using conjugate
1446 quat_t inv_rotation = glm::conjugate(get_rotation()); // Efficient and correct for unit quaternions
1447
1448 // Using inverse
1449 // quat_t inv_rotation = glm::inverse(get_rotation()); // Also correct but less efficient for unit quaternions
1450
1451 // Apply inverse transformations
1452 vec3_t result = inv_rotation * (v - get_position());
1453 result *= inv_scale;
1454
1455 return result;
1456 }
1457
1458 // Use matrix multiplication
1459 if(!components_need_recompute_ && is_perspective_identity_cached_)

1461 // Affine matrix: the affine inverse is cheaper and w stays exactly 1.
1462 vec4_t result = glm::affineInverse(get_matrix()) * vec4_t(v, T(1));
1463 return vec3_t(result);
1464 }
1465 vec4_t result = glm::inverse(get_matrix()) * vec4_t(v, T(1));
1466 if(abs(result.w) <= epsilon<T>())
1467 {
1468 // Point on the eye plane of a perspective transform: no finite projection.
1469 return vec3_t(result);
1470 }
1471 return vec3_t(result) / result.w;

1474template<typename T, precision Q>
1475TRANSFORM_INLINE auto transform_t<T, Q>::transform_normal(const vec3_t& v) const noexcept ->
1476 typename transform_t<T, Q>::vec3_t
1477{
1478 // Contract: the direction of the inverse-transpose (normal matrix) with the input's
1479 // length preserved. Backs the script TransformDirection, so length must not pick up
1480 // the object's scale, and both branches must agree.
1481 if(can_use_simplified_calculations())
1482 {
1483 // Uniform scale s: the normal matrix is (1/s) * R, so the direction is
1484 // sign(s) * (R * v) and the length of v is already preserved.
1485 const vec3_t rotated = get_rotation() * v;
1486 return get_scale().x < T(0) ? -rotated : rotated;
1487 }
1488
1489 return transform_normal_general(v);
1490}

1492template<typename T, precision Q>
1493TRANSFORM_NOINLINE auto transform_t<T, Q>::transform_normal_general(const vec3_t& v) const noexcept ->
1494 typename transform_t<T, Q>::vec3_t
1495{
1496 // Use matrix multiplication
1497 // Extract the linear (rotation, scaling, skew) part of the transformation matrix
1498 mat3_t linear_matrix = mat3_t(get_matrix());
1499
1500 // Compute the inverse transpose of the linear matrix
1501 mat3_t normal_matrix = glm::transpose(glm::inverse(linear_matrix));
1502
1503 // Transform the normal vector using the normal matrix
1504 vec3_t result = normal_matrix * v;
1505
1506 // Rescale to the input length so both branches share one magnitude contract.
1507 const T result_len2 = dot(result, result);
1508 const T input_len2 = dot(v, v);
1509 if(result_len2 > T(0) && input_len2 > T(0))
1510 {
1511 result *= sqrt(input_len2 / result_len2);
1512 }
1513 return result;
1514}

1606template<typename T, precision Q>
1607TRANSFORM_INLINE auto transform_t<T, Q>::operator*(const transform_t& t) const noexcept -> transform_t<T, Q>
1608{
1609 // Hottest path in the engine (scene-graph world = parent * local).
1610 //
1611 // Component-space compose is legal when both operands are skew/perspective
1612 // free with valid components and OUR scale is uniform: S1 = s*I commutes
1613 // with R2, so T1 R1 S1 T2 R2 S2 = T(p1 + R1 s p2) (R1 R2) (s * S2). The
1614 // right-hand scale may be non-uniform. This keeps the result's components
1615 // exact, so a later get_position()/get_rotation() costs nothing - the
1616 // matrix fallback instead defers a full matrix decompose (~10x the cost of
1617 // the multiply) to the first component read.
1618 if(can_use_simplified_calculations() && t.can_use_simplified_calculations_without_uniform_scale())
1619 {
1620 transform_t result;
1621 result.position_ = position_ + rotation_ * (scale_ * t.position_);
1622 result.rotation_ = glm::normalize(rotation_ * t.rotation_);
1623 result.scale_ = scale_ * t.scale_;
1624 // Our scale is uniform, so the product's uniformity is the right-hand's.
1625 result.is_scale_uniform_cached_ = t.is_scale_uniform_cached_;
1626 // Skew/perspective stay identity-initialized; the euler hint stays dirty.
1627 if(!matrix_needs_recompute_ && !t.matrix_needs_recompute_)
1628 {
1629 // Both operand matrices are already resolved (steady-state scene
1630 // graph): the SIMD multiply is cheaper than a deferred recompose.
1631 result.matrix_ = matrix_ * t.matrix_;
1632 }
1633 else
1634 {
1635 result.matrix_needs_recompute_ = true;
1636 }
1637 return result;
1638 }
1639
1640 // Fallback to matrix multiplication and decomposition
1641 return get_matrix() * t.get_matrix();
1642}

1697template<typename T, precision Q>
1698TRANSFORM_INLINE void transform_t<T, Q>::update_matrix() const noexcept
1699{
1700 if(matrix_needs_recompute_)
1701 {
1702 if(can_use_simplified_calculations())
1703 {
1704 // Direct T*R*S compose: write the scaled rotation columns straight into
1705 // the matrix. The previous translate(I) * mat4_cast(q) * scale(I) form
1706 // paid two full mat4 multiplies for the same result.
1707 const mat3_t rotation_matrix = glm::mat3_cast(rotation_);
1708 matrix_[0] = vec4_t(rotation_matrix[0] * scale_.x, T(0));
1709 matrix_[1] = vec4_t(rotation_matrix[1] * scale_.y, T(0));
1710 matrix_[2] = vec4_t(rotation_matrix[2] * scale_.z, T(0));
1711 matrix_[3] = vec4_t(position_, T(1));
1712 }
1713 else
1714 {
1715 glm_recompose(matrix_, get_scale(), get_rotation(), get_position(), get_skew(), get_perspective());
1716 }
1717
1718 matrix_needs_recompute_ = false;
1719 }
1720}

1832// Rotation that turns +Z onto forward and +Y toward upwards (approx); backs the script Quaternion.LookRotation.
1833inline auto look_rotation(const glm::vec3& forward, const glm::vec3& upwards) -> glm::quat
1834{
1835 // Build the basis directly: +Z = forward by construction, independent of the
1836 // GLM_FORCE_LEFT_HANDED configuration that glm::lookAt depends on, and safe for
1837 // the degenerate inputs (zero forward, forward parallel to upwards) that made
1838 // the lookAt-based version produce NaNs.
1839 constexpr float k_degenerate_len2 = 1e-12f;
1840 const float forward_len2 = glm::dot(forward, forward);
1841 if(forward_len2 < k_degenerate_len2)
1842 {
1843 return glm::identity<glm::quat>();
1844 }
1845 const glm::vec3 f = forward / glm::sqrt(forward_len2);
1846 glm::vec3 right = glm::cross(upwards, f);
1847 float right_len2 = glm::dot(right, right);
1848 if(right_len2 < k_degenerate_len2)
1849 {
1850 // upwards is collinear with forward (or zero): substitute any axis
1851 // orthogonal to f so the basis stays well defined.
1852 const glm::vec3 orth =
1853 glm::abs(f.x) > glm::abs(f.z) ? glm::vec3(-f.y, f.x, 0.0f) : glm::vec3(0.0f, -f.z, f.y);
1854 right = glm::cross(orth, f);
1855 right_len2 = glm::dot(right, right);
1856 }
1857 right /= glm::sqrt(right_len2);
1858 const glm::vec3 up = glm::cross(f, right);
1859 return glm::quat_cast(glm::mat3(right, up, f));
1860}

1862inline auto from_to_rotation(const glm::vec3& from, const glm::vec3& to) -> glm::quat
1863{
1864 // 1. Guard degenerate inputs (a zero vector has no direction) and normalize.
1865 constexpr float k_degenerate_len2 = 1e-12f;
1866 const float from_len2 = glm::dot(from, from);
1867 const float to_len2 = glm::dot(to, to);
1868 if(from_len2 < k_degenerate_len2 || to_len2 < k_degenerate_len2)
1869 {
1870 return glm::identity<glm::quat>();
1871 }
1872 const glm::vec3 f = from / glm::sqrt(from_len2);
1873 const glm::vec3 t = to / glm::sqrt(to_len2);
1874
1875 const float dot_product = glm::dot(f, t);
1876
1877 // 2. Nearly opposite: a 180-degree turn about any axis orthogonal to 'from'.
1878 // The half-vector construction below degenerates to a zero quaternion here.
1879 if(dot_product <= -1.0f + 1e-6f)
1880 {
1881 glm::vec3 orth = glm::abs(f.x) > glm::abs(f.z) ? glm::vec3(-f.y, f.x, 0.0f) : glm::vec3(0.0f, -f.z, f.y);
1882 orth = glm::normalize(orth);
1883 return glm::angleAxis(glm::pi<float>(), orth);
1884 }
1885
1886 // 3. Half-vector construction: normalize(w = 1 + dot, xyz = cross(f, t)) rotates
1887 // f onto t. Trig-free and more accurate near dot = +/-1 than acos + angleAxis
1888 // (also exact for the nearly-parallel case, which needs no special handling).
1889 const glm::quat q(1.0f + dot_product, glm::cross(f, t));
1890 return glm::normalize(q);
1891}
