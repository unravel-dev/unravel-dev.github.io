95auto is_root_of(const std::string& root, const std::string& path_string) -> bool
96{
97 if(root.empty() || path_string.size() < root.size())
98 {
99 return false;
100 }
101 const bool ignore_case = is_case_insensitive();

103 {
104 return ignore_case ? fold_ascii_case(lhs) == fold_ascii_case(rhs) : lhs == rhs;
105 };
106 if(!std::equal(root.begin(), root.end(), path_string.begin(), chars_match))
107 {

111 {
112 return true;
113 }

117 };
118 return is_separator(root.back()) || is_separator(path_string[root.size()]);

144{

152

168 return false;

170
171 auto const char_count = end_pos - start_pos;
172
173 if(!in.seekg(start_pos) || !in.good())
174 {
175 return false;
176 }
177
178 container.resize(static_cast<std::size_t>(char_count));
179
180 if(!container.empty())
181 {
182 in.read(reinterpret_cast<CharT*>(&container[0]), char_count);
183 container.resize(in.gcount());

309auto match_protocol_root(const std::string& string_path, const path& fallback) -> path
310{
311 const auto& protocols = get_path_protocols();
312
313 const protocols_t::value_type* best_protocol{};
314 for(const auto& protocol_pair : protocols)
315 {
316 const auto& resolved_protocol = protocol_pair.second;
317 if(!detail::is_root_of(resolved_protocol, string_path))
318 {
319 continue;
320 }
321 if(!best_protocol || best_protocol->second.size() < resolved_protocol.size())
322 {
323 best_protocol = &protocol_pair;
324 }
325 }
326 if(!best_protocol)

328 return fallback;
329 }
330 // The matched root is replaced by length: it may differ from the path in letter case.

333 {
334 relative.insert(relative.begin(), '/');
