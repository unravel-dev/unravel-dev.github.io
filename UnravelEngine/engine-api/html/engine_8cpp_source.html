166void on_app_event(os::events type)
167{
168 auto* ctx = engine::try_context();
169 if(ctx == nullptr)
170 {
171 return;
172 }
173 auto& ev = ctx->get_cached<events>();
174 switch(type)
175 {
176 case os::events::app_will_enter_background:
177 APPLOG_INFO("The application enters the background.");
178 ev.on_app_enter_background(*ctx);
179 APPLOG_FLUSH();
180 break;
181 case os::events::app_did_enter_foreground:
182 APPLOG_INFO("The application is back in the foreground.");
183 // The time spent in the background is not a frame.
184 ctx->get_cached<simulation>().reset_delta_clock();
185 ev.on_app_enter_foreground(*ctx);
186 break;
187 case os::events::app_low_memory:
188 // Android reports every memory trim request this way, the routine one on entering the
189 // background included.
190 APPLOG_INFO("The system asks the application to use less memory.");
191 ev.on_app_low_memory(*ctx);
192 break;
193 case os::events::app_terminating:
194 APPLOG_INFO("The system ends the application.");
195 engine::interrupt();
196 APPLOG_FLUSH();
197 break;
198 default:
199 break;
200 }
201}

617auto engine::process() -> int
618{
619 auto& ctx = engine::context();
620
621 auto& sim = ctx.get_cached<simulation>();
622 auto& ev = ctx.get_cached<events>();
623 auto& rend = ctx.get_cached<renderer>();
624 auto& thr = ctx.get_cached<threader>();
625 auto& input = ctx.get_cached<input_system>();
626
627 {
628 APP_SCOPE_PERF("Frame Loop");
629
630 {
631 APP_SCOPE_PERF("Threader Process");
632 thr.process();
633 }
634
635 sim.run_one_frame(true);
636
637 auto dt = sim.get_delta_time();
638 auto& play = ctx.get_cached<play_mode>();
639
640 // First simulation frame: zero dt so Start/init work cannot integrate
641 // a hitch. Increment frames_running at frame end so consumers can also
642 // detect this via frames_running() == 0 (e.g. mid-frame enter_running).
643 if(play.is_simulation_running() && play.frames_running() == 0)
644 {
645 dt = {};
646 }
647
648 if(play.is_paused())
649 {
650 // A requested frame step runs this paused frame like a running one, so every system
651 // (scripts, late update, physics, animation, seq) advances once in its usual order.
652 dt = play.begin_frame_step() ? delta_t(play_mode::FRAME_STEP_SECONDS) * sim.get_time_scale() : delta_t{};
653 }
654
655 update_input_zone(rend, input);
656
657 input.manager.before_events_update();
658 update_cursor_visibility(rend, input);
659
660 bool should_quit = is_shutting_down;
661
662 {
663 APP_SCOPE_PERF("Poll OS Events");
664 os::event e{};
665 while(!should_quit && os::poll_event(e))
666 {
667 ev.on_os_event(ctx, e);
668
669 input.manager.on_os_event(e);
670
671 should_quit |= rend.get_main_window() == nullptr;
672 }
673 }
674
675 {
676 APP_SCOPE_PERF("After Events Update");
677 input.manager.after_events_update();
678 }
679
680 if(should_quit)
681 {
682 ctx.get_cached<play_mode>().set_active(ctx, false);
683 const bool restart = is_restart_requested.exchange(false);
684 is_shutting_down = false;
685 if(restart)
686 {
687 APPLOG_INFO("Engine process returning restart action");
688 return SERVICE_RESULT_RESTART;
689 }
690 return SERVICE_RESULT_EXIT;
691 }
692
693 {
694 APP_SCOPE_PERF("Frame Begin");
695 ev.on_frame_begin(ctx, dt);
696 }
697
698 {
699 APP_SCOPE_PERF("Seq Update");
700 seq::update(dt);
701 }
702
703
704 {
705 APP_SCOPE_PERF("Frame Update");
706 ev.on_frame_update(ctx, dt);
707 }
708
709 {
710 APP_SCOPE_PERF("Seq Update Zero");
711 seq::update(delta_t::zero());
712 }
713
714 {
715 APP_SCOPE_PERF("Frame Before Render");
716 ev.on_frame_before_render(ctx, dt);
717 }
718
719 {
720 APP_SCOPE_PERF("Frame Render");
721 ev.on_frame_render(ctx, dt);
722 }
723
724 {
725 APP_SCOPE_PERF("Frame End");
726 ev.on_frame_end(ctx, dt);
727 }
728
729 // Before the step ends, so a stepped frame counts as simulated.
730 if(play.is_simulation_running())
731 {
732 play.on_simulation_frame();
733 }
734
735 play.end_frame_step();
736 }
737
738 get_app_profiler()->swap();
739
740 return 1;
741}
