227void audio_voice_pool::free_clip_overflow(const hpp::uuid& clip_uid)
228{
229 auto it = limits_.find(clip_uid);
230 if(it == limits_.end())
231 {
232 return;
233 }
234 const int max_voices = it->second.max_voices;
235 while(max_voices > 0)
236 {
237 int count = 0;
238 uint32_t oldest = 0;
239 uint64_t oldest_serial = std::numeric_limits<uint64_t>::max();
240 for(uint32_t i = 0; i < uint32_t(voices_.size()); ++i)
241 {
242 const auto& v = voices_[i];
243 if(!v.source || !v.is_one_shot || v.clip_uid != clip_uid)
244 {
245 continue;
246 }
247 ++count;
248 if(v.serial < oldest_serial)
249 {
250 oldest_serial = v.serial;
251 oldest = i;
252 }
253 }
254 if(count < max_voices)
255 {
256 return;
257 }
258 release(oldest);
259 }

301{

353auto audio_voice_pool::play(const std::shared_ptr<audio_clip>& clip,
354 const hpp::uuid& clip_uid,
355 const one_shot_params& params) -> uint64_t
356{
357 const auto now = std::chrono::steady_clock::now();
358 if(!clip || !clip->is_valid() || !is_start_allowed(clip_uid, now))
359 {
360 return 0;
361 }
362 auto source = create_source();
363 if(!source)
364 {
365 return 0;
366 }
367 // A clip at its own limit makes room by stopping its oldest voice, before anything else is freed.
368 free_clip_overflow(clip_uid);
369 make_room();
370 apply_voice_space(*source, params.spatial, one_shot_rolloff);
371 source->set_distance(audio_default_min_distance, audio_default_max_distance);
372 if(params.spatial)
373 {
374 source->set_position({{params.position.x, params.position.y, params.position.z}});
375 }
376 source->set_pitch(clamp_pitch(params.pitch));
377 const owned_voice_params voice_params{params.volume, params.bus, params.priority, false};
378 const auto index = add_voice(std::move(source), voice_params, true);
379 auto& v = voices_[index];
380 v.clip = clip;
381 v.clip_uid = clip_uid;
382 mark_started(clip_uid, now);
383 v.source->bind(*clip);
384 v.source->play();
385 return make_id(index);
386}
