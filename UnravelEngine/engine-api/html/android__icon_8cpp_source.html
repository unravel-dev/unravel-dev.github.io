63auto load_image(const fs::path& file) -> premultiplied_image
64{
65 const auto bytes = read_bytes(file);
66 if(bytes.empty())
67 {
68 return {};
69 }
70 bx::DefaultAllocator allocator;
71 bimg::ImageContainer* container =
72 bimg::imageParse(&allocator, bytes.data(), static_cast<std::uint32_t>(bytes.size()), bimg::TextureFormat::RGBA32F);
73 if(container == nullptr)
74 {
75 return {};
76 }
77 premultiplied_image image;
78 bimg::ImageMip mip;
79 if(bimg::imageGetRawData(*container, 0, 0, container->m_data, container->m_size, mip))
80 {
81 const auto* first = reinterpret_cast<const float*>(mip.m_data);
82 image.width = mip.m_width;
83 image.height = mip.m_height;
84 image.texels.assign(first, first + std::size_t(mip.m_width) * mip.m_height * CHANNELS);
85 }
86 bimg::imageFree(container);
87 for(std::size_t i = 0; i < image.texels.size(); i += CHANNELS)
88 {
89 for(std::uint32_t channel = 0; channel < ALPHA; ++channel)
90 {
91 image.texels[i + channel] *= image.texels[i + ALPHA];
92 }
93 }
94 return image;
95}

99auto sample(const premultiplied_image& image, float x, float y) -> std::array<float, CHANNELS>
100{
101 const float column = std::clamp(x - 0.5f, 0.0f, float(image.width - 1));
102 const float row = std::clamp(y - 0.5f, 0.0f, float(image.height - 1));
103 const auto left = static_cast<std::uint32_t>(column);
104 const auto top = static_cast<std::uint32_t>(row);
105 const std::uint32_t right = std::min(left + 1, image.width - 1);
106 const std::uint32_t bottom = std::min(top + 1, image.height - 1);
107 const float across = column - float(left);
108 const float down = row - float(top);
109 std::array<float, CHANNELS> value{};
110 for(std::uint32_t channel = 0; channel < CHANNELS; ++channel)
111 {
112 const float upper = image.get(left, top, channel) * (1.0f - across) + image.get(right, top, channel) * across;
113 const float lower = image.get(left, bottom, channel) * (1.0f - across) + image.get(right, bottom, channel) * across;
114 value[channel] = upper * (1.0f - down) + lower * down;
115 }
116 return value;
117}

121auto resize(const premultiplied_image& image, std::uint32_t width, std::uint32_t height) -> premultiplied_image
122{
123 premultiplied_image result{width, height, std::vector<float>(std::size_t(width) * height * CHANNELS)};
124 const float step_x = float(image.width) / float(width);
125 const float step_y = float(image.height) / float(height);
126 const auto samples_x = std::clamp(static_cast<std::uint32_t>(std::ceil(step_x)), 1u, MAX_SAMPLES_PER_AXIS);
127 const auto samples_y = std::clamp(static_cast<std::uint32_t>(std::ceil(step_y)), 1u, MAX_SAMPLES_PER_AXIS);
128 const float weight = 1.0f / float(samples_x * samples_y);
129 for(std::uint32_t y = 0; y < height; ++y)
130 {
131 for(std::uint32_t x = 0; x < width; ++x)
132 {
133 float* target = result.texels.data() + (std::size_t(y) * width + x) * CHANNELS;
134 for(std::uint32_t sy = 0; sy < samples_y; ++sy)
135 {
136 const float source_y = (float(y) + (float(sy) + 0.5f) / float(samples_y)) * step_y;
137 for(std::uint32_t sx = 0; sx < samples_x; ++sx)
138 {
139 const float source_x = (float(x) + (float(sx) + 0.5f) / float(samples_x)) * step_x;
140 const auto value = sample(image, source_x, source_y);
141 for(std::uint32_t channel = 0; channel < CHANNELS; ++channel)
142 {
143 target[channel] += value[channel] * weight;
144 }
145 }
146 }
147 }
148 }
149 return result;
150}

154auto make_foreground(const premultiplied_image& image, std::uint32_t scale_percent) -> premultiplied_image
155{
156 const std::uint32_t layer = ANDROID_ICON_LAYER_DP * scale_percent / PERCENT;
157 const std::uint32_t visible = ANDROID_ICON_VISIBLE_DP * scale_percent / PERCENT;
158 const float fit = std::min(float(visible) / float(image.width), float(visible) / float(image.height));
159 const auto width = std::max(1u, static_cast<std::uint32_t>(std::lround(float(image.width) * fit)));
160 const auto height = std::max(1u, static_cast<std::uint32_t>(std::lround(float(image.height) * fit)));
161 const auto scaled = resize(image, width, height);
162 premultiplied_image foreground{layer, layer, std::vector<float>(std::size_t(layer) * layer * CHANNELS, 0.0f)};
163 const std::uint32_t left = (layer - width) / 2;
164 const std::uint32_t top = (layer - height) / 2;
165 for(std::uint32_t row = 0; row < height; ++row)
166 {
167 const auto source = scaled.texels.begin() + std::ptrdiff_t(std::size_t(row) * width * CHANNELS);
168 const auto target = foreground.texels.begin() + std::ptrdiff_t((std::size_t(top + row) * layer + left) * CHANNELS);
169 std::copy(source, source + std::ptrdiff_t(width * CHANNELS), target);
170 }
171 return foreground;
172}

180auto to_rgba8(const premultiplied_image& image) -> std::vector<std::uint8_t>
181{
182 std::vector<std::uint8_t> bytes(image.texels.size());
183 for(std::size_t i = 0; i < image.texels.size(); i += CHANNELS)
184 {
185 const float alpha = image.texels[i + ALPHA];
186 for(std::uint32_t channel = 0; channel < ALPHA; ++channel)
187 {
188 bytes[i + channel] = alpha > 0.0f ? to_byte(image.texels[i + channel] / alpha) : 0;
189 }
190 bytes[i + ALPHA] = to_byte(alpha);
191 }
192 return bytes;
193}
