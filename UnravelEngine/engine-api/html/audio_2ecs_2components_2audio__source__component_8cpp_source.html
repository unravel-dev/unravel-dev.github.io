36void audio_source_component::update(const math::transform& t)
37{
38 if(get_autoplay() && !auto_play_consumed_)
39 {
40 if(play())
41 {
42 auto_play_consumed_ = true;
43 }
44 }
45
46 auto source = get_source();
47 if(!source)
48 {
49 return;
50 }
51
52 if(spatial_)
53 {
54 auto pos = t.get_position();
55 auto forward = t.z_unit_axis();
56 auto up = t.y_unit_axis();
57 source->set_position({{pos.x, pos.y, pos.z}});
58 source->set_orientation({{forward.x, forward.y, forward.z}}, {{up.x, up.y, up.z}});
59 }
60 else
61 {
62 // Listener-relative at the listener origin = non-spatial / UI / music.
63 source->set_position({{0.0f, 0.0f, 0.0f}});
64 }
65}

178auto audio_source_component::play() -> bool
179{
180 // Check the clip first: a source without a playable clip would hold a voice it never uses.
181 if(!sound_)
182 {
183 return false;
184 }
185 // get() hands back the shared empty clip for a failed load.
186 auto clip = sound_.get();
187 if(!clip || !clip->is_valid())
188 {
189 return false;
190 }
191
192 if(!voice_.is_alive() && !create_source())
193 {
194 return false;
195 }
196
197 auto source = get_source();
198 source->bind(*clip);
199 source->play();
200 return true;
201}

354auto audio_source_component::create_source() -> bool
355{
356 auto pool = find_voice_pool();
