196auto write_text_file(const fs::path& dst, const std::string& content, std::string* error) -> bool
197{
198 fs::error_code err;
199 const auto absolute = fs::absolute(dst);
200 fs::create_directories(absolute.parent_path(), err);
201 err.clear();
202
203 try
204 {
205 asset_writer::atomic_write_file(
206 absolute,
207 [&](const fs::path& temp)
208 {
209 std::ofstream output(temp, std::ios::binary | std::ios::trunc);
210 if(!output.is_open())
211 {
212 throw std::runtime_error("Failed to open temp file for write: " + temp.generic_string());
213 }
214 output.write(content.data(), static_cast<std::streamsize>(content.size()));
215 if(!output.good())

227 if(err)

427auto restore_from_trash(const std::string& batch, std::vector<fs::path>& restored, std::string* error) -> bool
428{
429 const auto batch_root = get_trash_directory() / batch;
430 fs::error_code ec;
431 if(batch.empty() || batch.find_first_of("/\\") != std::string::npos || batch.find("..") != std::string::npos ||
432 !fs::is_directory(batch_root, ec))
433 {
434 return report_error(error, "No trash batch " + batch);
435 }
436 std::vector<fs::path> files;
437 std::vector<fs::path> folders;
438 const auto meta_format = ex::get_meta_format();
439 for(fs::recursive_directory_iterator it(batch_root, ec), end; it != end && !ec; it.increment(ec))
440 {
441 fs::error_code entry_ec;
442 if(it->is_directory(entry_ec))
443 {
444 folders.push_back(it->path());
445 }
446 else if(it->path().extension().generic_string() != meta_format)
447 {
448 files.push_back(it->path());
449 }
450 }
451 const auto data_root = get_data_directory();
452 std::vector<std::string> taken;
453 {
454 fs::watcher::scoped_pause pause;
455 for(const auto& folder : folders)
456 {
457 fs::create_directories(data_root / folder.lexically_relative(batch_root), ec);
458 }
459 for(const auto& file : files)
460 {
461 const auto target = data_root / file.lexically_relative(batch_root);
462 fs::error_code exists_ec;
463 std::string move_error;
464 if(fs::exists(target, exists_ec) || !move_with_meta(file, target, &move_error))
465 {
466 taken.push_back(file.lexically_relative(batch_root).generic_string());
467 continue;
468 }
469 restored.push_back(target);
470 }
471 }
472 if(!taken.empty())
473 {
474 return report_error(error, fmt::format("Left in the trash, their place is taken: {}", fmt::join(taken, ", ")));
475 }
476 fs::remove_all(batch_root, ec);
477 return true;
478}
