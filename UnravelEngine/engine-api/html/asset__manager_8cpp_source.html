217void asset_manager::save_database(const std::string& protocol, const fs::path& path, const database_filter& include)
218{
219 std::lock_guard<std::mutex> lock(db_mutex_);
220 auto& db = get_database(protocol);
221 if(!include)
222 {
223 save_to_file(path.string(), db);
224 return;
225 }
226 asset_database::database_t kept;

285auto asset_manager::get_metadata_for_key(const std::string& key) const -> asset_database::meta
286{
287 // `databases_` is keyed by *protocol* (e.g. "app"), not by full asset key.
288 // The asset's metadata lives inside the database for its protocol, so we
289 // need to extract the protocol from the key before doing the map lookup.
290 std::lock_guard<std::mutex> lock(db_mutex_);
291 const auto protocol = fs::extract_protocol(fs::path(key)).generic_string();
292 const auto it = databases_.find(protocol);
293 if(it == databases_.cend())
294 {
295 static const asset_database::meta empty;
296 return empty;
297 }
298 return it->second.get_metadata(key);
299}

302{
303 asset_meta meta;
