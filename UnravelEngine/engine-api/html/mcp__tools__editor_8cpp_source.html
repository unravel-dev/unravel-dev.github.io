13auto level_to_string(level::level_enum level) -> const char*
14{
15 switch(level)
16 {
17 case level::trace:
18 return "trace";
19 case level::debug:
20 return "debug";
21 case level::info:
22 return "info";
23 case level::warn:
24 return "warn";
25 case level::err:
26 return "error";
27 case level::critical:
28 return "critical";
29 default:
30 return "off";
31 }
32}

65 }
