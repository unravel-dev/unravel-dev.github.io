65void intersect_packet(RTCScene scene, hpp::span<const ray> rays, hpp::span<ray_hit> hits)
66{
67 alignas(32) std::array<int, triangle_scene::packet_size> valid{};
68 RTCRayHit8 query{};
69 for(size_t i = 0; i < triangle_scene::packet_size; ++i)
70 {
71 query.hit.geomID[i] = RTC_INVALID_GEOMETRY_ID;
72 query.hit.instID[0][i] = RTC_INVALID_GEOMETRY_ID;
73 if(i >= rays.size())
74 {
75 continue;
76 }
77 const ray& lane = rays[i];
78 valid[i] = -1;
79 query.ray.org_x[i] = lane.origin.x;
80 query.ray.org_y[i] = lane.origin.y;
81 query.ray.org_z[i] = lane.origin.z;
82 query.ray.dir_x[i] = lane.direction.x;
83 query.ray.dir_y[i] = lane.direction.y;
84 query.ray.dir_z[i] = lane.direction.z;
85 query.ray.tnear[i] = lane.t_near;
86 query.ray.tfar[i] = lane.t_far;
87 query.ray.mask[i] = 0xFFFFFFFFu;
88 }
89 RTCIntersectArguments arguments;
90 rtcInitIntersectArguments(&arguments);
91 rtcIntersect8(valid.data(), scene, &query, &arguments);
92 for(size_t i = 0; i < rays.size(); ++i)
93 {
94 ray_hit& hit = hits[i];
95 hit = {};
96 hit.t = rays[i].t_far;
97 if(query.hit.geomID[i] != RTC_INVALID_GEOMETRY_ID)
98 {
99 hit.t = query.ray.tfar[i];
100 hit.triangle = query.hit.primID[i];
101 hit.is_hit = true;
102 }
103 }
104}

118auto distance_squared_between_bounds(const math::bbox& a, const math::bbox& b) -> float
119{
120 const math::vec3 gap = math::max(math::max(a.min - b.max, b.min - a.max), math::vec3(0.0f));
121 return math::dot(gap, gap);
122}

126auto collect_triangle(RTCPointQueryFunctionArguments* arguments) -> bool
127{
128 auto* collection = static_cast<triangle_collection*>(arguments->userPtr);
129 const uint32_t capacity = uint32_t(collection->out.size());
130 if(collection->count > capacity)
131 {
132 return false;
133 }
134 const uint32_t triangle = arguments->primID;
135 math::bbox bounds;
136 bounds.reset();
137 for(uint32_t corner = 0; corner < 3; ++corner)
138 {
139 const float* vertex = collection->vertices + size_t(collection->triangles[size_t(triangle) * 3 + corner]) * 3;
140 bounds.add_point(math::vec3(vertex[0], vertex[1], vertex[2]));
141 }
142 if(distance_squared_between_bounds(bounds, collection->region) > collection->reach_squared)
143 {
144 return false;
145 }
146 // A high quality build splits large triangles across leaves, so the same triangle can come back.
147 const auto found = collection->out.begin() + collection->count;
148 if(std::find(collection->out.begin(), found, triangle) != found)
149 {
150 return false;
151 }
152 if(collection->count == capacity)
153 {
154 // Full: flag the overflow and shrink the query to nothing, which ends the traversal's descent.
155 collection->count = capacity + 1;
156 arguments->query->radius = 0.0f;
157 return true;
158 }
159 collection->out[collection->count++] = triangle;
160 return false;
161}

179 auto build(const std::vector<math::vec3>& positions, const std::vector<uint32_t>& indices) -> bool override
180 {
181 const size_t triangle_count = indices.size() / 3;
182 RTCDevice device = get_device();
183 if(device == nullptr)
184 {
185 return false;
186 }
187 RTCScene scene = rtcNewScene(device);
188 // A bake queries one scene millions of times, so the slower, tighter build pays for itself.
189 rtcSetSceneBuildQuality(scene, RTC_BUILD_QUALITY_HIGH);
190 RTCGeometry geometry = rtcNewGeometry(device, RTC_GEOMETRY_TYPE_TRIANGLE);
191 rtcSetGeometryBuildQuality(geometry, RTC_BUILD_QUALITY_HIGH);
192 auto* vertices = static_cast<float*>(rtcSetNewGeometryBuffer(geometry,
193 RTC_BUFFER_TYPE_VERTEX,
194 0,
195 RTC_FORMAT_FLOAT3,
196 3 * sizeof(float),
197 positions.size()));
198 auto* triangles = static_cast<uint32_t*>(rtcSetNewGeometryBuffer(geometry,
199 RTC_BUFFER_TYPE_INDEX,
200 0,
201 RTC_FORMAT_UINT3,
202 3 * sizeof(uint32_t),
203 triangle_count));
204 if(vertices == nullptr || triangles == nullptr)
205 {
206 rtcReleaseGeometry(geometry);
207 rtcReleaseScene(scene);
208 return false;
209 }
210 for(size_t i = 0; i < positions.size(); ++i)
211 {
212 vertices[i * 3 + 0] = positions[i].x;
213 vertices[i * 3 + 1] = positions[i].y;
214 vertices[i * 3 + 2] = positions[i].z;
215 }
216 std::copy(indices.begin(), indices.begin() + std::ptrdiff_t(triangle_count * 3), triangles);
217 // Lets a query pass its own filter (skip_triangle_filter) through its arguments.
218 rtcSetGeometryEnableFilterFunctionFromArguments(geometry, true);
219 rtcCommitGeometry(geometry);
220 rtcAttachGeometry(scene, geometry);
221 rtcReleaseGeometry(geometry);
222 rtcCommitScene(scene);
223 if(rtcGetDeviceError(device) != RTC_ERROR_NONE)
224 {
225 rtcReleaseScene(scene);
226 return false;
227 }
228 scene_ = scene;
229 vertices_ = vertices;
230 triangles_ = triangles;
231 return true;
232 }

274 void intersect(hpp::span<const ray> rays, hpp::span<ray_hit> hits) const override
275 {
276 for(size_t first = 0; first < rays.size(); first += triangle_scene::packet_size)
277 {
278 const size_t count = std::min<size_t>(triangle_scene::packet_size, rays.size() - first);
279 if(count == 1)
280 {
281 // A lone ray is cheaper on its own than as a packet of one.
282 const ray& lone = rays[first];
283 hits[first] =
284 intersect(lone.origin, lone.direction, lone.t_near, triangle_scene::invalid_triangle, lone.t_far);
285 continue;
286 }
287 intersect_packet(scene_, rays.subspan(first, count), hits.subspan(first, count));
288 }
289 }

291 auto collect_triangles(const math::bbox& region, float reach, hpp::span<uint32_t> out) const -> uint32_t override
292 {
293 triangle_collection collection;
294 collection.vertices = vertices_;
295 collection.triangles = triangles_;
296 collection.region = region;
297 collection.reach_squared = reach * reach;
298 collection.out = out;
299 // The sphere around the region's centre that holds every point within reach of the region.
300 const math::vec3 center = region.get_center();
301 RTCPointQuery query{};
302 query.x = center.x;
303 query.y = center.y;
304 query.z = center.z;
305 query.radius = math::length(region.max - center) + reach;
306 RTCPointQueryContext context;
307 rtcInitPointQueryContext(&context);
308 rtcPointQuery(scene_, &query, &context, collect_triangle, &collection);
309 return collection.count;
310 }
