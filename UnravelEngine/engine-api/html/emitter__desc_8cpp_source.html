19auto is_constant_channel(const math::curve_channel& channel, float value) -> bool
20{
21 const auto& keys = channel.get_keys();
22 const bool has_smooth_key = std::any_of(keys.begin(),
23 keys.end(),
24 [](const math::curve_key& key)
25 {
26 return key.interpolation == math::curve_interpolation::smooth;
27 });
28 return std::all_of(keys.begin(),
29 keys.end(),
30 [&](const math::curve_key& key)
31 {
32 const bool is_level = !has_smooth_key || (math::abs(key.in_tangent) <= k_constant_tolerance &&
33 math::abs(key.out_tangent) <= k_constant_tolerance);
34 return is_level && math::abs(key.value - value) <= k_constant_tolerance;
35 });
36}

374{

391auto emitter_sim_state::calculate_smoothed_emitter_speed(float max_speed) const -> float
392{
393 if(temporal_count < temporal_buffer_size)
394 {
395 return max_speed;
396 }
397 float total_distance = 0.0f;
398 float total_time = 0.0f;
399 for(uint32_t i = 1; i < temporal_count; ++i)
400 {
401 const math::vec3 delta = temporal_positions[i] - temporal_positions[i - 1];
402 total_distance += math::length(delta);
403 total_time += temporal_dts[i];
404 }
405 if(total_time > 0.0f)
406 {
407 return total_distance / total_time;
408 }
409 return 0.0f;
410}
