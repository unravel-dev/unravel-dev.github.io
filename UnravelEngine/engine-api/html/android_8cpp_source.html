60auto read_stamp(const fs::path& file) -> std::map<std::string, std::string>
61{
62 std::map<std::string, std::string> values;
63 std::ifstream stream(file);
64 std::string line;
65 while(std::getline(stream, line))
66 {
67 if(!line.empty() && line.back() == '\r')
68 {
69 line.pop_back();
70 }
71 const auto separator = line.find('=');
72 if(separator != std::string::npos)
73 {
74 values[line.substr(0, separator)] = line.substr(separator + 1);

76 }

97auto make_id_part(const std::string& text, const char* fallback) -> std::string
98{
99 std::string part;
100 for(const char c : text)
101 {
102 const auto code = static_cast<unsigned char>(c);
103 const bool is_kept = code < 0x80 && std::isalnum(code) != 0;
104 const char mapped = is_kept ? static_cast<char>(std::tolower(code)) : '_';
105 if(mapped == '_' && (part.empty() || part.back() == '_'))
106 {
107 continue;
108 }
109 part += mapped;
110 }
111 while(!part.empty() && part.back() == '_')
112 {
113 part.pop_back();
114 }
115 if(part.empty())
116 {
117 return fallback;
118 }
119 if(!is_letter(part.front()))
120 {
121 part.insert(0, LEADING_LETTERS);
122 }
123 return part;
124}

411auto parse_android_devices(const std::string& adb_output) -> std::vector<android_device>
412{
413 std::vector<android_device> devices;
414 std::stringstream lines(adb_output);
415 std::string line;
416 while(std::getline(lines, line))
417 {
418 std::stringstream words(line);
419 std::vector<std::string> tokens;
420 for(std::string word; words >> word;)
421 {
422 tokens.push_back(word);
423 }
424 // The serial can hold spaces, so the state word ends it. The header and the server's start
425 // messages ("* daemon started successfully") have none.
426 const auto state = tokens.empty() ? tokens.end() : std::find_if(tokens.begin() + 1, tokens.end(), is_device_state);
427 if(state == tokens.end())
428 {
429 continue;
430 }
431 android_device device;
432 for(auto it = tokens.begin(); it != state; ++it)
433 {
434 device.serial += device.serial.empty() ? *it : " " + *it;
435 }
436 device.state = *state == NO_PERMISSIONS_STATE_WORD ? NO_PERMISSIONS_STATE : *state;
437 for(auto it = state + 1; it != tokens.end(); ++it)
438 {
439 if(it->rfind("model:", 0) == 0)
440 {
441 device.model = it->substr(6);
442 std::replace(device.model.begin(), device.model.end(), '_', ' ');
443 }
444 }
445 devices.push_back(std::move(device));
446 }
447 return devices;
448}

450auto parse_logcat_line(const std::string& text, logcat_line& line) -> bool
451{
452 // "10-18 20:44:29.426 25848 25903 I spdlog : message"
453 std::stringstream words(text);
454 std::string date;
455 std::string time;
456 std::string thread;
457 std::string level;
458 if(!(words >> date >> time >> line.pid >> thread >> level) || level.size() != 1 || date.size() != 5)
459 {
460 return false;
461 }
462 line.level = level.front();
463 const auto rest_begin = static_cast<std::size_t>(words.tellg());
464 const auto separator = text.find(':', rest_begin);
465 if(separator == std::string::npos)
466 {
467 return false;
468 }
469 line.tag = text.substr(rest_begin, separator - rest_begin);
470 const auto tag_begin = line.tag.find_first_not_of(' ');
471 const auto tag_end = line.tag.find_last_not_of(' ');
472 line.tag = tag_begin == std::string::npos ? std::string() : line.tag.substr(tag_begin, tag_end - tag_begin + 1);
473 const auto message_begin = separator + 1 < text.size() && text[separator + 1] == ' ' ? separator + 2 : separator + 1;
474 line.message = message_begin < text.size() ? text.substr(message_begin) : std::string();
475 return true;
476}

564{

572 tools.adb = sdk / "platform-tools" / adb_name;

574 }

576 const auto java_name = get_executable_name("java");
577 for(const auto& jdk : search.jdk_candidates)

582 break;
583 }
584 }
585 if(tools.sdk.empty())

587 tools.error = "No Android SDK was found. Install Android Studio, or set ANDROID_HOME to the SDK folder.";

590 {
591 tools.error = "No Java was found. Install Android Studio, which bundles one, or set JAVA_HOME.";
