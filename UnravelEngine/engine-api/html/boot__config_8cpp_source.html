203auto resolve_boot_config(const cmd_line::parser& parser, const boot_config& project_hint) -> boot_config
204{
205 boot_config result = project_hint;
206 std::string preferred_renderer_arg;
207 if(parser.try_get("renderer", preferred_renderer_arg) && !preferred_renderer_arg.empty())
208 {
209 const preferred_renderer from_cli = preferred_renderer_from_string(preferred_renderer_arg);
210 if(preferred_renderer_arg != "auto" && from_cli != preferred_renderer::auto_detect)
211 {
212 result.cli.renderer = true;
213 result.renderer = from_cli;
214 }
215 }
216 std::string physics_arg;
217 if(parser.try_get("physics", physics_arg) && !physics_arg.empty() && physics_arg != "auto")
218 {
219 result.cli.physics = true;
220 result.physics = physics_backend_from_string(physics_arg);
221 }
222 return result;
223}

225auto peek_project_boot_config(const fs::path& project_path) -> boot_config
226{
227 const fs::path settings_path = project_path / "settings" / "settings.cfg";
228 fs::error_code err;
229 if(!fs::exists(settings_path, err))
230 {
231 return {};
232 }
233 // Boot configuration is peeked before any system is initialized, so only the cold
234 // sections may be read: a full settings load resolves asset handles through the
235 // asset manager, whose per-type storages do not exist until asset_manager::init.
236 settings project_settings{};
237 if(!load_boot_sections_from_file(settings_path.string(), project_settings))
238 {
239 APPLOG_WARNING("Failed to peek project settings at {}", settings_path.string());
240 return {};
241 }
242 return boot_config_from_settings(project_settings);
243}

245auto boot_config_requires_restart(const boot_config& active, const boot_config& project) -> bool
246{
247 boot_config effective = project;
248 if(active.cli.renderer)
249 {
250 effective.renderer = active.renderer;
251 }
252 if(active.cli.physics)
253 {
254 effective.physics = active.physics;
255 }
256 // Auto and an explicit choice of the same engine boot the same backend; do not
257 // restart over the spelling.
258 effective.physics = resolve_physics_backend(effective.physics);
259 boot_config resolved_active = active;
260 resolved_active.physics = resolve_physics_backend(active.physics);
261 return resolved_active != effective;
262}
