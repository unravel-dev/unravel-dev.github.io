21auto json_escape(const std::string& value) -> std::string
22{
23 std::string out;
24 out.reserve(value.size() + 8);
25 for(char c : value)
26 {
27 switch(c)
28 {
29 case '\\':
30 out += "\\\\";
31 break;
32 case '"':
33 out += "\\\"";
34 break;
35 case '\b':
36 out += "\\b";
37 break;
38 case '\f':
39 out += "\\f";
40 break;
41 case '\n':
42 out += "\\n";
43 break;
44 case '\r':
45 out += "\\r";
46 break;
47 case '\t':
48 out += "\\t";
49 break;
50 default:
51 out += c;
52 break;
53 }
54 }
55 return out;
56}
