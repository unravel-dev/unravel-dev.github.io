25auto json_escape(const std::string& value) -> std::string
26{
27 std::string out;
28 out.reserve(value.size() + 8);
29 for(char c : value)
30 {
31 switch(c)
32 {
33 case '\\':
34 out += "\\\\";
35 break;
36 case '"':
37 out += "\\\"";
38 break;
39 case '\b':
40 out += "\\b";
41 break;
42 case '\f':
43 out += "\\f";
44 break;
45 case '\n':
46 out += "\\n";
47 break;
48 case '\r':
49 out += "\\r";
50 break;
51 case '\t':
52 out += "\\t";
53 break;
54 default:
55 out += c;
56 break;
57 }
58 }
59 return out;
60}

92auto find_entity_by_id(scene& scn, const std::string& id) -> entt::handle
93{
94 if(id.empty() || !scn.registry)
95 {
96 return {};
97 }
98 // A uuid that matches nothing is a stale id: it names no entity, whatever its leading digits read as.
99 if(auto uuid = hpp::uuid::from_string(id))
100 {
101 return scn.find_entity_by_uuid(*uuid);
102 }
103 // Entities without an id component are addressed by their entt value, all digits.
104 std::uint32_t value{};
105 const auto* const end = id.data() + id.size();
106 const auto [ptr, error] = std::from_chars(id.data(), end, value);
107 const auto integral = static_cast<entt::entity>(value);
108 if(error != std::errc{} || ptr != end || !scn.registry->valid(integral))
109 {
110 return {};
111 }
112 return entt::handle(*scn.registry, integral);
113}

268 size_t& nodes_emitted,

303 continue;
304 }
305 if(!first)
306 {
307 children_json += ",";
308 }
309 first = false;
310 children_json += child_json;

313 }
314 }
