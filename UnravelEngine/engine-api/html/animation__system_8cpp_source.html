37auto get_node_parent_frame(const model_component& model, entt::handle model_entity, int node_index) -> math::mat4
38{
39 math::mat4 frame(1.0f);
40 const auto node = node_index >= 0 ? model.get_armature_by_index(size_t(node_index)) : entt::handle{};
41 if(!node)
42 {
43 return frame;
44 }
45 for(entt::handle parent = node.get<transform_component>().get_parent(); parent && parent != model_entity;)
46 {
47 const auto& parent_transform = parent.get<transform_component>();
48 frame = parent_transform.get_transform_local().get_matrix() * frame;
49 parent = parent_transform.get_parent();
50 }
51 return frame;
52}

55auto get_frame_rotation(const math::mat4& frame) -> math::quat
56{
57 math::mat3 axes(frame);
58 for(int i = 0; i < 3; ++i)
59 {
60 const float length = math::length(axes[i]);
61 axes[i] = length > 0.0f ? axes[i] / length : axes[i];
62 }
63 return math::normalize(glm::quat_cast(axes));
64}

82void move_by_root_motion(const animation_pose::root_motion_result& motion,
83 const model_component& model,
84 entt::handle model_entity,
85 transform_component& entity_transform,
86 character_controller_component* character)
87{
88 math::vec3 move(0.0f);
89 if(motion.root_position_node_index >= 0)
90 {
91 const math::vec3 delta =
92 math::mix(math::vec3(0.0f), motion.root_transform_delta.get_translation(), motion.root_position_weights);
93 move = math::mat3(get_node_parent_frame(model, model_entity, motion.root_position_node_index)) * delta;
94 }
95 math::quat turn = math::identity<math::quat>();
96 if(motion.root_rotation_node_index >= 0)
97 {
98 const math::quat frame = get_frame_rotation(get_node_parent_frame(model, model_entity, motion.root_rotation_node_index));
99 const math::quat delta =
100 math::slerp(math::identity<math::quat>(), motion.root_transform_delta.get_rotation(), motion.root_rotation_weight);
101 turn = math::normalize(frame * delta * glm::inverse(frame));
102 }
103 if(!is_finite(move) || !is_finite(turn))
104 {
105 return;
106 }
107 // The step is in the entity's frame before this frame's turn, so it moves first.
108 if(character != nullptr)
109 {
110 character->move(math::mat3(entity_transform.get_transform_global().get_matrix()) * move);
111 }
112 else
113 {
114 entity_transform.set_position_local(entity_transform.get_position_local() +
115 entity_transform.get_rotation_local() * (entity_transform.get_scale_local() * move));
116 }
117 entity_transform.set_rotation_local(math::normalize(entity_transform.get_rotation_local() * turn));
118}

141void describe_skeleton(const model_component& model, animation_skeleton_desc& desc)
142{
143 const auto& armature = model.get_armature_entities();
144 desc.names.resize(armature.size());
145 desc.parents.assign(armature.size(), -1);
146 std::unordered_map<entt::entity, int> index_of;

148 for(size_t i = 0; i < armature.size(); ++i)
149 {

152 index_of[armature[i].entity()] = int(i);

155 desc.names[i] = tag->name;
156 }
157 }
158 }

198auto animation_system::make_skeleton_binding(animation_component& animation, const model_component& model)
199 -> animation_skeleton_binding
200{
201 animation_skeleton_binding binding;
202 if(animation.get_retargeting_mode() != animation_retargeting_mode::name_based)
203 {
204 return binding;
205 }
206 binding.version = get_skeleton_version(model) * 2;
207 binding.resolve = [&model](const animation_channel& channel)
208 {
209 const int index = model.get_armature_index_by_name_cached(channel.node_name);
210 // Rigs of one family prefix their bones differently ("mixamorig:" and "mixamorig1:").
211 return index >= 0 ? index : model.get_armature_index_by_base_name_cached(channel.node_name);
212 };
213 if(animation.get_pose_retargeting() == animation_pose_retargeting::rest_aligned)
214 {
215 // A changed retarget pose rebuilds the skeleton and every binding made against it.
216 size_t version = size_t(binding.version + 1);
217 utils::hash_combine(version, animation.get_retarget_revision());
218 binding.version = version;
219 binding.rest_aligned_target = get_target_skeleton(animation, model);
220 }
221 return binding;
222}

302void animation_system::on_update(scene& scn, delta_t dt)
303{
304 APP_SCOPE_PERF("Animation/System Update");
305
306 static const std::string thread_name = "Animation/System Update Thread";
307 auto& ctx = engine::context();
308 auto& th = ctx.get_cached<threader>();
309 // Create a view for entities with transform_component and submesh_component
310 auto view = scn.registry->view<model_component, animation_component, transform_component>();
311
312 // Pose application writes armature transforms, and transform dirty flags
313 // propagate down the hierarchy. Two animated entities whose hierarchies
314 // overlap (e.g. an animated prop parented under another character's bone)
315 // must therefore be updated by the same task. Tag every animated entity
316 // with its topmost animated ancestor and turn runs of equal roots into
317 // tasks - disjoint hierarchies (the common case) still run fully in
318 // parallel. The scratch buffers are members reused across frames, so the
319 // steady state performs no heap allocations here.
320 grouping_scratch_.clear();
321 bool any_nested = false;
322 for(auto entity : view)
323 {
324 entt::entity group_root = entity;
325 auto parent = view.get<transform_component>(entity).get_parent();
326 // World transforms resolve lazily into caches: an ancestor several characters share is
327 // resolved here, on one thread, before the pool reads it (character controllers move in
328 // world space).
329 if(parent)
330 {
331 parent.get<transform_component>().get_transform_global();
332 }
333 while(parent)
334 {
335 if(parent.all_of<model_component, animation_component, transform_component>())
336 {
337 group_root = parent.entity();
338 }
339 parent = parent.get<transform_component>().get_parent();
340 }
341 any_nested |= group_root != entity;
342 grouping_scratch_.emplace_back(group_root, entity);
343 }
344
345 if(any_nested)
346 {
347 // Bring equal roots together. Entities can only share a root when
348 // nesting exists, so most frames skip the sort entirely.
349 std::sort(grouping_scratch_.begin(), grouping_scratch_.end());
350 }
351
352 group_ranges_.clear();
353 for(size_t begin = 0; begin < grouping_scratch_.size();)
354 {
355 size_t end = begin + 1;
356 while(end < grouping_scratch_.size() && grouping_scratch_[end].first == grouping_scratch_[begin].first)
357 {
358 ++end;
359 }
360 group_ranges_.emplace_back(begin, end);
361 begin = end;
362 }
363
364 // Character controllers only move while the simulation runs.
365 const bool is_simulating = ctx.has<play_mode>() && ctx.get_cached<play_mode>().is_active();
366
367 auto update_entity = [&](entt::entity entity)
368 {
369 APP_SCOPE_PERF_THREAD("Animation/Update Poses", "Pool Thread");
370
371 // This is needed as we call .get on the animations inside the player
372 tpp::this_thread::register_this_thread(thread_name, true);
373
374 auto& animation_comp = view.get<animation_component>(entity);
375 auto& model_comp = view.get<model_component>(entity);
376
377 bool should_update_poses = true;
378 if(animation_comp.get_culling_mode() == animation_component::culling_mode::renderer_based)
379 {
380 if(!model_comp.was_used_last_frame())
381 {
382 should_update_poses = false;
383 }
384 }
385
386 auto& player = animation_comp.get_player();
387
388 // Apply speed to delta time
389 auto speed = animation_comp.get_speed();
390 auto adjusted_dt = dt * speed;
391
392 // A controller decides what the player plays before time advances, so a
393 // transition it fires is visible in this frame's pose.
394 if(player.is_playing())
395 {
396 if(auto* controller = animation_comp.try_get_controller_instance())
397 {
398 controller->update(player,
399 model_comp.get_armature_version(),
400 [&model_comp](animation_skeleton_desc& desc)
401 {
402 describe_skeleton(model_comp, desc);
403 });
404 }
405 }
406
407 bool updated = player.update_time(adjusted_dt);
408
409 if(updated && should_update_poses)
410 {
411 auto& transform_comp = view.get<transform_component>(entity);
412
413 const bool apply_root_motion = animation_comp.get_apply_root_motion();
414 auto* character = is_simulating
415 ? scn.registry->try_get<character_controller_component>(entity)
416 : nullptr;
417
418 // Clips are sampled straight onto armature indices: name-based retargeting
419 // resolves each channel once per (clip, armature) instead of per bone per
420 // frame, and clips exported from different files blend bone by bone.
421 const animation_skeleton_binding binding = make_skeleton_binding(animation_comp, model_comp);
422
423 animation_pose::root_motion_result root_motion;
424 bool has_root_motion = false;
425 player.update_poses(
426 model_comp.get_bind_pose(),
427 binding,
428 apply_root_motion,
429 [&](const animation_pose::node_desc& desc,
430 const math::transform& transform,
431 const animation_pose::root_motion_result& motion_result)
432 {
433 auto armature = model_comp.get_armature_by_index(desc.index);
434 if(!armature)
435 {
436 APPLOG_WARNING("Cannot find armature with index {}", desc.index);
437 return;
438 }
439 auto& armature_transform_comp = armature.template get<transform_component>();
440
441 const bool is_root_position_node = int(desc.index) == motion_result.root_position_node_index;
442 const bool is_root_rotation_node = int(desc.index) == motion_result.root_rotation_node_index;
443 // A root node takes what the root motion does not own from the pose:
444 // a pelvis that only translates still turns with the animation.
445 math::transform local = transform;
446 if(apply_root_motion && is_root_position_node)
447 {
448 local.set_position(math::mix(armature_transform_comp.get_position_local(),
449 transform.get_position(),
450 motion_result.bone_position_weights));
451 }
452 if(apply_root_motion && is_root_rotation_node)
453 {
454 local.set_rotation(math::slerp(armature_transform_comp.get_rotation_local(),
455 transform.get_rotation(),
456 motion_result.bone_rotation_weight));
457 }
458 armature_transform_comp.set_transform_local(local);
459 if(is_root_position_node || is_root_rotation_node)
460 {
461 root_motion = motion_result;
462 has_root_motion = true;
463 }
464 });
465
466 // Once per frame, after the pose is written: the root node's parent frame
467 // is the posed one.
468 if(apply_root_motion && has_root_motion)
469 {
470 move_by_root_motion(root_motion, model_comp, entt::handle(*scn.registry, entity), transform_comp, character);
471 }
472 }
473 else if(updated)
474 {
475 // Poses were skipped: the root motion they missed is not made up in one jump
476 // when the character is seen again.
477 player.reset_root_motion();
478 }
479 };
480
481 // A vector of ranges, not a view: already random access, so the plain range form is the
482 // right one here (for_each_entity_par is for entt views, whose iterators are forward-only).
483 poolstl::for_each_par_if(true,
484 group_ranges_.begin(),
485 group_ranges_.end(),
486 [&](const std::pair<size_t, size_t>& range)
487 {
488 for(size_t i = range.first; i < range.second; ++i)
489 {
490 update_entity(grouping_scratch_[i].second);
491 }
492 });
493}
