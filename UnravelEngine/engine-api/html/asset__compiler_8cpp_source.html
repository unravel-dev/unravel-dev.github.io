363auto select_bc_format(bgfx::TextureFormat::Enum input_format,
364 const fs::path& extension,
365 texture_importer_meta::compression_quality quality) -> bgfx::TextureFormat::Enum
366{
367 if(input_format == bgfx::TextureFormat::BC1)
368 {
369 return bgfx::TextureFormat::BC3;
370 }
371
372 if(gfx::is_compressed_format(input_format))
373 {
374 return input_format;
375 }
376
377 auto info = gfx::get_format_info(input_format);
378
379 if(extension == ".hdr" || extension == ".exr")
380 {
381 info.is_hdr = true;
382 }
383
384 // 1) HDR? Use BC6H for color data, ignoring alpha (HDR with alpha is non-trivial).
385 if(info.is_hdr)
386 {
387 // BC6H: color (RGB) 16F
388 // No standard BC format for HDR alpha in the block-compression range.
389 return bgfx::TextureFormat::BC6H;
390 }
391
392 // 2) Single channel => BC4
393 // e.g., for grayscale height map or single-channel mask
394 if(info.num_channels == 1)
395 {
396 return bgfx::TextureFormat::BC4;
397 }
398
399 // 3) Two channel => BC5
400 // e.g., typical for 2D vector data, normal map XY
401 if(info.num_channels == 2)
402 {
403 return bgfx::TextureFormat::BC5;
404 }
405
406 // 4) If we reach here, we have 3 or 4 channels in LDR.
407
408 // 4a) No alpha needed => choose BC1 or BC7, etc.
409 if(!info.has_alpha_channel)
410 {
411 switch(quality)
412 {
413 case texture_importer_meta::compression_quality::low_quality:
414 // BC1 is cheap and has no alpha
415 return bgfx::TextureFormat::BC1;
416 case texture_importer_meta::compression_quality::normal_quality:
417 // BC1 is standard for color w/out alpha
418 return bgfx::TextureFormat::BC1;
419 case texture_importer_meta::compression_quality::high_quality:
420 // BC7 is higher quality for color, also supports alpha but not needed here.
421 // It is also really slow for encoding so don't use it for now.
422 return bgfx::TextureFormat::BC1;
423 default:
424 break;
425 }
426 // fallback
427 return bgfx::TextureFormat::BC1;
428 }
429 else
430 {
431 // 4b) We do have alpha => choose BC2, BC3, or BC7.
432 // BC2 (DXT3) is old and rarely used except for sharp alpha transitions.
433 // BC3 (DXT5) is the typical solution for alpha textures if BC7 is not an option.
434 // BC7 is better (but bigger decode cost).
435 switch(quality)
436 {
437 case texture_importer_meta::compression_quality::low_quality:
438 return bgfx::TextureFormat::BC3;
439 case texture_importer_meta::compression_quality::normal_quality:
440 return bgfx::TextureFormat::BC3; // DXT5
441 case texture_importer_meta::compression_quality::high_quality:
442 // BC7 is best BC for RGBA
443 // It is also really slow for encoding so don't use it for now.
444 return bgfx::TextureFormat::BC3;
445 default:
446 break;
447 }
448 // fallback
449 return bgfx::TextureFormat::BC3;
450 }
451
452 return input_format;
453}

519auto bake_normal_map_input_if_needed(const fs::path& input_path,
520 const texture_importer_meta& importer,
521 fs::path& temp_baked_path) -> fs::path
522{
523 temp_baked_path.clear();
524
525 if(!importer.invert_normal_y)
526 {
527 return input_path;
528 }
529
530 bimg::ImageContainer* image = imageLoad(bx::FilePath(input_path.string().c_str()));
531 if(nullptr == image)
532 {
533 APPLOG_ERROR("Failed to load texture for normal Y bake: {0}", input_path.string());
534 return input_path;
535 }
536
537 if(!imageFlipTangentSpaceNormalY(image))
538 {
539 bimg::imageFree(image);
540 APPLOG_ERROR("Failed to flip normal map Y for: {0}", input_path.string());
541 return input_path;
542 }
543
544 if(!imagePrepareNormalMapBakePng(image))
545 {
546 bimg::imageFree(image);
547 APPLOG_ERROR("Failed to prepare baked normal map PNG for: {0}", input_path.string());
548 return input_path;
549 }
550
551 fs::error_code err;
552 // Always feed texturec a lossless RGBA8 PNG (avoids broken BC re-encode and zero-alpha PNG previews).
553 // Unique per compile: textures that share a name, or one texture compiled for two targets, can bake at once.
554 temp_baked_path =
555 fs::temp_directory_path(err) / fmt::format("{}_normal_y_{}.png", input_path.stem().string(), hpp::to_string(generate_uuid()));
556 if(err || temp_baked_path.empty())
557 {
558 bimg::imageFree(image);
559 APPLOG_ERROR("Failed to resolve temp path for normal Y bake: {0}", input_path.string());
560 return input_path;
561 }
562
563 if(!imageSave(temp_baked_path.string().c_str(), image))
564 {
565 bimg::imageFree(image);
566 fs::remove(temp_baked_path, err);
567 temp_baked_path.clear();
568 APPLOG_ERROR("Failed to write baked normal map: {0}", input_path.string());
569 return input_path;
570 }
571
572 bimg::imageFree(image);
573 APPLOG_INFO("Baked invert normal Y for {0} -> {1}", input_path.filename().string(), temp_baked_path.filename().string());
574 return temp_baked_path;
575}

577auto compile_texture_to_file(const fs::path& input_path,
578 const fs::path& output_path,
579 const texture_importer_meta& importer,
580 const std::string& protocol,
581 const texture_compile_parameters& parameters) -> bool
582{
583 fs::path temp_baked_path;
584 const fs::path compile_input = bake_normal_map_input_if_needed(input_path, importer, temp_baked_path);
585 const bool using_temp_input = !temp_baked_path.empty();
586
587 std::string str_input = compile_input.string();
588 std::string str_output = output_path.string();
589
590 bool try_compress = protocol == "app";
591
592 const auto& quality = parameters;
593 const auto input_info = get_input_texture_info(compile_input, quality.max_size);
594
595 // Resolve the effective color space. Data-typed imports are always linear no
596 // matter the tag: normal maps are vectors, and float sources (.hdr/.exr) have
597 // no display encoding to begin with. Everything else honors the meta, with
598 // `automatic` meaning raw/linear (legacy behavior -- UI textures, icons and
599 // untagged content keep sampling exactly as before). One exception: LDR
600 // equirect imports are display-encoded by construction (authored/rendered sky
601 // panoramas feeding the skybox dome and the irradiance bake), and no other
602 // stage can tag them (the mesh importer only classifies material slots) -- so
603 // `automatic` resolves to sRGB for those. An explicit `linear` tag still wins.
604 const auto source_ext = string_utils::to_lower(compile_input.extension().string());
605 const bool source_is_float = source_ext == ".hdr" || source_ext == ".exr";
606 const bool is_normal_map = importer.type == texture_importer_meta::texture_type::normal_map;
607 const bool auto_srgb_equirect = importer.colorspace == texture_importer_meta::color_space::automatic &&
608 importer.type == texture_importer_meta::texture_type::equirect;
609 const bool resolved_srgb = (importer.colorspace == texture_importer_meta::color_space::srgb ||
610 auto_srgb_equirect) &&
611 !is_normal_map && !source_is_float;
612 // A texture sampled through an sRGB view needs a format that has one (select_srgb_format). The project's
613 // textures take the family's color format; the others keep their source's format, RGBA8 when it has none.
614 auto format = select_compressed_format(input_info, compile_input.extension(), parameters);
615 if(resolved_srgb)
616 {
617 format = select_srgb_format(format, parameters.family);
618 }
619 const bool needs_srgb_format = resolved_srgb && !try_compress && !has_srgb_view(input_info.format);
620
621 const bool needs_format_conversion = input_info.format != format;
622 const bool needs_downscale = !input_info.fits_max_size;
623 // Equirect must always run through texturec; a raw copy cannot produce a cubemap.
624 const bool needs_equirect_projection = importer.type == texture_importer_meta::texture_type::equirect;
625 // sRGB-tagged textures must run through texturec even when a raw copy would do:
626 // only the compiled KTX2 container carries the sRGB tag the runtime loader
627 // turns into BGFX_TEXTURE_SRGB. A raw-copied PNG cannot express it.
628 const bool needs_srgb_container = resolved_srgb;
629 // Raster sources (PNG, JPG, ...) parse with a single mip, so a raw copy would drop
630 // the requested chain. Containers that already carry mips are kept: texturec passes
631 // an authored chain through unchanged when -m is set, so running it adds nothing.
632 const bool needs_mip_generation = importer.generate_mipmaps && input_info.num_mips <= 1;
633
634 if(needs_format_conversion || needs_downscale || needs_equirect_projection || needs_srgb_container ||
635 needs_mip_generation)
636 {
637 if(needs_downscale && !needs_format_conversion)
638 {
639 APPLOG_INFO("Downscaling {0} ({1}x{2}) to fit max size {3}",
640 compile_input.filename().string(),
641 input_info.width,
642 input_info.height,
643 texture_size_to_pixel_limit(quality.max_size));
644 }
645
646 // KTX2 container: unlike legacy DDS fourcc, its data format descriptor
647 // carries the sRGB/linear transfer tag, which the runtime loader maps to
648 // BGFX_TEXTURE_SRGB (see loadTexture). texturec tags the container sRGB
649 // unless --linear is passed, so the flag below must mirror resolved_srgb.
650 std::vector<std::string> args_array = {
651 "-f",
652 str_input,
653 "-o",
654 str_output,
655 "--as",
656 "ktx2",
657 };
658
659 if(!resolved_srgb)
660 {
661 args_array.emplace_back("--linear");
662 }
663
664 if(try_compress)
665 {
666 const auto output_format = get_texturec_output_format(format);
667 args_array.emplace_back("-t");
668 args_array.emplace_back(gfx::to_string(output_format));
669
670 if(output_format == bgfx::TextureFormat::BC7 || output_format == bgfx::TextureFormat::BC6H)
671 {
672 APPLOG_INFO("Compressing to {0}. May take a while.", gfx::to_string(output_format));
673
674 args_array.emplace_back("-q");
675 args_array.emplace_back("fastest");
676 }
677 else if(needs_format_conversion
678 && quality.compression == texture_importer_meta::compression_quality::high_quality)
679 {
680 args_array.emplace_back("-q");
681 args_array.emplace_back("highest");
682 }
683 }
684 else if(needs_srgb_format)
685 {
686 args_array.emplace_back("-t");
687 args_array.emplace_back(gfx::to_string(bgfx::TextureFormat::RGBA8));
688 }
689
690 if(importer.generate_mipmaps)
691 {
692 args_array.emplace_back("-m");
693 }
694
695 append_texture_max_size_args(args_array, quality.max_size);
696
697 switch(importer.type)
698 {
699 case texture_importer_meta::texture_type::equirect:
700 {
701 args_array.emplace_back("--equirect");
702 break;
703 }
704
705 case texture_importer_meta::texture_type::normal_map:
706 {
707 // texturec's normal map mode stores an ASTC normal as XXXY (Y in alpha), and the shaders read
708 // the normal's XYZ, so ASTC normal maps are encoded as color.
709 if(parameters.family != texture_compression_family::astc)
710 {
711 args_array.emplace_back("--normalmap");
712 }
713 break;
714 }
715
716 default:
717 break;
718 }
719
720 std::string error;
721
722 // Create an empty file at the output location so the process can write to it
723 {
724 std::ofstream output_file(str_output);
725 (void)output_file;
726 }
727
728 auto texturec = fs::resolve_protocol("binary:/texturec");
729
730 // Run the texture compiler directly to the temporary output location
731 bool compiled = run_process(texturec.string(), args_array, false, error);
732 if(!compiled)
733 {
734 APPLOG_ERROR("Failed compilation of {0} with error: {1}", str_input, error);
735 fs::error_code remove_err;
736 fs::remove(str_output, remove_err);
737 if(using_temp_input)
738 {
739 fs::remove(temp_baked_path, remove_err);
740 }
741 return false;
742 }
743 }
744 else
745 {
746 copy_compiled_file(compile_input, output_path);
747 }
748
749 if(using_temp_input)
750 {
751 fs::error_code remove_err;
752 fs::remove(temp_baked_path, remove_err);
753 }
754
755
756 return true;
757}

759auto compile_shader_to_file(const fs::path& input_path,
760 const fs::path& output_path,
761 bgfx::RendererType::Enum renderer) -> bool
762{
763
764 std::string str_input = input_path.string();
765 std::string str_output = output_path.string();
766
767 std::string file = input_path.stem().string();
768 fs::path dir = input_path.parent_path();
769
770 fs::path include = fs::resolve_protocol("engine:/data/shaders");
771 std::string str_include = include.string();
772
773 fs::path varying = dir / (file + ".io");
774
775 fs::error_code err;
776 if(!fs::exists(varying, err))
777 {
778 varying = dir / "varying.def.io";
779 }
780 if(!fs::exists(varying, err))
781 {
782 varying = dir / "varying.def.sc";
783 }
784
785 std::string str_varying = varying.string();
786
787 std::string str_platform;
788 std::string str_profile;
789 std::string str_type;
790
791 bool optimize = true;
792// #if UNRAVEL_DEBUG
793// optimize = false;
794// #endif
795
796 std::string str_opt = optimize ? "3" : "0";
797
798 bool vs = hpp::string_view(file).starts_with("vs_");
799 bool fs = hpp::string_view(file).starts_with("fs_");
800 bool cs = hpp::string_view(file).starts_with("cs_");
801
802 if(renderer == bgfx::RendererType::Vulkan)
803 {
804 str_platform = "windows";
805 str_profile = "spirv";
806 }
807
808 if(renderer == bgfx::RendererType::Direct3D11 || renderer == bgfx::RendererType::Direct3D12)
809 {
810 str_platform = "windows";
811
812 if(vs || fs)
813 {
814 str_profile = "s_5_0";
815 if(renderer == bgfx::RendererType::Direct3D12)
816 {
817 str_profile = "s_6_0";
818 }
819 }
820 else if(cs)
821 {
822 str_profile = "s_5_0";
823 if(renderer == bgfx::RendererType::Direct3D12)
824 {
825 str_profile = "s_6_0";
826 }
827 str_opt = optimize ? "1" : "0";
828 }
829 }
830 else if(renderer == bgfx::RendererType::OpenGLES)
831 {
832 // OpenGL ES 3.0 is the bgfx baseline and compute needs ES 3.1. shaderc raises a
833 // vertex/fragment shader to ES 3.1 itself when it uses storage buffers or images.
834 str_platform = "android";
835 str_profile = cs ? "310_es" : "300_es";
836 }
837 else if(renderer == bgfx::RendererType::OpenGL)
838 {
839 // OpenGL 4.3 is the bgfx baseline: the backend runs every shader as GLSL 4.30.
840 str_platform = "linux";
841 str_profile = "430";
842 }
843 else if(renderer == bgfx::RendererType::Metal)
844 {
845 str_platform = "osx";
846 str_profile = "metal";
847 }
848
849 if(vs)
850 str_type = "vertex";
851 else if(fs)
852 str_type = "fragment";
853 else if(cs)
854 str_type = "compute";
855 else
856 str_type = "unknown";
857
858 std::vector<std::string> args_array = {
859 "-f",
860 str_input,
861 "-o",
862 str_output,
863 "-i",
864 str_include,
865 "--varyingdef",
866 str_varying,
867 "--type",
868 str_type,
869 "--define",
870 "BGFX_CONFIG_MAX_BONES=" + std::to_string(gfx::get_max_blend_transforms())
871 // "--Werror"
872 };
873
874 if(!str_platform.empty())
875 {
876 args_array.emplace_back("--platform");
877 args_array.emplace_back(str_platform);
878 }
879
880 if(!str_profile.empty())
881 {
882 args_array.emplace_back("-p");
883 args_array.emplace_back(str_profile);
884 }
885
886 if(!str_opt.empty())
887 {
888 args_array.emplace_back("-O");
889 args_array.emplace_back(str_opt);
890 }
891
892 std::string error;
893
894 // Create an empty file at the output location
895 {
896 std::ofstream output_file(str_output);
897 (void)output_file;
898 }
899
900 auto shaderc = fs::resolve_protocol("binary:/shaderc");
901
902 if(!run_process(shaderc.string(), args_array, true, error))
903 {
904 APPLOG_ERROR("Failed compilation of {0} -> {1} with error: {2}", str_input, output_path.filename().string(), error);
905 // Non-throwing: this runs inside atomic_write_file's noexcept callback, where any removal error would
906 // terminate the process. Whatever survives here, atomic_write_file's temp guard reclaims.
907 fs::remove(output_path, err);
908 return false;
909 }
910 return true;
911}

1000auto get_submesh_face_count(const mesh::load_data& data, uint32_t lod, size_t submesh_index) -> uint32_t
1001{
1002 if(lod == 0)
1003 {
1004 return data.submeshes[submesh_index].face_count;
1005 }

1018void build_submesh_gi_cards(mesh::load_data& data,
1019 const std::vector<importer::imported_material>& materials,
1020 uint32_t max_cards,
1021 uint32_t lod_index,
1022 const std::string& source_name)
1023{
1024 APP_SCOPE_PERF("Build Surface Cache Cards");
1025 const auto start = std::chrono::steady_clock::now();
1026 data.submesh_cards.assign(data.submeshes.size(), gi_mesh_cards{});
1027 const uint32_t lod = math::min<uint32_t>(lod_index, uint32_t(data.lods.size()));
1028 data.cards_lod_index = lod;
1029 std::atomic<uint64_t> card_count{0};
1030 const auto build_submesh = [&](size_t i)
1031 {
1032 const uint32_t data_group = data.submeshes[i].data_group_id;
1033 const auto& mat = data_group < materials.size() ? materials[data_group].mat : nullptr;
1034 const bool two_sided = mat && mat->get_cull_type() == cull_type::none;
1035 gi_mesh_cards& cards = data.submesh_cards[i];
1036 cards.is_mostly_two_sided = two_sided;
1037 if(i >= data.submesh_sdfs.size() || !data.submesh_sdfs[i].is_sampleable())
1038 {
1039 return;
1040 }
1041 sdf_source_geometry geometry;
1042 if(!extract_sdf_source_geometry(data, lod, i, geometry))
1043 {
1044 return;
1045 }
1046 build_gi_mesh_cards(geometry, two_sided, max_cards, cards);
1047 card_count += cards.cards.size();
1048 };
1049 // A submesh's build takes from milliseconds to tens of seconds, the most triangles the longest, so the builds
1050 // start in that order, each pool worker pulling the next. Each build runs serially.
1051 std::vector<uint32_t> face_counts(data.submeshes.size());
1052 for(size_t i = 0; i < face_counts.size(); ++i)
1053 {
1054 face_counts[i] = get_submesh_face_count(data, lod, i);
1055 }
1056 poolstl::for_each_costliest_first_par_if(true, face_counts, build_submesh);
1057 const auto elapsed = std::chrono::duration<double, std::milli>(std::chrono::steady_clock::now() - start);
1058 APPLOG_INFO("Built {0} surface cache cards for {1} ({2} submeshes, LOD {3}) in {4:.1f} ms",
1059 card_count.load(),
1060 source_name,
1061 data.submeshes.size(),
1062 lod,
1063 elapsed.count());
1064}

1323template<>
1324auto compile<mesh>(asset_manager& am, const fs::path& key, const fs::path& output, uint32_t flags) -> bool
1325{
1326 APP_SCOPE_PERF("Compile Mesh");
1327 // Try to import first.
1328 auto base_importer = read_importer<mesh>(am, key);
1329
1330 if(!base_importer)
1331 {
1332 return true;
1333 }
1334
1335 auto importer = std::static_pointer_cast<mesh_importer_meta>(base_importer);
1336
1337 auto absolute_path = resolve_input_file(key);
1338
1339 std::string str_input = absolute_path.string();
1340
1341 fs::path file = absolute_path.stem();
1342 fs::path dir = absolute_path.parent_path();
1343
1344 mesh::load_data data;
1345 std::vector<animation_clip> animations;
1346 std::vector<importer::imported_material> materials;
1347 std::vector<importer::imported_texture> textures;
1348 // load_mesh_data_from_file waits for multi-file companions (.gltf/.bin, .obj/.mtl)
1349 // before Assimp runs, so incomplete copies cannot produce empty mesh buffers.
1350 if(!importer::load_mesh_data_from_file(am, absolute_path, *importer, data, animations, materials, textures))
1351 {
1352 APPLOG_ERROR("Failed compilation of {0}", str_input);
1353 return false;
1354 }
1355 // Never write a manifest / "successful" compile for an empty mesh. That happens when a
1356 // multi-file source (.gltf + .bin) was observed before companions finished copying.
1357 if(data.vertex_data.empty())
1358 {
1359 APPLOG_ERROR("Failed compilation of {0}: imported mesh has no vertex data "
1360 "(source companions may still be incomplete)",
1361 str_input);
1362 return false;
1363 }
1364 // IMPORTANT:
1365 // For skinned meshes, the skin binding step can duplicate vertices and rewrite triangle indices
1366 // to ensure a consistent bone palette per submesh. LODs must be generated AFTER this rewrite,
1367 // otherwise the stored LOD index buffers will reference the wrong vertices at runtime.
1368 if(data.skin_data.has_bones())
1369 {
1370 APP_SCOPE_PERF("Apply Skin to Load Data");
1371 if(!mesh::apply_skin_to_load_data(data))
1372 {
1373 APPLOG_ERROR("Failed to apply skinning data before generating LODs for {0}", str_input);
1374 return false;
1375 }
1376 }
1377 // Generate LODs offline during compilation (no GPU buffers created)
1378 if(importer->model.generate_lods)
1379 {
1380 // Use custom LOD configs if provided, otherwise use defaults
1381 auto lod_configs = mesh::generate_default_lod_configs(data, importer->model.lod_target_error);
1382 if(!lod_configs.empty())
1383 {
1384 APP_SCOPE_PERF("Generate LODs for Load Data");
1385 mesh::generate_lods_for_load_data(data, lod_configs);
1386 }
1387 }
1388 // Bake the GI distance field from the final LOD0 topology. This runs after skinning and
1389 // LOD generation because both can rewrite the vertex and index buffers; baking earlier
1390 // would produce a field that no longer matches the geometry that ships.
1391 if(importer->sdf.generate_sdf)
1392 {
1393 APPLOG_INFO("Baking SDF for {0}", str_input);
1394 APP_SCOPE_PERF("Bake Mesh SDF");
1395 mesh_sdf_bake_settings sdf_settings;
1396 sdf_settings.target_voxel_size = importer->sdf.target_voxel_size;
1397 sdf_settings.resolution = importer->sdf.resolution;
1398 sdf_settings.min_voxel_size = importer->sdf.min_voxel_size;
1399 sdf_settings.max_voxel_size = importer->sdf.max_voxel_size;
1400 sdf_settings.max_total_voxels = importer->sdf.max_total_voxels;

2445auto parse_compilation_entries(const std::string& log, hpp::string_view severity)
2446 -> std::vector<script_compilation_entry>
2447{
2448 // Path cannot contain '(' or newlines; keeps matching linear.
2449 const std::string pattern =
2450 std::string(R"(^([^\n(]+)\‍((\d+),\d+\):\s*)") + std::string(severity) + R"(\b.*)";
2451 const std::regex entry_regex(pattern, std::regex::ECMAScript | std::regex::optimize);
2452
2453 std::vector<script_compilation_entry> entries;
2454 std::istringstream stream(log);
2455 std::string line;
2456 while(std::getline(stream, line))
2457 {
2458 if(!line.empty() && line.back() == '\r')
2459 {
2460 line.pop_back();
2461 }
2462 std::smatch match;
2463 if(!std::regex_match(line, match, entry_regex) || match.size() < 3)
2464 {
2465 continue;
2466 }
2467 script_compilation_entry entry;
2468 entry.file = match[1].str();
2469 entry.line = std::stoi(match[2].str());
2470 entry.msg = match[0].str();
2471 entries.emplace_back(std::move(entry));
2472 }
2473 return entries;
2474}
