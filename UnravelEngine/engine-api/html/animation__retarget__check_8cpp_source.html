47auto count_role_matches_missed(const std::shared_ptr<animation_clip>& clip,
48 animation_skeleton_binding binding,
49 const animation_retarget_check& check) -> size_t
50{
51 binding.version += 1;
52 binding.rest_aligned_target = check.skeleton;
53 const auto aligned = animation_player::describe_binding(clip, binding, *check.skeleton);
54 const auto matched = size_t(std::count_if(aligned.channels.begin(),
55 aligned.channels.end(),
56 [](const animation_binding_report::channel_entry& channel)
57 {
58 return channel.match != match_kind::none;
59 }));
60 const size_t current = count_matched_channels(check);
61 return matched > current ? matched - current : 0;
62}

64void add_setting_warnings(const animation_component& animation,
65 size_t role_matches_missed,
66 animation_retarget_check& check)
67{
68 const bool wants_rest_aligned = animation.get_pose_retargeting() == animation_pose_retargeting::rest_aligned;
69 if(wants_rest_aligned && animation.get_retargeting_mode() != animation_retargeting_mode::name_based)
70 {
71 check.warnings.emplace_back("Rest Aligned needs Retargeting Mode Name Based: Index Based plays clips by node "
72 "order, copied.");
73 }
74 else if(wants_rest_aligned && !check.report.has_source_rest)
75 {
76 check.warnings.emplace_back("The clip recorded no rest pose (imported before rest poses were kept), so it "
77 "plays copied. Reimport its model to retarget it.");
78 }
79 if(!wants_rest_aligned && check.report.rest_mismatches > 0)
80 {
81 check.warnings.push_back(std::to_string(check.report.rest_mismatches) +
82 " bones rest differently in the clip than on this model: it was made on another "
83 "skeleton or another export. Copy keeps that skeleton's bone axes and lengths; set "
84 "Pose Retargeting to Rest Aligned.");
85 }
86 if(role_matches_missed > 0)
87 {
88 check.warnings.push_back(std::to_string(role_matches_missed) +
89 " more channels would reach a bone with Pose Retargeting Rest Aligned, which also "
90 "matches bones by humanoid role where names differ.");
91 }

94void add_binding_warnings(animation_retarget_check& check)
95{
96 const auto& report = check.report;
97 if(!report.channels.empty() && count_matched_channels(check) == 0)
98 {
99 check.warnings.emplace_back("No channel reaches a bone: the names differ, and no humanoid roles match them.");
100 }
101 for(const auto& difference : report.rest_differences)
102 {
103 if(difference.degrees <= REST_POSE_WARNING_DEGREES)
104 {
105 continue;
106 }
107 const std::string what = std::string(get_humanoid_bone_name(difference.role)) + " rests " +
108 std::to_string(int(std::lround(difference.degrees))) + " degrees apart on the two skeletons";
109 check.warnings.push_back(
110 check.aligns_to_t_pose
111 ? what + " even straightened to a T-pose: check its role in Humanoid Mapping on both sides."
112 : what + ": they rest in different poses (a T-pose against an A-pose?) and every pose keeps that "
113 "offset. Set Retarget Pose to T-Pose to straighten both first.");
114 }
115 if(report.is_rest_aligned && check.skeleton)
116 {
117 const auto roles = find_unreached_roles(report, *check.skeleton);
118 if(!roles.empty() && roles.size() < HUMANOID_BONE_COUNT)
119 {
120 check.warnings.push_back("The clip has no bone for these roles, which keep the bind pose: " +
121 get_role_list(roles) + ".");
122 }
123 }
124 const bool lost_root_motion = (report.root_position_channel >= 0 && report.root_position_bone < 0) ||
125 (report.root_rotation_channel >= 0 && report.root_rotation_bone < 0);
126 if(lost_root_motion)
127 {
128 check.warnings.emplace_back("The clip's root motion node reaches no bone here: its root motion moves nothing.");
129 }

156auto check_animation_retarget(animation_component& animation,
157 const model_component& model,
158 const std::shared_ptr<animation_clip>& clip) -> animation_retarget_check
159{
160 animation_retarget_check check;
161 check.skeleton = animation_system::get_target_skeleton(animation, model);
162 if(!clip || !check.skeleton)
163 {
164 return check;
165 }
166 const animation_skeleton_binding binding = animation_system::make_skeleton_binding(animation, model);
167 check.report = animation_player::describe_binding(clip, binding, *check.skeleton);
168 check.aligns_to_t_pose = check.skeleton->aligns_to_t_pose;
169 check.pose_offset_count = animation.get_retarget_pose_offsets().size();
170 for(const auto& channel : check.report.channels)
171 {
172 ++check.match_counts[size_t(channel.match)];
173 }
174 const bool can_match_roles = animation.get_retargeting_mode() == animation_retargeting_mode::name_based &&
175 !check.report.is_rest_aligned && check.report.has_source_rest;
176 const size_t role_matches_missed = can_match_roles ? count_role_matches_missed(clip, binding, check) : 0;
177 add_setting_warnings(animation, role_matches_missed, check);
178 add_binding_warnings(check);
179 return check;
180}

251auto set_model_humanoid_override(rtti::context& ctx,
252 const std::string& model_key,
253 humanoid_bone role,
254 const std::optional<std::string>& bone) -> bool
255{
256 auto meta = ctx.get_cached<asset_manager>().get_metadata_for_key(model_key);
257 const auto importer = std::dynamic_pointer_cast<mesh_importer_meta>(meta.meta.importer);
258 if(!importer)
259 {
260 return false;
261 }
262 auto& overrides = importer->rig.humanoid_overrides;
263 overrides.erase(std::remove_if(overrides.begin(),
264 overrides.end(),
265 [role](const humanoid_bone_override& entry)
266 {
267 return entry.role == role;
268 }),
269 overrides.end());
270 if(bone)
271 {
272 overrides.push_back({role, *bone});
273 }
274 return editor_actions::save_import_settings(model_key, meta.meta);
