18auto quote_windows_argument(std::string_view argument) -> std::string
19{
20 // Follow Windows CommandLineToArgvW rules used by CreateProcess.
21 const bool needs_quotes =
22 argument.empty() || argument.find_first_of(" \t\"") != std::string_view::npos;
23 if(!needs_quotes)
24 {
25 return std::string(argument);
26 }
27 std::string result;
28 result.reserve(argument.size() + 2);
29 result.push_back('"');
30 std::size_t backslash_count = 0;
31 for(char ch : argument)
32 {
33 if(ch == '\\')
34 {
35 ++backslash_count;
36 continue;
37 }
38 if(ch == '"')
39 {
40 result.append(backslash_count * 2 + 1, '\\');
41 result.push_back('"');
42 backslash_count = 0;
43 continue;
44 }
45 if(backslash_count > 0)
46 {
47 result.append(backslash_count, '\\');
48 backslash_count = 0;
49 }
50 result.push_back(ch);
51 }
52 if(backslash_count > 0)
53 {
54 result.append(backslash_count * 2, '\\');
55 }
56 result.push_back('"');
57 return result;
58}
