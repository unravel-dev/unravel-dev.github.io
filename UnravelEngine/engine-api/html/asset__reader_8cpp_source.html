44{

121auto validate(const std::string& key, const std::string& compiled_ext, source_fallback fallback, std::string& out)
122 -> bool
123{
124 if(!fs::has_known_protocol(key))
125 {
126 log_unknown_protocol_for_key(key);
127 return false;
128 }
129
130 auto compiled_absolute_path = resolve_compiled_path(key).string() + compiled_ext;
131
132 // A deployed game reads its compiled files from mounted packs; the editor from its cache.
133 if(!fs::is_readable_file(compiled_absolute_path))
134 {
135 if(fallback == source_fallback::none)
136 {
137 log_uncompiled_asset_for_key(compiled_absolute_path);
138 return false;
139 }
140 log_missing_compiled_asset_for_key(compiled_absolute_path);
141
142 compiled_absolute_path = resolve_path(key).string();
143 }
144
145 if(!fs::is_readable_file(compiled_absolute_path))
146 {
147 log_missing_raw_asset_for_key(key);
148 return false;
149 }
150
151 out = compiled_absolute_path;
152 return true;
153}

158template<typename T, typename CreateFromPathFn>
159auto schedule_load(tpp::thread_pool& pool, asset_handle<T>& output, const std::string& key,
160 const std::string& compiled_ext, source_fallback fallback, CreateFromPathFn create_fn,
161 load_mode mode) -> bool
162{
163 if(mode == load_mode::deferred)
164 {
165 auto deferred_fn = [key, compiled_ext, fallback, create_fn]() -> std::shared_ptr<T>
166 {
167 std::string path{};
168 if(!validate(key, compiled_ext, fallback, path))
169 {
170 return nullptr;
171 }
172 return create_fn(path);
173 };
174 output.set_internal_job(pool.create_job(get_job_name<T>(), deferred_fn).share());
175 return true;
176 }
177
178 std::string path{};
179 if(!validate(key, compiled_ext, fallback, path))
180 {
181 return false;
182 }
183 auto immediate_fn = [path, create_fn]() -> std::shared_ptr<T>
184 {
185 return create_fn(path);
186 };
187 output.set_internal_job(pool.schedule(get_job_name<T>(), immediate_fn).share());
188 return true;
189}
