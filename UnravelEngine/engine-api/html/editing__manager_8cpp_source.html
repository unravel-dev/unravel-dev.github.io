489auto has_whole_component_statement(const prefab& document) -> bool
490{
491 constexpr hpp::string_view key = "\"component_path\"";
492 const hpp::string_view text(reinterpret_cast<const char*>(document.buffer.data.data()), document.buffer.data.size());
493 for(auto at = text.find(key); at != hpp::string_view::npos; at = text.find(key, at + key.size()))
494 {
495 const auto open = text.find('"', at + key.size() + 1);
496 const auto close = open == hpp::string_view::npos ? open : text.find('"', open + 1);
497 if(close == hpp::string_view::npos)
498 {
499 return false;
500 }
501 if(text.substr(open + 1, close - open - 1).find('/') == hpp::string_view::npos)
502 {
503 return true;
504 }
505 }
506 return false;
507}

510void editing_manager::prune_stale_statements_in_dependents(rtti::context& ctx, const asset_handle<prefab>& changed)
511{
512 auto& am = ctx.get_cached<asset_manager>();
513
514 // Every loaded prefab, read once: what it nests, and whether it could hold a stale statement.
515 // Peeked, never loaded: one that is not loaded is read fresh - and pruned by its replays -
516 // when it is next used.
517 struct candidate
518 {
519 asset_handle<prefab> handle;
520 std::vector<hpp::uuid> references;
521 bool has_whole_statement{};
522 };
523 std::vector<candidate> candidates;
524 for(const auto& handle : am.get_assets<prefab>())
525 {
526 if(const auto loaded = handle.peek(); loaded && handle.uid() != changed.uid())
527 {
528 candidates.push_back({handle, asset_deps::get_referenced_uids(*loaded), has_whole_component_statement(*loaded)});
529 }
530 }
531
532 // Those that nest `changed`, directly or through others.
533 std::set<hpp::uuid> nesting{changed.uid()};
534 std::vector<asset_handle<prefab>> dependents;
535 for(bool grew = true; grew;)
536 {
537 grew = false;
538 for(const auto& entry : candidates)
539 {
540 if(nesting.contains(entry.handle.uid()))
541 {
542 continue;
543 }
544 const bool nests = std::any_of(entry.references.begin(),
545 entry.references.end(),
546 [&](const hpp::uuid& uid)
547 {
548 return nesting.contains(uid);
549 });
550 if(nests)
551 {
552 nesting.insert(entry.handle.uid());
553 grew = true;
554 if(entry.has_whole_statement)
555 {
556 dependents.push_back(entry.handle);
557 }
558 }
559 }
560 }
561
562 for(const auto& dependent : dependents)
563 {
564 if(is_prefab_mode() && dependent == edited_prefab)
565 {
566 continue;
567 }
568 scene scratch{"stale_statement_check"};
569 const auto dropped_before = get_dropped_stale_statement_count();
570 auto root = scratch.instantiate(dependent);
571 if(root && get_dropped_stale_statement_count() > dropped_before)
572 {
573 APPLOG_INFO("Dropping statements in {} about content {} no longer has", dependent.id(), changed.id());
574 asset_writer::atomic_save_to_file(fs::resolve_protocol(dependent.id()).string(), root);
575 }
576 scratch.unload();
577 }

580void editing_manager::sync_prefab_entity(rtti::context& ctx, entt::handle entity, const asset_handle<prefab>& pfb)
581{
582 // The one funnel every editor-side sync goes through, so this is where the authoring root
583 // is kept out. It is upstream of its file: a replay can only restore what was last saved,
584 // which undoes any revert made since, and it re-labels the root's own authoring on its
585 // nested instances as "inherited from the containing prefab". Both are the file flowing
586 // the wrong way.
587 if(is_authoring_root(entity))
588 {
589 return;
590 }
591
592 queue_action("Sync Prefab Entity",
593 [&ctx, entity, pfb]() mutable
594 {
595 auto& play = ctx.get_cached<play_mode>();
596 if(play.is_active())
597 {
598 return;
599 }
600
601 if(!entity.valid() || !pfb.is_valid())
602 {
603 return;
604 }
605
606 // The sync itself lives in the engine. It is the version that knows about nesting:
607 // it snapshots each nested instance's locally-made overrides and removals before the
608 // replay and puts them back after, filters the document's snapshot of a nested
609 // instance through that instance's override set, and copies the prefab_component
610 // rather than holding a reference across a load that adds and removes them (the pool
611 // swap-and-pops). The editor used to keep its own pre-nesting copy of this dance,
612 // which replayed stale snapshots over live nested instances.
613 if(auto* prefab_comp = entity.try_get<prefab_component>())
614 {
615 prefab_comp->source = pfb;
616 }
617
618 sync_prefab_instance(entity);
619 });
620}

626auto find_outermost_instance(entt::handle entity) -> entt::handle
627{
628 if(!entity)
629 {
630 return {};
631 }
632
633 entt::handle outermost = entity.all_of<prefab_component>() ? entity : entt::handle{};
634 const auto* trans_comp = entity.try_get<transform_component>();
635 auto current = trans_comp != nullptr ? trans_comp->get_parent() : entt::handle{};
636 while(current)
637 {
638 if(current.all_of<prefab_component>())
639 {
640 outermost = current;
641 }
642 const auto* parent_trans = current.try_get<transform_component>();
643 current = parent_trans != nullptr ? parent_trans->get_parent() : entt::handle{};
644 }
645 return outermost;
646}

668void editing_manager::sync_after_override_change(rtti::context& ctx, entt::handle entity)
669{
670 auto outermost = find_outermost_instance(entity);
671
672 if(outermost && !is_authoring_root(outermost))
673 {
674 if(const auto* prefab_comp = outermost.try_get<prefab_component>())
675 {
676 sync_prefab_entity(ctx, outermost, prefab_comp->source);
677 }
678 return;
679 }
680
681 // Under an authoring root, or under nothing at all. No document above to replay, so each
682 // instance directly under it is brought back in line with its own prefab - their own
683 // syncs cascade into anything nested further down.
684 const auto base = outermost ? outermost : entity;
685 std::vector<entt::handle> nested;
686 collect_direct_nested_instances(base, nested);
687 for(auto& instance : nested)
688 {
689 if(const auto* prefab_comp = instance.try_get<prefab_component>())
690 {
691 sync_prefab_entity(ctx, instance, prefab_comp->source);
692 }
693 }

696void editing_manager::sync_prefab_instances(rtti::context& ctx, scene* scn)
697{
698 // Top-level instances only. An instance nested inside another is refreshed by its
699 // container's sync - load_from_prefab_out cascades into everything nested - so syncing
700 // it from here as well repeated the whole replay once per nesting level.
701 scn->registry->view<prefab_component>().each(
702 [&](auto e, auto&& comp)
703 {
704 auto owner = comp.get_owner();
705 const auto* trans_comp = owner.template try_get<transform_component>();
706 auto parent = trans_comp != nullptr ? trans_comp->get_parent() : entt::handle{};
707 while(parent)
708 {
709 if(parent.template all_of<prefab_component>())
710 {
711 return;
712 }
713 const auto* parent_trans = parent.template try_get<transform_component>();
714 parent = parent_trans != nullptr ? parent_trans->get_parent() : entt::handle{};
715 }
716 sync_prefab_entity(ctx, owner, comp.source);
717 });

826void editing_manager::enter_prefab_mode(rtti::context& ctx, const asset_handle<prefab>& prefab, bool auto_save)
827{
828 auto& play = ctx.get_cached<play_mode>();
829 if(play.is_active())
830 {
831 return;
832 }
833
834
835 auto on_continue = [this,&ctx, prefab]()
836 {
837 // Store the prefab we're editing
838 edited_prefab = prefab;
839 current_mode = editing_mode::prefab;
840
841 // Clear selection
842 unselect();
843
844 // Create a new scene for prefab editing if it doesn't exist
845 prefab_scene.unload();
846
847 // Set up a default 3D scene with lighting
848 defaults::create_default_3d_scene_for_editing(ctx, prefab_scene);
849
850 // Instantiate the prefab in our editing scene
851 prefab_entity = prefab_scene.instantiate(prefab);
852 prefab_has_unsaved_changes_ = false;
853
854 // The edit root stays an instance of the prefab - prefab mode is a scene holding one
855 // instance, and Save is Apply All - but it is an instance that is *upstream* of its
856 // file: its live state is the file's next content. The tag is what keeps the two
857 // instance behaviours that assume the opposite away from it - syncing against the
858 // file (sync_prefab_entity) and recording its own content as overrides of the file
859 // (find_prefab_root_entity). What the document states about its nested content is
860 // adopted as the root's own local list, so it reads as the author's own work - not as
861 // inherited - and a save folds it back into the file.
862 if(prefab_entity)
863 {
864 prefab_entity.emplace<authoring_root_tag>();
865 scene::adopt_document_statements(prefab_entity);
866 }
867
868 // Select the prefab entity
869 if (prefab_entity)
870 {
871 select(prefab_entity);
872 }
873
874 APPLOG_INFO("Entered prefab editing mode for: {}", prefab.id());
875 };
876
877 if (is_prefab_mode())
878 {
879 // Already in prefab mode, check if we need to save changes
880 if (edited_prefab != prefab)
881 {
882 auto on_save = [this,&ctx]()
883 {
884 save_prefab_changes(ctx);
885 };
886
887 if(auto_save)
888 {
889 on_save();
890 }
891 else
892 {
893 prompt_save_changes(ctx, on_save, on_continue);
894 return;
895 }
896 }
897 else
898 {
899 select(prefab_entity);
900
901 // Already editing this prefab, nothing to do
902 return;
903 }
904 }
905
906 on_continue();
907
908

1184{

1188 return;

1191 // back before the old mesh rebinds them by name.
