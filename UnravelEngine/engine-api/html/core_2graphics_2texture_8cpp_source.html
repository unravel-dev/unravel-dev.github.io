53auto try_make_room_for(std::uint64_t bytes, const char* what, eviction::reclaim_kind kind) -> bool
54{
55 if(bytes == 0 || !eviction::is_supported())
56 {
57 return true;
58 }
59 if(kind == eviction::reclaim_kind::evictable)
60 {
61 if(eviction::would_allocation_fit(bytes))
62 {
63 return true;
64 }
65 const std::string msg = std::string("Eviction: Insufficient headroom for ") + what + " (" +
66 format_bytes(bytes) + "); Skipping allocation (fallback texture).";
67 gfx::log("warning", msg, __FILE__, __LINE__);
68 return false;
69 }
70 const auto result = eviction::reclaim_for(bytes, kind);
71 if(result != eviction::reclaim_result::insufficient)
72 {
73 return true;
74 }
75 const std::string msg = std::string("Eviction: Insufficient headroom for ") + what + " (" +
76 format_bytes(bytes) + "); Allocation may fail.";
77 gfx::log("warning", msg, __FILE__, __LINE__);
78 return true;
79}

119void calc_desc_info(bgfx::TextureInfo& info, const texture_desc& desc)
120{
121 const std::uint16_t width = to_dimension(desc.size.width);
122 const std::uint16_t height = to_dimension(desc.size.height);
123 switch(desc.kind)
124 {
125 case texture_kind::tex_3d:
126 bgfx::calcTextureSize(info, width, height, desc.depth, false, desc.has_mips, 1, desc.format);
127 return;
128 case texture_kind::tex_cube:
129 // Six width x width faces at depth 1. Describing a width^3 volume would inflate the
130 // eviction estimate (a mipped 256 RGBA16F cube: ~153 MB instead of ~4 MB) and leave
131 // info.cubeMap false.
132 bgfx::calcTextureSize(info, width, width, 1, true, desc.has_mips, desc.layers, desc.format);
133 return;
134 case texture_kind::tex_2d:
135 break;
136 }
137 bgfx::calcTextureSize(info, width, height, 1, false, desc.has_mips, desc.layers, desc.format);
138}

316texture::texture(const texture_desc& desc, const bgfx::Memory* mem) : flags(desc.flags), desc_(desc)
317{

319 assert((desc.kind != texture_kind::tex_3d || desc.layers == 1) && "A 3D texture has no array layers");
320 assert((desc.kind != texture_kind::tex_cube || desc.size.width == desc.size.height) && "Cube faces are square");

323 const eviction::reclaim_kind kind = texture_reclaim_kind(desc.flags);
324 const bool can_allocate = try_make_room_for(estimated_size, get_kind_label(desc.kind), kind);
325 if(can_allocate)

327 handle_ = create_desc_handle(desc, mem);
