245auto editor::create_main_window(rtti::context& ctx, const cmd_line::parser& parser) -> bool
246{
247 // The backend is not up yet; show_main_window adds its name to the title.
248 const std::string title = "Unravel Editor";
249 auto& rend = ctx.get_cached<renderer>();
250 std::string window_geometry;
251 int32_t x = 0;
252 int32_t y = 0;
253 uint32_t width = 0;
254 uint32_t height = 0;
255 bool maximized = false;
256 if(parser.try_get("window", window_geometry) &&
257 renderer::parse_window_geometry(window_geometry, x, y, width, height, maximized))
258 {
259 uint32_t flags = os::window::resizable;
260 if(maximized)
261 {
262 flags |= os::window::maximized;
263 }
264 rend.create_window(title, x, y, width, height, flags);
265 return true;
266 }
267 const uint32_t flags = os::window::resizable | os::window::maximized;
268 const auto primary_display = os::display::get_primary_display_index();
269 rend.create_window_for_display(primary_display, title, flags);
270 return true;
271}
