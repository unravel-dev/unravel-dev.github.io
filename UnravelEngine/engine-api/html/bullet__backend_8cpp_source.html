213static constexpr int COMBINE_BITS = 2;
214static constexpr int COMBINE_MASK = (1 << COMBINE_BITS) - 1; // 0b11
215static constexpr int FRICTION_SHIFT = COMBINE_BITS; // friction in bits [3..2]
216static constexpr int RESTITUTION_SHIFT = 0; // bounce in bits [1..0]

218inline int encode_combine_modes(unravel::combine_mode friction, unravel::combine_mode bounce)
219{
220 int f = (static_cast<int>(friction) & COMBINE_MASK) << FRICTION_SHIFT;
221 int b = (static_cast<int>(bounce) & COMBINE_MASK) << RESTITUTION_SHIFT;
222 return f | b;
223}

235//------------------------------------------------------------------------------
236// 2) Helper to pick a single combine-mode when two bodies collide.
237// If both bodies requested the same mode, we use that. Otherwise, default to Average.
238// You can adjust this tie-breaking however you like.
239//------------------------------------------------------------------------------
240static unravel::combine_mode pick_combine_mode(unravel::combine_mode modeA, unravel::combine_mode modeB)
241{
242 if(modeA == modeB)
243 {
244 return modeA;
245 }
246 // If only one of them left at default 0 (Multiply) and you want to treat that
247 // differently, you could check for that here. For simplicity we go to Average any time
248 // they differ:
249 return unravel::combine_mode::average;
250}

252//------------------------------------------------------------------------------
253// 3) The global callback that Bullet will call for each new contact.
254// We read userIndex2 from each body to decide how to combine their restitutions.
255//------------------------------------------------------------------------------
256static btScalar per_body_combine(const btCollisionObject* body0,
257 const btCollisionObject* body1,
258 btScalar e0,
259 btScalar e1,
260 unravel::combine_mode mode0,
261 unravel::combine_mode mode1)
262{
263 // 3.3) Pick final combine mode:
264 auto mode = pick_combine_mode(mode0, mode1);
265
266 // 3.5) Compute combined restitution according to chosenMode:
267 btScalar combined;
268 switch(mode)
269 {
270 case unravel::combine_mode::multiply:
271 combined = e0 * e1;
272 break;
273
274 case unravel::combine_mode::average:
275 combined = (e0 + e1) * btScalar(0.5);
276 break;
277
278 case unravel::combine_mode::minimum:
279 combined = btMin(e0, e1);
280 break;
281
282 case unravel::combine_mode::maximum:
283 combined = btMax(e0, e1);
284 break;
285
286 default:
287 combined = e0 * e1; // fallback if somehow we get out-of-range
288 break;
289 }
290
291 // 3.7) Return true to indicate “we handled it.”
292 return combined;
293}

394// --- Bullet internal profiling ----------------------------------------------------
395//
396// Every BT_PROFILE inside Bullet expands to a CProfileSample, whose constructor calls
397// through btEnterProfileZone -> a function pointer that defaults to an empty function.
398// That indirection is present even in this BT_NO_PROFILE build, so redirecting it
399// surfaces stepSimulation's internal breakdown - predictUnconstraintMotion,
400// dispatchAllCollisionPairs, solveConstraints, integrateTransforms - without rebuilding
401// Bullet or defining BT_ENABLE_PROFILE.
402//
403// Cost when the profiler is not capturing is one relaxed atomic load per zone, because
404// profile_begin gates on the process capture flag before touching thread-local state.
405
409constexpr int max_profile_zone_depth = 4;
410
411thread_local int t_profile_zone_depth = 0;
412thread_local uint32_t t_profile_zone_stack[max_profile_zone_depth]{};
413
414void enter_profile_zone(const char* name)
415{
416 const int depth = t_profile_zone_depth++;
417 if(depth < max_profile_zone_depth)
418 {
419 // The lane name is only consumed the first time a given thread reports, so this
420 // labels Bullet's worker threads and is ignored on the main thread, which
421 // already has a lane.
422 t_profile_zone_stack[depth] = unravel::profile_begin(name, "Physics Thread");
423 }
424}
425
426void leave_profile_zone()
427{
428 if(t_profile_zone_depth <= 0)
429 {
430 return;
431 }
432
433 const int depth = --t_profile_zone_depth;
434 if(depth < max_profile_zone_depth)
435 {
436 // profile_end ignores UINT32_MAX, which is what profile_begin returns while the
437 // profiler is idle, so depth tracking stays balanced either way.
438 unravel::profile_end(t_profile_zone_stack[depth]);
439 }
440}

982 void release_contacts_for(entt::handle entity, unravel::contact_end_reason reason, bool notify)
983 {
984 auto* links = find_contact_links(entity);
985 if(links == nullptr || links->head == unravel::contact_links::npos)
986 {
987 return;
988 }
989
990 const bool wants_events = notify && links->flush_pending != 0;
991
992 unravel::physics_vector<contact_manifold, 8> pending;
993 if(wants_events)
994 {
995 pending.reserve(links->flush_pending);
996 }
997
998 contacts.visit(entity,
999 [&](uint32_t id, contact_slot& slot)
1000 {
1001 // An enter that never reached script code must not be
1002 // answered with an exit; as far as gameplay is concerned
1003 // that pair never happened.
1004 if(wants_events && slot.flush_on_destroy && slot.enter_dispatched)
1005 {
1006 // Moved, not copied: the slot is erased two lines down.
1007 pending.push_back(std::move(slot.payload));
1008
1009 auto& cm = pending.back();
1010 cm.event = event_type::exit;
1011 // process_manifold states the reason relative to `a`.
1012 cm.reason = (slot.a == entity) ? reason : unravel::mirror_contact_end_reason(reason);
1013 }
1014
1015 contacts.erase(id);
1016 return false;
1017 });
1018
1019 if(pending.empty())
1020 {
1021 return;
1022 }
1023
1024 auto& ctx = unravel::engine::context();
1025 auto& scripting = ctx.get_cached<unravel::script_system>();
1026 for(const auto& cm : pending)
1027 {
1028 process_manifold(scripting, cm);
1029 }
1030 }

1068 static void copy_manifold_points(btPersistentManifold* m,
1069 std::vector<unravel::manifold_point>& out,
1070 bool swap_sides)
1071 {
1072 const int count = m->getNumContacts();
1073 out.clear();
1074 out.reserve(static_cast<size_t>(count));
1075
1076 for(int j = 0; j < count; ++j)
1077 {
1078 const auto& p = m->getContactPoint(j);
1079
1080 unravel::manifold_point mp;
1081 if(swap_sides)
1082 {
1083 mp.a = from_bullet(p.getPositionWorldOnB());
1084 mp.b = from_bullet(p.getPositionWorldOnA());
1085 mp.normal_on_a = from_bullet(p.m_normalWorldOnB);
1086 mp.normal_on_b = -mp.normal_on_a;
1087 }
1088 else
1089 {
1090 mp.a = from_bullet(p.getPositionWorldOnA());
1091 mp.b = from_bullet(p.getPositionWorldOnB());
1092 mp.normal_on_b = from_bullet(p.m_normalWorldOnB);
1093 mp.normal_on_a = -mp.normal_on_b;
1094 }
1095 mp.impulse = p.getAppliedImpulse();
1096 mp.distance = p.getDistance();
1097
1098 out.push_back(mp);
1099 }

1110 void touch_pair(btPersistentManifold* m, entt::handle a, entt::handle b, manifold_type type, bool swap_sides)
1111 {
1112 // Cheapest gate first: one or two sparse-set probes and out. A pile of
1113 // unscripted debris never reaches the graph at all, so it costs no slots, no
1114 // contact-point buffers and no per-step diffing. If a listener disappears from
1115 // an already-tracked pair, skipping the refresh here lets the next sweep expire
1116 // the slot.
1117 if(!has_contact_listener(a, b, type))
1118 {
1119 return;
1120 }
1121
1122 const uint32_t existing = find_slot_for_pair(a, b, type);
1123 if(existing != unravel::contact_links::npos)
1124 {
1125 contacts.get(existing).seen_stamp = contacts.stamp();
1126 return;
1127 }
1128
1129 const auto flags = effective_contact_flags(a, b, type);
1130 if(!should_track_contact(flags, type))
1131 {
1132 return;
1133 }
1134
1135 const uint32_t id = contacts.insert(a, b, should_flush_on_destroy(flags, type));
1136 if(id == unravel::contact_links::npos)
1137 {
1138 return;
1139 }
1140
1141 auto& slot = contacts.get(id);
1142 slot.payload.type = type;
1143 slot.payload.event = event_type::enter;
1144 slot.payload.reason = unravel::contact_end_reason::separated;
1145 slot.payload.a = a;
1146 slot.payload.b = b;
1147 copy_manifold_points(m, slot.payload.contacts, swap_sides);
1148
1149 to_enter.push_back(slot_ref{id, slot.generation});
1150 }

1152 void process_manifolds()
1153 {
1154 APP_SCOPE_PERF("Physics/Bullet/Process Manifolds");
1155 auto& ctx = unravel::engine::context();
1156 auto& scripting = ctx.get_cached<unravel::script_system>();
1157 auto& ec = ctx.get_cached<unravel::ecs>();
1158
1159 auto* dispatcher = dynamics_world->getDispatcher();
1160 const int nm = dispatcher->getNumManifolds();
1161
1162 // Opening a step is a counter bump. Pairs seen during it record the stamp;
1163 // anything still carrying an older one has separated. Replaces a full pass over
1164 // every live contact just to clear a flag.
1165 const uint32_t stamp = contacts.advance_stamp();
1166
1167 to_enter.clear();
1168 to_exit.clear();
1169
1170 // Nothing in the scene can receive a contact callback, so there is nothing to
1171 // fold in. Existing pairs still expire below, which drains the graph.
1172 auto& registry = *ec.get_scene().registry;
1173 const bool any_listeners = !registry.storage<unravel::script_component>().empty();
1174
1175 // Phase 1: fold every current manifold into the graph.
1176 for(int i = 0; any_listeners && i < nm; ++i)
1177 {
1178 auto* m = dispatcher->getManifoldByIndexInternal(i);
1179 if(m->getNumContacts() == 0)
1180 {
1181 continue;
1182 }
1183
1184 const auto* obj_a = m->getBody0();
1185 const auto* obj_b = m->getBody1();
1186 const bool is_sensor_a = (obj_a->getCollisionFlags() & btCollisionObject::CF_NO_CONTACT_RESPONSE) != 0;
1187 const bool is_sensor_b = (obj_b->getCollisionFlags() & btCollisionObject::CF_NO_CONTACT_RESPONSE) != 0;
1188
1189 auto entity_a = get_entity_from_user_index(ec, obj_a->getUserIndex());
1190 auto entity_b = get_entity_from_user_index(ec, obj_b->getUserIndex());
1191 if(!entity_a || !entity_b)
1192 {
1193 continue;
1194 }
1195
1196 if(is_sensor_a || is_sensor_b)
1197 {
1198 // Sensor pairs are directional - the sensor is always the `a` side,
1199 // and two overlapping sensors each report the other.
1200 if(is_sensor_a)
1201 {
1202 touch_pair(m, entity_a, entity_b, manifold_type::sensor, false);
1203 }
1204 if(is_sensor_b)
1205 {
1206 touch_pair(m, entity_b, entity_a, manifold_type::sensor, true);
1207 }
1208 continue;
1209 }
1210
1211 touch_pair(m, entity_a, entity_b, manifold_type::collision, false);
1212 }
1213
1214 // Phase 2: whatever was not seen has separated. Walking backwards keeps the
1215 // swap-and-pop inside contact_graph::erase from skipping an entry.
1216 const auto& live = contacts.live();
1217 for(size_t i = live.size(); i != 0; --i)
1218 {
1219 const uint32_t id = live[i - 1];
1220 auto& slot = contacts.get(id);
1221 if(slot.seen_stamp == stamp)
1222 {
1223 continue;
1224 }
1225
1226 if(slot.enter_dispatched)
1227 {
1228 // Moved, not copied: the slot is freed on the next line, so its contact
1229 // buffer can be handed over instead of duplicated.
1230 to_exit.push_back(std::move(slot.payload));
1231 to_exit.back().event = event_type::exit;
1232 to_exit.back().reason = unravel::contact_end_reason::separated;
1233 }
1234
1235 contacts.erase(id);
1236 }
1237
1238 // Phase 3: dispatch. Script code called here may destroy entities, which frees
1239 // slots via release_contacts_for, so every queued enter is revalidated against
1240 // the slot generation before use. Exits carry their own copy and cannot dangle.
1241 for(const auto& ref : to_enter)
1242 {
1243 if(!contacts.is_live(ref.slot, ref.generation))
1244 {
1245 continue;
1246 }
1247
1248 auto& slot = contacts.get(ref.slot);
1249 slot.enter_dispatched = true;
1250 process_manifold(scripting, slot.payload);
1251 }
1252
1253 for(const auto& cm : to_exit)
1254 {
1255 process_manifold(scripting, cm);
1256 }
1257 }

1348 // Then the function itself
1349 auto sphere_cast_closest(const math::vec3& origin,
1350 const math::vec3& direction,
1351 float radius,
1352 float max_distance,
1353 int layer_mask,
1354 bool query_sensors) -> hpp::optional<unravel::raycast_hit>
1355 {
1356 if(!dynamics_world)
1357 {
1358 return {};
1359 }
1360
1361 // Convert origin, direction to bullet
1362 btVector3 btOrigin = to_bullet(origin);
1363 btVector3 btEnd = to_bullet(origin + direction * max_distance);
1364
1365 // Create a temporary sphere shape
1366 // (We do *not* add this shape to the world, just use it for sweeping)
1367 btSphereShape shape(radius);
1368 // shape.setMargin(0.f); // optionally set margin=0
1369
1370 // Build transform from=to
1371 btTransform start, end;
1372 start.setIdentity();
1373 end.setIdentity();
1374 start.setOrigin(btOrigin);
1375 end.setOrigin(btEnd);
1376
1377 // Setup our custom callback
1378 bullet::sphere_closest_convex_result_callback cb(btOrigin, btEnd, layer_mask, query_sensors);
1379
1380 // Perform the sweep
1381 dynamics_world->convexSweepTest(&shape, start, end, cb);
1382
1383 // Check if we got a hit
1384 if(!cb.hasHit())
1385 return {}; // no hit
1386
1387 // Build a raycast_hit
1388 unravel::raycast_hit hit;
1389 const btCollisionObject* obj = cb.m_hitCollisionObject;
1390 float fraction = cb.m_closestHitFraction;
1391 btVector3 hitPoint = btOrigin.lerp(btEnd, fraction);
1392 btVector3 normal = cb.m_hitNormalWorld;
1393 hit.entity = resolve_entity_from_collision_object(obj);
1394 hit.point = from_bullet(hitPoint);
1395 hit.normal = from_bullet(normal.normalized());
1396 hit.distance = fraction * max_distance; // approximate
1397 return hit;
1398 }

1400 auto sphere_cast_all(const math::vec3& origin,
1401 const math::vec3& direction,
1402 float radius,
1403 float max_distance,
1404 int layer_mask,
1405 bool query_sensors) -> unravel::physics_vector<unravel::raycast_hit>
1406 {
1407 if(!dynamics_world)
1408 {
1409 return {};
1410 }
1411 // bullet transforms
1412 btVector3 btOrigin = to_bullet(origin);
1413 btVector3 btEnd = to_bullet(origin + direction * max_distance);
1414
1415 btTransform start, end;
1416 start.setIdentity();
1417 end.setIdentity();
1418 start.setOrigin(btOrigin);
1419 end.setOrigin(btEnd);
1420
1421 // shape
1422 btSphereShape shape(radius);
1423
1424 // custom callback
1425 sphere_all_convex_result_callback cb(layer_mask, query_sensors);
1426
1427 dynamics_world->convexSweepTest(&shape, start, end, cb);
1428
1429 // Now cb.hits has all hits in the order they were encountered
1430 // Typically not sorted by fraction, so let's sort them:
1431 std::sort(cb.hits.begin(),
1432 cb.hits.end(),
1433 [](auto& a, auto& b)
1434 {
1435 return a.fraction < b.fraction;
1436 });
1437
1438 // Build the final results
1439 unravel::physics_vector<unravel::raycast_hit> hits;
1440 hits.reserve(cb.hits.size());
1441
1442 for(const auto& hi : cb.hits)
1443 {
1444 auto& hit = hits.emplace_back();
1445 hit.entity = resolve_entity_from_collision_object(hi.object);
1446 btVector3 hitPoint = btOrigin.lerp(btEnd, hi.fraction);
1447 hit.point = from_bullet(hitPoint);
1448 hit.normal = from_bullet(hi.normal.normalized());
1449 hit.distance = hi.fraction * max_distance;
1450 }
1451
1452 return hits;
1453 }

1568// Builds one Bullet collision shape per submesh, each paired with the submesh's node
1569// transform (relative to the model root). The vertex buffer stores positions in node-local
1570// space; keeping a separate shape per submesh lets us apply the matching node transform via
1571// addChildShape so the collision geometry stays aligned with the rendered mesh.
1572auto create_bullet_mesh_shapes(const physics_mesh_shape& shape)
1573 -> std::vector<std::pair<btCollisionShape*, btTransform>>
1574{
1575 std::vector<std::pair<btCollisionShape*, btTransform>> result;
1576 const auto& mesh_ref = shape.mesh_asset.get();
1577
1578 // Get vertex and index data from mesh
1579 auto* vertex_data = mesh_ref->get_system_vb();
1580 auto* index_data = mesh_ref->get_system_ib();
1581 auto vertex_count = mesh_ref->get_vertex_count();
1582 auto face_count = mesh_ref->get_face_count();
1583 const auto& vertex_format = mesh_ref->get_vertex_format();
1584
1585 if(!vertex_data || !index_data || vertex_count == 0 || face_count == 0)
1586 {
1587 return result;
1588 }
1589
1590 // Find position attribute offset in vertex format
1591 auto position_offset = vertex_format.getOffset(bgfx::Attrib::Position);
1592
1593 if(position_offset == UINT16_MAX)
1594 {
1595 return result; // No position data
1596 }
1597
1598 const auto& submeshes = mesh_ref->get_submeshes();
1599 const auto node_transforms = mesh_ref->get_submesh_node_transforms();
1600 result.reserve(submeshes.size());
1601
1602 for(size_t s = 0; s < submeshes.size(); ++s)
1603 {
1604 const auto* submesh = submeshes[s];
1605 if(!submesh || submesh->face_start < 0 || submesh->face_count == 0)
1606 {
1607 continue;
1608 }
1609 const auto face_begin = static_cast<uint32_t>(submesh->face_start);
1610 if(face_begin >= face_count)
1611 {
1612 continue;
1613 }
1614
1615 // Build a triangle mesh from this submesh's faces only.
1616 auto* triangle_mesh = new btTriangleMesh(true, false); // 32-bit indices, 3-component vertices
1617 const auto face_end = std::min(face_begin + submesh->face_count, face_count);
1618 for(uint32_t f = face_begin; f < face_end; ++f)
1619 {
1620 uint32_t i0 = index_data[f * 3 + 0];
1621 uint32_t i1 = index_data[f * 3 + 1];
1622 uint32_t i2 = index_data[f * 3 + 2];
1623
1624 float v0[4];
1625 float v1[4];
1626 float v2[4];
1627 bgfx::vertexUnpack(v0, bgfx::Attrib::Position, vertex_format, vertex_data, i0);
1628 bgfx::vertexUnpack(v1, bgfx::Attrib::Position, vertex_format, vertex_data, i1);
1629 bgfx::vertexUnpack(v2, bgfx::Attrib::Position, vertex_format, vertex_data, i2);
1630
1631 btVector3 vertex0(v0[0], v0[1], v0[2]);
1632 btVector3 vertex1(v1[0], v1[1], v1[2]);
1633 btVector3 vertex2(v2[0], v2[1], v2[2]);
1634 triangle_mesh->addTriangle(vertex0, vertex1, vertex2);
1635 }
1636
1637 // Create appropriate collision shape based on type
1638 btCollisionShape* collision_shape = nullptr;
1639 if(shape.collision_type == mesh_collision_type::convex)
1640 {
1641 // Create convex hull shape (can be dynamic)
1642 collision_shape = new btConvexTriangleMeshShape(triangle_mesh);
1643 }
1644 else
1645 {
1646 // Create concave BVH triangle mesh shape (static only, but accurate)
1647 collision_shape = new btBvhTriangleMeshShape(triangle_mesh, true); // Use quantized AABB compression
1648 }
1649
1650 // Apply the submesh's node transform. btTransform only carries rotation/translation,
1651 // so any node scale is applied via the shape's local scaling.
1652 btTransform child_transform = btTransform::getIdentity();
1653 if(s < node_transforms.size())
1654 {
1655 const auto& node_transform = node_transforms[s];
1656 child_transform.setRotation(bullet::to_bullet(node_transform.get_rotation()));
1657 child_transform.setOrigin(bullet::to_bullet(node_transform.get_position()));
1658 collision_shape->setLocalScaling(bullet::to_bullet(node_transform.get_scale()));
1659 }
1660
1661 result.emplace_back(collision_shape, child_transform);
1662 }
1663
1664 return result;
1665}

1667auto make_rigidbody_shape(physics_component& comp) -> std::shared_ptr<btCompoundShape>
1668{
1669 // use an ownning compound shape. When sharing is implemented we can go back to non owning
1670 auto cp = std::make_shared<bullet::btCompoundShapeOwning>();
1671
1672 auto compound_shapes = comp.get_shapes();
1673 if(compound_shapes.empty())
1674 {
1675 return cp;
1676 }
1677
1678 for(const auto& s : compound_shapes)
1679 {
1680 if(hpp::holds_alternative<physics_box_shape>(s.shape))
1681 {
1682 const auto& shape = hpp::get<physics_box_shape>(s.shape);
1683 auto half_extends = shape.extends * 0.5f;
1684
1685 btBoxShape* box_shape = new btBoxShape({half_extends.x, half_extends.y, half_extends.z});
1686
1687 btTransform local_transform = btTransform::getIdentity();
1688 local_transform.setOrigin(bullet::to_bullet(shape.center));
1689 cp->addChildShape(local_transform, box_shape);
1690 }
1691 else if(hpp::holds_alternative<physics_sphere_shape>(s.shape))
1692 {
1693 const auto& shape = hpp::get<physics_sphere_shape>(s.shape);
1694
1695 btSphereShape* sphere_shape = new btSphereShape(shape.radius);
1696
1697 btTransform local_transform = btTransform::getIdentity();
1698 local_transform.setOrigin(bullet::to_bullet(shape.center));
1699 cp->addChildShape(local_transform, sphere_shape);
1700 }
1701 else if(hpp::holds_alternative<physics_capsule_shape>(s.shape))
1702 {
1703 const auto& shape = hpp::get<physics_capsule_shape>(s.shape);
1704
1705 btCapsuleShape* capsule_shape = new btCapsuleShape(shape.radius, shape.length);
1706
1707 btTransform local_transform = btTransform::getIdentity();
1708 local_transform.setOrigin(bullet::to_bullet(shape.center));
1709 cp->addChildShape(local_transform, capsule_shape);
1710 }
1711 else if(hpp::holds_alternative<physics_cylinder_shape>(s.shape))
1712 {
1713 const auto& shape = hpp::get<physics_cylinder_shape>(s.shape);
1714
1715 btVector3 half_extends(shape.radius, shape.length * 0.5f, shape.radius);
1716 btCylinderShape* cylinder_shape = new btCylinderShape(half_extends);
1717
1718 btTransform local_transform = btTransform::getIdentity();
1719 local_transform.setOrigin(bullet::to_bullet(shape.center));
1720 cp->addChildShape(local_transform, cylinder_shape);
1721 }
1722 else if(hpp::holds_alternative<physics_mesh_shape>(s.shape))
1723 {
1724 const auto& shape = hpp::get<physics_mesh_shape>(s.shape);
1725
1726 // Only create mesh shape if we have a valid mesh asset
1727 if(shape.mesh_asset && shape.mesh_asset.is_ready())
1728 {
1729 // One collision shape per submesh, each carrying its own node transform.
1730 auto mesh_shapes = create_bullet_mesh_shapes(shape);
1731 for(auto& [mesh_shape, node_transform] : mesh_shapes)
1732 {
1733 if(!mesh_shape)
1734 {
1735 continue;
1736 }
1737 // Apply the shape center on top of the submesh node transform.
1738 btTransform local_transform = node_transform;
1739 local_transform.setOrigin(local_transform.getOrigin() + bullet::to_bullet(shape.center));
1740 cp->addChildShape(local_transform, mesh_shape);
1741 }
1742 }
1743 }
1744 }
1745
1746 return cp;
1747}

2013void destroy_phyisics_body(bullet::world& world, entt::handle entity, bool from_physics_component)
2014{
2015 auto body = entity.try_get<bullet::rigidbody>();
2016
2017 if(body)
2018 {
2019 // Anything this body was still touching leaves the graph with it either way: a
2020 // slot outliving its participant would later report an exit against a dead
2021 // handle, which is the failure this whole path exists to remove.
2022 //
2023 // Whether that is worth telling anyone about depends on why we are here.
2024 // Suppression is raised only for the span of entt::registry::destroy, so:
2025 //
2026 // suppressed -> entity teardown. The pre-destroy phase already reported
2027 // these while the whole subtree was intact; saying it twice
2028 // would be wrong and running script code from inside
2029 // registry::destroy would be worse.
2030 // not -> the component is being taken off an entity that lives on
2031 // (inspector, RemoveComponent, a body swap). The overlap has
2032 // genuinely ended and nobody else will ever say so.
2033 //
2034 // A raw handle.destroy() that skipped scene::destroy_entity also lands in the
2035 // second case. The physics layer cannot tell it apart from a plain component
2036 // removal from in here, and reporting from a slightly riskier context beats
2037 // dropping the event, so it takes the same path.
2038 const bool notify = !scene::is_destroy_suppressed();
2039
2040 world.release_contacts_for(entity, unravel::contact_end_reason::other_disabled, notify);
2041
2042 if(body->internal)
2043 {
2044 world.remove_rigidbody(*body);
2045 }
2046 }
2047
2048 if(from_physics_component)
2049 {
2050 entity.remove<bullet::rigidbody>();
2051 }

2495auto add_force(btRigidBody* body, const btVector3& force, force_mode mode) -> bool
2496{
2497 if(force.fuzzyZero())
2498 {
2499 return false;
2500 }
2501 // Apply force based on ForceMode
2502 switch(mode)
2503 {
2504 case force_mode::force: // Continuous force
2505 body->applyCentralForce(force);
2506 break;
2507
2508 case force_mode::acceleration:
2509 { // Force independent of mass
2510 btVector3 acceleration_force = force * body->getMass();

2512 break;
2513 }
2514
2515 case force_mode::impulse: // Instantaneous impulse
2516 body->applyCentralImpulse(force);
2517 break;
2518
2519 case force_mode::velocity_change: // Direct velocity change
2520 {
2521 btVector3 new_velocity = body->getLinearVelocity() + force; // Accumulate velocity
2522 body->setLinearVelocity(new_velocity);
2523 break;
2524 }
2525 }

2530{

2532 {
2533 return false;
2534 }
2535 // Apply force based on ForceMode
2536 switch(mode)
2537 {
2538 case force_mode::force: // Continuous torque

2547 inertia_tensor.getY() != 0 ? torque.getY() * (1.0f / inertia_tensor.getY()) : 0.0f,

2551 break;

2553 case force_mode::impulse: // Angular impulse
2554 body->applyTorqueImpulse(torque);
2555 break;
2556
2557 case force_mode::velocity_change: // Direct angular velocity change
2558 {
2559 btVector3 new_velocity = body->getLinearVelocity() + torque; // Accumulate velocity
2560 body->setAngularVelocity(new_velocity);

2807void bullet_backend::apply_explosion_force(physics_component& comp,
2808 float explosion_force,
2809 const math::vec3& explosion_position,
2810 float explosion_radius,

2813{

2815 {
2816 return;
2817 }

2820 if(!world)

2822 return;
2823 }

2825 auto owner = comp.get_owner();
2826 // Flush pending prefab/spawn dirty state so the impulse hits a correct dynamic body.
2827 const bool force_recreate = !owner.all_of<bullet::rigidbody>();
2828 bullet::sync_physics_body(*world, comp, force_recreate);

2842 {
2843 return;
2844 }

2847 {
2848 direction /= distance;
