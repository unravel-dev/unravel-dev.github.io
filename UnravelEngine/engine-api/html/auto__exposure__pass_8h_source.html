105 // -- LOCAL EXPOSURE (a bilateral grid of the metered log luminance). One exposure for
106 // the whole frame has to choose between a sunlit exterior and the interior around
107 // it; local exposure keeps the global choice and redistributes CONTRAST around each
108 // pixel's own neighbourhood level, so both stay readable without the flat look of a
109 // tone curve pulled down. Neutral (every scale 1) is a no-op and costs nothing.
110 //
111 // The contrast scales default to 0.8 / 0.8, so every scene starts with a mild
112 // compression of both highlights and shadows. The shadow lift also raises whatever
113 // little light an unlit room holds.
116 float local_highlight_contrast = 0.8f;
118 float local_shadow_contrast = 0.8f;
121 float local_detail_strength = 1.0f;
124 float local_blurred_blend = 0.6f;
126 float local_blurred_kernel_percent = 50.0f;
128 float local_middle_grey_bias = 0.0f;
129
132 auto is_local_exposure_enabled() const -> bool
133 {
134 return local_highlight_contrast != 1.0f || local_shadow_contrast != 1.0f ||
135 local_detail_strength != 1.0f;
136 }
