32void internal_m2n_animation_blend(entt::entity id, int layer, hpp::uuid guid, float seconds, bool loop, bool phase_sync)
33{
34 if(!is_layer_index(layer))
35 {
36 return;
37 }
38 if(auto comp = safe_get_component<animation_component>(id))
39 {
40 auto& ctx = engine::context();
41 auto& am = ctx.get_cached<asset_manager>();
42 auto asset = am.get_asset<animation_clip>(guid);
43 // Stopping a layer the controller drives stops it there too, or it would resume the state.
44 auto* controller = comp->try_get_controller_instance();
45 if(!asset && controller != nullptr && size_t(layer) < controller->get_layer_count())
46 {
47 controller->stop_layer(size_t(layer));
48 }
49 comp->get_player().blend_to(layer, asset, animation_player::seconds_t(seconds), loop, phase_sync);
50 }
51}

238 auto* controller = comp->get_controller_instance();
239 if(controller != nullptr && size_t(layer) < controller->get_layer_count())

241 controller->set_layer_blend_mode(comp->get_player(), size_t(layer), mode);
242 return;

244 comp->get_player().set_layer_blend_mode(size_t(layer), mode);
