134 ImDrawList* draw_list = ImGui::GetWindowDrawList();

145 draw_list->ChannelsSetCurrent(0);
146 const float rounding = to_pixels(ABOUT_BOX_ROUNDING);
147 draw_list->AddRectFilled(box_min, box_max, ABOUT_BOX_COLOR, rounding);
148 draw_list->AddRect(box_min, box_max, ABOUT_BOX_BORDER_COLOR, rounding);
149 draw_list->ChannelsMerge();
150 ImGui::SetCursorScreenPos(box_min);
151 ImGui::Dummy(box_max - box_min);

187 ImGuiWindowFlags_AlwaysAutoResize | ImGuiWindowFlags_NoSavedSettings;
188 const bool is_visible = ImGui::BeginPopupModal(ABOUT_POPUP_ID, nullptr, flags);
189 // Only the window itself: tooltips opened from it keep the theme's padding.
190 ImGui::PopStyleVar(2);
191 if(!is_visible)
192 {
193 is_open_ = false;
194 return;
195 }
196
197 draw_header(ctx);
198 draw_section_gap();
199 draw_details();
200 draw_section_gap();
201 draw_links();
202 draw_section_gap();
203 draw_footer();
204
205 if(ImGui::IsKeyChordPressed(ImGuiMod_Ctrl | ImGuiKey_C) && !ImGui::IsAnyItemActive())
206 {
207 copy_version();
208 }
209 const bool is_escape_pressed = ImGui::IsKeyPressed(ImGuiKey_Escape, false);
210 if(draw_close_button() || is_escape_pressed)
211 {
212 is_open_ = false;
213 ImGui::CloseCurrentPopup();
214 }
215 ImGui::EndPopup();

271void about_window::draw_links()
272{
273 const float spacing = ImGui::GetStyle().ItemSpacing.x;
274 const float link_count = static_cast<float>(ABOUT_LINKS.size());
275 const float button_width = (ImGui::GetContentRegionAvail().x - spacing * (link_count - 1.0f)) / link_count;
276 ImGui::PushStyleVar(ImGuiStyleVar_FrameRounding, to_pixels(ABOUT_BUTTON_ROUNDING));
277 for(std::size_t i = 0; i < ABOUT_LINKS.size(); ++i)
278 {
279 const about_link& link = ABOUT_LINKS[i];
280 if(i > 0)
281 {
282 ImGui::SameLine();
283 }
284 if(ImGui::Button(link.label, ImVec2(button_width, to_pixels(ABOUT_BUTTON_HEIGHT))))
285 {
286 ImGui::OpenInShell(link.url);
287 }
288 ImGui::SetItemTooltipEx("%s", link.url);
289 }
290 ImGui::PopStyleVar();
291}
