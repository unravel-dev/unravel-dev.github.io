203 template<typename T>
204 auto get_asset(const hpp::uuid& uid,
205 load_flags flags = load_flags::standard,
206 load_mode mode = load_mode::immediate) -> asset_handle<T>
207 {
208 // Fast path: an already-requested asset is indexed by uid, so we can skip resolving
209 // the uid to a location through the metadata databases and re-hashing the key. A
210 // reload must go through the slow path since it has to re-dispatch the load task.
211 if(flags != load_flags::reload)
212 {
213 auto& storage = get_storage<T>();
214 std::unique_lock<std::recursive_mutex> lock(storage.container_mutex);
215 auto it = storage.container_by_uid.find(uid);
216 if(it != storage.container_by_uid.end())
217 {
218 return it->second;
219 }
220 }
221
222 auto meta = get_metadata(uid);
223 if(!meta.location.empty())
224 {
225 const auto& key = meta.location;
226 return get_asset<T>(key, flags, mode);
227 }
228
229 if(parent_)
230 {
231 return parent_->get_asset<T>(uid, flags, mode);
232 }
233 return {};
234 }

373 template<typename T>
374 void rename_asset(const std::string& key, const std::string& new_key)
375 {
376 for(auto& kvp : databases_)
377 {
378 auto& db = kvp.second;
379 db.rename_asset(key, new_key);
380 }
381
382 auto& storage = get_storage<T>();
383
384 std::lock_guard<std::recursive_mutex> lock(storage.container_mutex);
385 auto it = storage.container.find(key);
386 if(it != storage.container.end())
387 {
388 // A copy shares the handle's link; the entry goes before the new one is inserted, which may rehash.
389 auto handle = it->second;
390 storage.container.erase(it);
391 handle.set_internal_id(new_key);
392 // A load not done yet would read the old path, which the rename emptied: it is set up again for
393 // the new one, as an evicted asset's is. A loaded asset keeps what it holds.
394 if(handle.is_deferred() || handle.is_loading())
395 {
396 pool_.stop(handle.task_id());
397 handle.demote_to_deferred();
398 storage.load_from_file(pool_, handle, new_key, load_mode::deferred);
399 }
400 storage.container[new_key] = handle;
401 }
402
403 if(parent_)
404 {
405 parent_->rename_asset<T>(key, new_key);
406 }
407 }
