453 template<typename T, typename F>
454 auto load_asset_from_file_impl(const std::string& key,
455 load_flags flags,
456 load_mode mode,
457 std::recursive_mutex& container_mutex,
458 typename asset_storage<T>::request_container_t& container,
459 F&& load_func) -> asset_handle<T>
460 {
461 if(flags != load_flags::reload)
462 {
463 auto inst = find_asset_impl<T>(key, container_mutex, container);
464 if(inst)
465 {
466 return inst;
467 }
468 }
469
470 std::unique_lock<std::recursive_mutex> lock(container_mutex);
471
472 auto& handle = container[key];
473 if(load_func)
474 {
475 auto uid = add_asset(key);
476
477 if(handle.task_id())

479 pool_.stop(handle.task_id());
480 handle.invalidate();

482
483 handle.set_internal_ids(uid, key);
484 load_func(pool_, handle, key, mode);
485 }
486
487 return handle;
488 }
