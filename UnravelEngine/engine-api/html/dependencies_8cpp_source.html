63auto parse_dependency_line(const std::string& line, const fs::path& folder) -> std::string
64{
65 const std::size_t arrow = line.find(LDD_ARROW);

68 if(start == std::string::npos)

96auto scan_dependencies(const fs::path& executable, const fs::path& scanner) -> std::vector<std::string>
97{
98 const auto command = get_scanner_command(executable, scanner);
99 const auto result = subprocess::call(command);
100 if(result.retcode != 0 || result.out_output.empty())
101 {
102 throw std::runtime_error(fmt::format("The dependency scan of {} failed (exit code {}): {}",
103 executable.generic_string(),
104 result.retcode,
105 result.out_output.empty() ? std::string("no output") : result.out_output));
106 }
107 APPLOG_TRACE("Dependencies: \n{}", result.out_output);

145auto find_dependencies(const fs::path& executable, const fs::path& scanner) -> std::vector<std::string>
146{
147 struct scan

150 std::vector<std::string> dependencies;
151 };

154 std::lock_guard<std::mutex> lock(scans_mutex);
155 const auto found = scans.find(executable.string());
156 if(found != scans.end() && found->second.stamp == make_scan_stamp(executable, found->second.dependencies))
157 {
158 return found->second.dependencies;
159 }
160 // A failed scan throws before anything is kept, so the next deploy scans again.
161 auto dependencies = scan_dependencies(executable, scanner);
