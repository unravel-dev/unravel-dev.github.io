188auto create_scene_modal::complete_if_pending(defaults::scene_preset preset) -> bool
189{
190 auto& inst = get_instance();
191 if(!inst.show_requested_)
192 {
193 return false;
194 }
195
196 auto callback = std::move(inst.callback_);
197 inst.callback_ = {};
198 inst.show_requested_ = false;
199 inst.open_requested_ = false;
200 inst.selected_preset_ = preset;
201
202 if(callback)

217auto create_scene_modal::answer_if_pending(const std::string& label) -> bool
218{
219 if(!is_pending())
220 {
221 return false;
222 }
223 if(is_answer_label(label, CREATE_SCENE_CANCEL_LABEL))
224 {
225 cancel_if_pending();
226 return true;
227 }
228 if(is_answer_label(label, CREATE_SCENE_CREATE_LABEL))
229 {
230 return complete_if_pending(get_instance().selected_preset_);
231 }
232 const auto card = std::find_if(std::begin(presets),
233 std::end(presets),
234 [&label](const preset_card_info& info)
235 {
236 return is_answer_label(label, info.title);
237 });
238 return card != std::end(presets) && complete_if_pending(card->preset);
239}
