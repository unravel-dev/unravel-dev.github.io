60auto compute_slope_match(float speed, float transition_stops, float frame_time) -> float
61{
62 const float safe_speed = std::max(speed, 0.001f);
63 const float start_time = transition_stops / safe_speed;
64 const float step = 1.0f - std::pow(2.0f, -frame_time * safe_speed);
65 return frame_time / (step * start_time);
66}

121

123 if(cs_local_grid && cs_local_downsample && cs_local_blur)

133 // NOTE: the buffer's initial contents are undefined and CANNOT be seeded from the

247void auto_exposure_pass::run_histogram(gfx::render_view& rview, const run_params& params)
248{
249 const settings& config = params.config;
250 const auto input_size = params.input->get_size();
251 const std::uint32_t input_width = std::max(1u, input_size.width);
252 const std::uint32_t input_height = std::max(1u, input_size.height);
253
254 // One metering cell per 4x4 source block, capped on the long edge.
255 const auto meter_size = compute_metering_size(input_size);
256 const std::uint32_t meter_width = meter_size.width;
257 const std::uint32_t meter_height = meter_size.height;
258
259 // The per-cell log luminance the local exposure grid bins. Written every frame (one texel
260 // per cell), so the local exposure chain can be switched on without a warm-up frame.
261 const auto& log_lum_tex = rview.tex_acquire(
262 log_lum_key,
263 {.size = meter_size, .format = bgfx::TextureFormat::R16F, .flags = BGFX_TEXTURE_COMPUTE_WRITE});
264
265 gfx::render_pass pass("Auto Exposure/Histogram");
266
267 histogram_program_.program->begin();
268
269 gfx::set_texture(histogram_program_.s_hdr_input, 0, params.input->get_texture(), metering_sampler_flags);
270 params.pre_exposure.bind();
271
272 bgfx::setBuffer(1, histogram_buffer_, bgfx::Access::ReadWrite);
273 bgfx::setImage(2, log_lum_tex->native_handle(), 0, bgfx::Access::Write, bgfx::TextureFormat::R16F);
274
275 const float log_range = max_log_lum - min_log_lum;
276 const float histogram_params[4] = {min_log_lum, 1.0f / log_range, float(meter_width), float(meter_height)};
277 gfx::set_uniform(histogram_program_.u_histogram_params, histogram_params);
278
279 const float aspect_ratio = float(input_width) / float(input_height);
280 const float log2_pre_exposure = std::log2(std::max(params.pre_exposure.value, min_valid_pre_exposure));
281 const float metering_params[4] = {float(static_cast<int>(config.metering_mode)), config.metering_area, aspect_ratio, log2_pre_exposure};
282 gfx::set_uniform(histogram_program_.u_metering_params, metering_params);
283
284 const float metering_cell[4] = {float(input_width) / float(meter_width),
285 float(input_height) / float(meter_height),
286 1.0f / float(input_width),
287 1.0f / float(input_height)};
288 gfx::set_uniform(histogram_program_.u_metering_cell, metering_cell);
289
290 const std::uint32_t groups_x = (meter_width + 15) / 16;
291 const std::uint32_t groups_y = (meter_height + 15) / 16;
292 bgfx::dispatch(pass.id, histogram_program_.program->native_handle(), groups_x, groups_y, 1);
293
294 histogram_program_.program->end();
295}

297auto auto_exposure_pass::update_compensation_curve(gfx::render_view& rview, const settings& config)
298 -> gfx::texture::ptr
299{
300 if(!config.compensation_curve.is_valid())
301 {
302 rview.tex_remove(compensation_curve_key);
303 return nullptr;
304 }
305 // The curve compares by value, so the baked copy is both the cache key and the record of what
306 // was baked; no hash over the keys' bytes, which could not tell -0 from 0 anyway.
307 auto& baked_curve = rview.data().get_or_emplace<math::curve<float>>(compensation_curve_key_data);
308 bool is_built = false;
309 const auto& curve_tex =
310 rview.tex_acquire(compensation_curve_key,
311 {.size = {compensation_curve_samples, 1}, .format = bgfx::TextureFormat::R16F},
312 [&](const gfx::texture::ptr& /*tex*/)
313 {
314 is_built = true;
315 });
316 if(!curve_tex->is_valid())
317 {
318 // The allocation was refused (GPU memory budget). The texture has no initial contents to
319 // restore it from later, so drop it and build it again next frame; until then the
320 // neutral (white) curve is bound.
321 rview.tex_remove(compensation_curve_key);
322 return nullptr;
323 }
324 if(!is_built && baked_curve == config.compensation_curve)
325 {
326 return curve_tex;
327 }
328 // The curve LUT: 2^curve per sample in R16F, read back with linear filtering.
329 std::vector<std::uint16_t> lut(compensation_curve_samples);
330 const float step = (compensation_curve_max_ev - compensation_curve_min_ev) / float(compensation_curve_samples - 1);
331 for(std::uint16_t i = 0; i < compensation_curve_samples; ++i)
332 {
333 const float ev100 = compensation_curve_min_ev + step * float(i);
334 lut[i] = bx::halfFromFloat(std::exp2(config.compensation_curve.sample(ev100)));
335 }
336 const bgfx::Memory* payload = bgfx::copy(lut.data(), std::uint32_t(lut.size() * sizeof(std::uint16_t)));
337 bgfx::updateTexture2D(curve_tex->native_handle(), 0, 0, 0, 0, compensation_curve_samples, 1, payload);
338 baked_curve = config.compensation_curve;
339 return curve_tex;
340}

342void auto_exposure_pass::run_average(gfx::render_view& rview, const run_params& params)
343{
344 auto exposure_tex = rview.tex_get(exposure_key);
345 auto history_tex = rview.tex_get(history_key);
346 auto histogram_tex = rview.tex_get(histogram_key);
347 if(!exposure_tex || !history_tex || !histogram_tex)
348 {
349 return;
350 }
351
352 const settings& config = params.config;
353
354 float effective_dt = params.delta_time;
355 float force_target = 0.0f;
356 auto& state = rview.data().get_or_emplace<metering_state>(metering_state::view_key);
357 if(!histogram_bins_valid_)
358 {
359 // First dispatch since the buffer was created: its initial contents are
360 // undefined and CANNOT be seeded from the CPU -- bgfx asserts on update()
361 // for COMPUTE_WRITE buffers (the GI surface-list buffers document the same
362 // constraint). This pass zeroes the bins after reading them, so every later
363 // frame is clean; discard this one measurement (dt = 0 holds the current
364 // exposure) and leave any pending snap for the next, clean dispatch.
365 histogram_bins_valid_ = true;
366 effective_dt = 0.0f;
367 }
368 else
369 {
370 const bool snap_requested = state.force_snap;
371 state.force_snap = false;
372 // Force the target on camera cuts and when min >= max makes the range a fixed exposure.
373 if(snap_requested || params.camera_cut || config.min_ev >= config.max_ev)
374 {
375 force_target = 1.0f;
376 }
377 }
378
379 const float history_texel = float(state.history_index % history_length);
380 state.history_index = (state.history_index + 1u) % history_length;
381
382 gfx::render_pass pass("Auto Exposure/Average");
383
384 average_program_.program->begin();
385
386 bgfx::setBuffer(0, histogram_buffer_, bgfx::Access::ReadWrite);
387 bgfx::setImage(1, exposure_tex->native_handle(), 0, bgfx::Access::ReadWrite, bgfx::TextureFormat::RGBA32F);
388 bgfx::setImage(2, history_tex->native_handle(), 0, bgfx::Access::Write, bgfx::TextureFormat::RGBA32F);
389 bgfx::setImage(3, histogram_tex->native_handle(), 0, bgfx::Access::Write, bgfx::TextureFormat::R32F);
390
391 const float log_range = max_log_lum - min_log_lum;
392 const float params0[4] = {min_log_lum, log_range, config.low_percentile, config.high_percentile};
393 gfx::set_uniform(average_program_.u_average_params0, params0);
394
395 const float params1[4] = {config.min_ev, config.max_ev, config.compensation, 0.0f};
396 gfx::set_uniform(average_program_.u_average_params1, params1);
397
398 const float params2[4] = {effective_dt, config.speed_up, config.speed_down, force_target};
399 gfx::set_uniform(average_program_.u_average_params2, params2);
400
401 const float slope_up = compute_slope_match(config.speed_up, exponential_transition_stops, slope_match_frame_time);
402 const float slope_down = compute_slope_match(config.speed_down, exponential_transition_stops, slope_match_frame_time);
403 const float params3[4] = {slope_up, slope_down, exponential_transition_stops, history_texel};
404 gfx::set_uniform(average_program_.u_average_params3, params3);
405
406 // The local exposure shape, for the average local exposure the pre-exposure folds in. The
407 // detail strength is deliberately absent: with no spatial term a bin's own luminance IS its
408 // base, so detail cancels out of the average.
409 const float params4[4] = {config.local_highlight_contrast,
410 config.local_shadow_contrast,
411 config.local_middle_grey_bias,
412 0.0f};
413 gfx::set_uniform(average_program_.u_average_params4, params4);
414
415 // The compensation curve LUT, addressed with texel centres at the sample EV100 values.
416 // Without a curve the white texture reads 1 (no extra stops).
417 const auto curve_tex = update_compensation_curve(rview, config);
418 const float samples = float(compensation_curve_samples);
419 const float lut_scale = (samples - 1.0f) / (samples * (compensation_curve_max_ev - compensation_curve_min_ev));
420 const float lut_bias = 0.5f / samples - compensation_curve_min_ev * lut_scale;
421 const float params5[4] = {lut_scale, lut_bias, 0.0f, 0.0f};
422 gfx::set_uniform(average_program_.u_average_params5, params5);
423 gfx::set_texture(average_program_.s_compensation_curve,
424 4,
425 curve_tex ? curve_tex : default_textures::get().white_texture(),
426 BGFX_SAMPLER_U_CLAMP | BGFX_SAMPLER_V_CLAMP);
427
428 bgfx::dispatch(pass.id, average_program_.program->native_handle(), 1, 1, 1);
429
430 average_program_.program->end();
431}

433void auto_exposure_pass::run_local_exposure(gfx::render_view& rview, const run_params& params)
434{
435 const settings& config = params.config;
436 const bool wants_local_exposure =
437 config.is_local_exposure_enabled() && local_grid_program_.program && local_grid_program_.program->is_valid() &&
438 local_downsample_program_.program && local_downsample_program_.program->is_valid() &&
439 local_blur_program_.program && local_blur_program_.program->is_valid();
440 if(!wants_local_exposure)
441 {
442 // Neutral settings: drop the buffers so the consumer's "no view" branch is the only
443 // state to reason about, and a view that never uses local exposure keeps its memory.
444 rview.tex_remove(local_grid_key);
445 rview.tex_remove(local_downsampled_key);
446 rview.tex_remove(local_blur_temp_key);
447 rview.tex_remove(local_blurred_key);
448 return;
449 }
450 auto log_lum_tex = rview.tex_find(log_lum_key);
451 if(!log_lum_tex)
452 {
453 return;
454 }
455
456 const auto meter_size = compute_metering_size(params.input->get_size());
457 const std::uint32_t tiles_x =
458 std::max(1u, (meter_size.width + local_exposure_tile_cells - 1) / local_exposure_tile_cells);
459 const std::uint32_t tiles_y =
460 std::max(1u, (meter_size.height + local_exposure_tile_cells - 1) / local_exposure_tile_cells);
461 // The grid is the flattened 3D atlas: one row per tile row, each tile's slices end to end.
462 const usize32_t grid_size{tiles_x * local_exposure_slices, tiles_y};
463 auto grid_tex = ensure_compute_target(rview, local_grid_key, grid_size, bgfx::TextureFormat::RGBA32F);
464
465 auto& layout = rview.data().get_or_emplace<local_exposure_view>(local_layout_key);
466 layout.tiles_x = float(tiles_x);
467 layout.tiles_y = float(tiles_y);
468 layout.grid_uv_scale = {float(meter_size.width) / float(tiles_x * local_exposure_tile_cells),
469 float(meter_size.height) / float(tiles_y * local_exposure_tile_cells)};
470
471 const float log_range = max_log_lum - min_log_lum;
472 {
473 gfx::render_pass pass("Auto Exposure/Local Grid");
474 local_grid_program_.program->begin();
475 gfx::set_texture(local_grid_program_.s_exposure_log_lum, 0, log_lum_tex, local_point_clamp);
476 bgfx::setImage(1, grid_tex->native_handle(), 0, bgfx::Access::Write, bgfx::TextureFormat::RGBA32F);
477 const float grid_params[4] = {float(meter_size.width),
478 float(meter_size.height),
479 float(local_exposure_tile_cells),
480 float(local_exposure_slices)};
481 gfx::set_uniform(local_grid_program_.u_local_grid_params, grid_params);
482 const float grid_range[4] = {min_log_lum, 1.0f / log_range, float(tiles_x), float(tiles_y)};
483 gfx::set_uniform(local_grid_program_.u_local_grid_range, grid_range);
484 // One workgroup per tile (cs_local_exposure_grid.sc).
485 bgfx::dispatch(pass.id, local_grid_program_.program->native_handle(), tiles_x, tiles_y, 1);
486 local_grid_program_.program->end();
487 }
488 run_local_exposure_blur(rview, config, meter_size);
489}

491void auto_exposure_pass::run_local_exposure_blur(gfx::render_view& rview,
492 const settings& config,
493 const usize32_t& meter_size)
494{
495 auto log_lum_tex = rview.tex_find(log_lum_key);
496 const usize32_t blur_size{std::max(1u, (meter_size.width + local_exposure_blur_cells - 1) / local_exposure_blur_cells),
497 std::max(1u, (meter_size.height + local_exposure_blur_cells - 1) / local_exposure_blur_cells)};
498 auto downsampled_tex = ensure_compute_target(rview, local_downsampled_key, blur_size, bgfx::TextureFormat::R16F);
499 auto temp_tex = ensure_compute_target(rview, local_blur_temp_key, blur_size, bgfx::TextureFormat::R16F);
500 auto blurred_tex = ensure_compute_target(rview, local_blurred_key, blur_size, bgfx::TextureFormat::R16F);
501
502 // The view covers meter / blur_cells texels; the last texel is partial.
503 const float covered_width = float(meter_size.width) / float(local_exposure_blur_cells);
504 const float covered_height = float(meter_size.height) / float(local_exposure_blur_cells);
505 auto& layout = rview.data().get_or_emplace<local_exposure_view>(local_layout_key);
506 layout.blurred_uv_scale = {covered_width / float(blur_size.width), covered_height / float(blur_size.height)};
507
508 const std::uint32_t groups_x = (blur_size.width + 7) / 8;
509 const std::uint32_t groups_y = (blur_size.height + 7) / 8;
510 {
511 gfx::render_pass pass("Auto Exposure/Local Downsample");
512 local_downsample_program_.program->begin();
513 gfx::set_texture(local_downsample_program_.s_exposure_log_lum, 0, log_lum_tex, local_point_clamp);
514 bgfx::setImage(1, downsampled_tex->native_handle(), 0, bgfx::Access::Write, bgfx::TextureFormat::R16F);
515 const float downsample_params[4] = {float(meter_size.width),
516 float(meter_size.height),
517 float(blur_size.width),
518 float(blur_size.height)};
519 gfx::set_uniform(local_downsample_program_.u_local_downsample_params, downsample_params);
520 bgfx::dispatch(pass.id, local_downsample_program_.program->native_handle(), groups_x, groups_y, 1);
521 local_downsample_program_.program->end();
522 }
523 // The kernel size is a DIAMETER as a percentage of the view width.
524 const float radius = std::clamp(covered_width * std::max(config.local_blurred_kernel_percent, 0.0f) * 0.01f * 0.5f,
525 1e-3f,
526 local_blur_max_radius);
527 const std::array<std::pair<gfx::texture::ptr, gfx::texture::ptr>, 2> blur_passes{
528 std::make_pair(downsampled_tex, temp_tex),
529 std::make_pair(temp_tex, blurred_tex)};
530 for(std::uint32_t axis = 0; axis < 2; ++axis)
531 {
532 gfx::render_pass pass(axis == 0 ? "Auto Exposure/Local Blur X" : "Auto Exposure/Local Blur Y");
533 local_blur_program_.program->begin();
534 gfx::set_texture(local_blur_program_.s_local_blur_input, 0, blur_passes[axis].first, local_point_clamp);
535 bgfx::setImage(1, blur_passes[axis].second->native_handle(), 0, bgfx::Access::Write, bgfx::TextureFormat::R16F);
536 const float blur_params[4] = {float(blur_size.width), float(blur_size.height), radius, float(axis)};
537 gfx::set_uniform(local_blur_program_.u_local_blur_params, blur_params);
538 bgfx::dispatch(pass.id, local_blur_program_.program->native_handle(), groups_x, groups_y, 1);
539 local_blur_program_.program->end();
540 }
541}

546

548

556void auto_exposure_pass::update_gpu_pre_exposure(gfx::render_view& rview,
557 const gfx::texture::ptr& environment_sh,
558 float manual_exposure,
559 bool is_auto_exposure_active)
560{
561 // begin() links the program once its shader has loaded (it may compile after init).
562 if(!environment_sh || !pre_exposure_program_.program || !pre_exposure_program_.program->begin())
563 {
564 return;
565 }
566 // Without auto exposure the share returns to 1, and the history correction carries that step too.
567 const auto exposure_tex = is_auto_exposure_active ? rview.tex_find(exposure_key) : nullptr;
568 const auto& input = exposure_tex ? exposure_tex : default_textures::get().white_texture();
569 const float params[4] = {manual_exposure, pre_exposure_state::min_value, pre_exposure_state::max_value, 0.0f};
570
571 gfx::render_pass pass("Auto Exposure/Pre-Exposure");
572 gfx::set_uniform(pre_exposure_program_.u_pre_exposure_params, params);
573 gfx::set_texture(pre_exposure_program_.s_exposure, 0, input);
574 bgfx::setImage(1, environment_sh->native_handle(), 0, bgfx::Access::ReadWrite, bgfx::TextureFormat::RGBA32F);
575 bgfx::dispatch(pass.id, pre_exposure_program_.program->native_handle(), 1, 1, 1);
576 pre_exposure_program_.program->end();
577}
