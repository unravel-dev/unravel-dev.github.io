81auto fit_text(ImFont* font, float size, const std::string& text, float max_width) -> std::string
82{
83 const auto get_width = [&](const std::string& candidate)
84 {
85 return font->CalcTextSizeA(size, FLT_MAX, 0.0f, candidate.c_str()).x;
86 };
87 if(get_width(text) <= max_width)
88 {
89 return text;
90 }
91 // Byte offset of every code point boundary: a cut never splits a UTF-8 sequence.
92 std::vector<size_t> boundaries{0};
93 const char* const begin = text.c_str();
94 const char* const end = begin + text.size();
95 for(const char* it = begin; it < end;)
96 {
97 unsigned int code_point = 0;
98 it += std::max(ImTextCharFromUtf8(&code_point, it, end), 1);
99 boundaries.push_back(size_t(it - begin));
100 }
101 // Binary search on the number of code points kept: `fits` does, `too_long` does not.
102 size_t fits = 0;
103 size_t too_long = boundaries.size() - 1;
104 while(too_long - fits > 1)
105 {
106 const size_t middle = (fits + too_long) / 2;
107 if(get_width(text.substr(0, boundaries[middle]) + ELLIPSIS) <= max_width)
108 {
109 fits = middle;
110 }
111 else
112 {
113 too_long = middle;
114 }
115 }
116 return text.substr(0, boundaries[fits]) + ELLIPSIS;
117}

215auto describe_condition(const animation_controller& controller, const animation_condition_def& condition) -> std::string
216{
217 const int index = controller.find_parameter(condition.parameter);
218 if(index < 0)
219 {
220 return fmt::format("{} (no such parameter)", condition.parameter.empty() ? "?" : condition.parameter);
221 }
222 const auto& name = condition.parameter;
223 switch(controller.parameters[size_t(index)].type)
224 {
225 case animation_parameter_type::float_value:
226 return fmt::format("{} {} {:.2f}", name, condition.mode == animation_condition_mode::less ? "<" : ">", condition.threshold);
227 case animation_parameter_type::int_value:
228 {
229 const char* op = condition.mode == animation_condition_mode::less ? "<"
230 : condition.mode == animation_condition_mode::equals ? "=="
231 : condition.mode == animation_condition_mode::not_equals ? "!="
232 : ">";
233 return fmt::format("{} {} {}", name, op, int(std::lround(condition.threshold)));
234 }
235 case animation_parameter_type::bool_value:
236 return fmt::format("{} is {}", name, condition.mode == animation_condition_mode::is_false ? "false" : "true");
237 case animation_parameter_type::trigger:
238 return fmt::format("{} (trigger)", name);
239 }
240 return name;
241}
