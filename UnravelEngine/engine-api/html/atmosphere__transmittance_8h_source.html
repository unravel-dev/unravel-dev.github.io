41inline auto compute_atmosphere_sun_transmittance(const math::vec3& direction_to_sun) -> math::vec3
42{
43 namespace ad = atmosphere_defaults;
44 const float elevation_sin = std::clamp(direction_to_sun.y, -1.0f, 1.0f);
45 const float elevation_cos = std::sqrt(std::max(0.0f, 1.0f - elevation_sin * elevation_sin));
46 // 2D: the transmittance is symmetric about the zenith, so only the elevation matters.
47 const float origin_x = 0.0f;
48 const float origin_y = ad::bottom_radius_km + ad::origin_altitude_km;
49 const float b = origin_y * elevation_sin;
50 const float c = origin_y * origin_y - ad::top_radius_km * ad::top_radius_km;
51 const float ray_length = -b + std::sqrt(std::max(0.0f, b * b - c));
52 const float step = ray_length / float(ad::transmittance_samples);
53 math::vec3 optical_depth(0.0f);
54 for(int i = 0; i < ad::transmittance_samples; ++i)
55 {
56 // Samples at the start of each step.
57 const float t = step * float(i);
58 const float x = origin_x + elevation_cos * t;
59 const float y = origin_y + elevation_sin * t;
60 const float height = std::sqrt(x * x + y * y) - ad::bottom_radius_km;
61 const float rayleigh_density = std::exp(-height / ad::rayleigh_scale_height_km);
62 const float mie_density = std::exp(-height / ad::mie_scale_height_km);
63 const float ozone_density =
64 std::clamp(1.0f - std::abs(height - ad::ozone_tip_altitude_km) / ad::ozone_half_width_km, 0.0f, 1.0f);
65 optical_depth += step * (rayleigh_density * math::vec3(ad::rayleigh_scattering_r,
66 ad::rayleigh_scattering_g,
67 ad::rayleigh_scattering_b) +
68 mie_density * math::vec3(ad::mie_extinction) +
69 ozone_density * math::vec3(ad::ozone_absorption_r,
70 ad::ozone_absorption_g,
71 ad::ozone_absorption_b));
72 }
73 return math::vec3(std::exp(-optical_depth.x), std::exp(-optical_depth.y), std::exp(-optical_depth.z));
74}
