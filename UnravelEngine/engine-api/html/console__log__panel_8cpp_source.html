230auto console_log_panel::snapshot_logs(level::level_enum min_level, size_t max_count, uint64_t after_id) const
231 -> std::vector<log_snapshot_entry>
232{
233 std::lock_guard<std::recursive_mutex> lock(entries_mutex_);
234 std::vector<log_snapshot_entry> matched;
235 matched.reserve(std::min(max_count, entries_.size()));
236 for(const auto& entry : entries_)
237 {
238 if(entry.id <= after_id)
239 {
240 continue;
241 }
242 if(entry.level < min_level)
243 {
244 continue;
245 }
246 log_snapshot_entry snap;
247 snap.id = entry.id;
248 snap.level = entry.level;
249 snap.text.assign(entry.formatted.begin(), entry.formatted.end());
250 snap.filename = entry.source.filename;
251 snap.funcname = entry.source.funcname;
252 snap.line = entry.source.line;
253 snap.is_device_system = entry.is_device_system;
254 matched.push_back(std::move(snap));
255 }
256 if(matched.size() > max_count)
257 {
258 matched.erase(matched.begin(), matched.end() - static_cast<std::ptrdiff_t>(max_count));
259 }
260 return matched;
261}

328void console_log_panel::notify_new_errors()
329{
330 if(first_unnotified_id_ == NO_ENTRY_ID)
331 {
332 return;
333 }
334 uint64_t errors_count = 0;
335 {
336 std::lock_guard<std::recursive_mutex> lock(entries_mutex_);
337 // By id, not by position: once the ring is full, every new entry pushes the oldest out.
338 const auto first_id = uint64_t(first_unnotified_id_.load());
339 errors_count = uint64_t(std::count_if(entries_.begin(),
340 entries_.end(),
341 [this, first_id](const log_entry& entry)
342 {
343 return entry.id >= first_id && entry.level == level::err &&
344 is_source_shown(entry);
345 }));
346 first_unnotified_id_ = NO_ENTRY_ID;
347 }
348 if(errors_count > 0)
349 {
350 const std::string message = fmt::format("{} Error(s)...", errors_count);
351 ImGui::PushNotification({ImGuiToastType_Error, CONSOLE_ERROR_TOAST_MILLISECONDS, message.c_str()});
352 }
