9// The velocity drag in 1/s. The velocity closes 63% of the gap to its target every
10// 1 / 10 s: full speed and a full stop both take about 0.3 s.
11constexpr float VELOCITY_DAMPING = 10.0f;
12// The rest of a coast at this speed is half a millimetre. Stop there, otherwise the view creeps
13// for seconds and never lets the temporal histories settle.
14constexpr float REST_SPEED = 0.005f;

18auto camera_controller::update(const math::vec3& move_direction, float fly_speed, float dt) -> math::vec3
19{
20 const math::vec3 target_velocity = move_direction * fly_speed;
21 const math::vec3 excess_velocity = velocity_ - target_velocity;
22 const float decay = std::exp(-VELOCITY_DAMPING * dt);
23 const math::vec3 displacement = target_velocity * dt + excess_velocity * ((1.0f - decay) / VELOCITY_DAMPING);
24 velocity_ = target_velocity + excess_velocity * decay;
25 const bool is_coasting = math::dot(target_velocity, target_velocity) == 0.0f;
26 if(is_coasting && math::dot(velocity_, velocity_) < REST_SPEED * REST_SPEED)
27 {
28 stop();
29 }
30 return displacement;
31}
