56auto sample_medium(float height_km) -> medium_sample
57{
58 medium_sample m;
59 const float rayleigh_density = std::exp(-height_km / ad::rayleigh_scale_height_km);
60 const float mie_density = std::exp(-height_km / ad::mie_scale_height_km);
61 const float ozone_density =
62 std::clamp(1.0f - std::abs(height_km - ad::ozone_tip_altitude_km) / ad::ozone_half_width_km, 0.0f, 1.0f);
63 m.rayleigh_scattering =
64 rayleigh_density * math::vec3(ad::rayleigh_scattering_r, ad::rayleigh_scattering_g, ad::rayleigh_scattering_b);
65 m.mie_scattering = mie_density * mie_scattering;
66 m.extinction = m.rayleigh_scattering + math::vec3(mie_density * ad::mie_extinction) +
67 ozone_density * math::vec3(ad::ozone_absorption_r, ad::ozone_absorption_g, ad::ozone_absorption_b);
68 return m;
69}

72auto intersect_sphere(const math::vec3& origin, const math::vec3& direction, float radius) -> float
73{
74 const float b = math::dot(origin, direction);
75 const float c = math::dot(origin, origin) - radius * radius;
76 const float discriminant = b * b - c;
77 if(discriminant < 0.0f)
78 {
79 return -1.0f;
80 }
81 const float root = std::sqrt(discriminant);
82 const float near_t = -b - root;
83 const float far_t = -b + root;
84 if(far_t < 0.0f)
85 {
86 return -1.0f;
87 }
88 return near_t > 0.0f ? near_t : far_t;
89}

112 auto get_ratio(float sun_elevation_sin) const -> float
113 {
114 const float x = std::clamp(sun_elevation_sin, 0.0f, 1.0f) * float(ratio_entries - 1);
115 const int i0 = std::min(int(x), ratio_entries - 2);
116 const float f = x - float(i0);
117 return ratios_[i0] * (1.0f - f) + ratios_[i0 + 1] * f;
118 }

121 void build_transmittance()
122 {
123 transmittance_.resize(size_t(transmittance_heights) * transmittance_cosines);
124 for(int h = 0; h < transmittance_heights; ++h)
125 {
126 const float u = float(h) / float(transmittance_heights - 1);
127 const float radius = ad::bottom_radius_km + (ad::top_radius_km - ad::bottom_radius_km) * u * u;
128 for(int c = 0; c < transmittance_cosines; ++c)
129 {
130 const float cosine = -1.0f + 2.0f * float(c) / float(transmittance_cosines - 1);
131 const math::vec3 direction(std::sqrt(std::max(0.0f, 1.0f - cosine * cosine)), cosine, 0.0f);
132 transmittance_[size_t(h) * transmittance_cosines + c] =
133 integrate_transmittance(math::vec3(0.0f, radius, 0.0f), direction);
134 }
135 }

138 auto integrate_transmittance(const math::vec3& origin, const math::vec3& direction) const -> math::vec3
139 {
140 if(hits_ground(origin, direction))
141 {
142 return math::vec3(0.0f);
143 }
144 const float length = intersect_sphere(origin, direction, ad::top_radius_km);
145 const float step = length / float(ad::transmittance_samples);
146 math::vec3 optical_depth(0.0f);
147 for(int i = 0; i < ad::transmittance_samples; ++i)
148 {
149 const math::vec3 p = origin + direction * ((float(i) + 0.5f) * step);
150 optical_depth += sample_medium(math::length(p) - ad::bottom_radius_km).extinction * step;
151 }
152 return exp3(-optical_depth);
153 }

188 auto integrate_multi_scattering(const math::vec3& origin,
189 const math::vec3& sun,
190 const std::vector<math::vec3>& directions) const -> math::vec3
191 {
192 math::vec3 second_order(0.0f);
193 math::vec3 transfer(0.0f);
194 const float isotropic_phase = 1.0f / (4.0f * pi);
195 for(const auto& direction : directions)
196 {
197 const float ground_t = hits_ground(origin, direction) ? intersect_sphere(origin, direction, ad::bottom_radius_km) : -1.0f;
198 const float length = ground_t > 0.0f ? ground_t : intersect_sphere(origin, direction, ad::top_radius_km);
199 const float step = length / float(multi_scattering_steps);
200 math::vec3 view_transmittance(1.0f);
201 for(int i = 0; i < multi_scattering_steps; ++i)
202 {
203 const math::vec3 p = origin + direction * ((float(i) + 0.5f) * step);
204 const medium_sample m = sample_medium(math::length(p) - ad::bottom_radius_km);
205 const math::vec3 scattering = m.rayleigh_scattering + math::vec3(m.mie_scattering);
206 const math::vec3 step_transmittance = exp3(-m.extinction * step);
207 const math::vec3 integral = (math::vec3(1.0f) - step_transmittance) / math::max(m.extinction, math::vec3(1e-9f));
208 second_order += view_transmittance * scattering * sun_transmittance(p, sun) * isotropic_phase * integral;
209 transfer += view_transmittance * scattering * integral;
210 view_transmittance *= step_transmittance;
211 }
212 if(ground_t > 0.0f)
213 {
214 const math::vec3 ground = origin + direction * ground_t;
215 const float sun_cosine = std::max(math::dot(math::normalize(ground), sun), 0.0f);
216 second_order += view_transmittance * sun_transmittance(ground, sun) * sun_cosine * ground_albedo / pi;
217 }
218 }
219 const float inv_count = 1.0f / float(directions.size());
220 second_order *= inv_count;
221 transfer *= inv_count;
222 return second_order / math::max(math::vec3(1.0f) - transfer, math::vec3(1e-4f));
223 }

225 auto multi_scattering(const math::vec3& p, const math::vec3& sun) const -> math::vec3
226 {
227 const float radius = math::length(p);

229 const float u = (radius - ad::bottom_radius_km) / (ad::top_radius_km - ad::bottom_radius_km);
230 return sample_table(multi_scattering_, multi_scattering_heights, multi_scattering_cosines, u, (cosine + 1.0f) * 0.5f, true);

234 auto integrate_sky_radiance(const math::vec3& origin, const math::vec3& direction, const math::vec3& sun) const -> math::vec3
235 {
236 const float length = intersect_sphere(origin, direction, ad::top_radius_km);
237 const float step = length / float(irradiance_ray_steps);
238 const float cosine = math::dot(direction, sun);
239 const float rayleigh_phase = 3.0f / (16.0f * pi) * (1.0f + cosine * cosine);
240 const float g2 = mie_anisotropy * mie_anisotropy;
241 const float mie_phase =
242 (1.0f - g2) / (4.0f * pi * std::pow(1.0f + g2 - 2.0f * mie_anisotropy * cosine, 1.5f));
243 math::vec3 radiance(0.0f);
244 math::vec3 view_transmittance(1.0f);
245 for(int i = 0; i < irradiance_ray_steps; ++i)
246 {
247 const math::vec3 p = origin + direction * ((float(i) + 0.5f) * step);
248 const medium_sample m = sample_medium(math::length(p) - ad::bottom_radius_km);
249 const math::vec3 scattering = m.rayleigh_scattering + math::vec3(m.mie_scattering);
250 const math::vec3 source =
251 sun_transmittance(p, sun) * (m.rayleigh_scattering * rayleigh_phase + math::vec3(m.mie_scattering * mie_phase)) +
252 multi_scattering(p, sun) * scattering;
253 const math::vec3 step_transmittance = exp3(-m.extinction * step);
254 radiance += view_transmittance * source * (math::vec3(1.0f) - step_transmittance) /
255 math::max(m.extinction, math::vec3(1e-9f));
256 view_transmittance *= step_transmittance;
257 }
258 return radiance;
259 }

261 auto integrate_horizontal_irradiance(float sun_elevation_sin) const -> float
262 {
263 const float sun_cos = std::sqrt(std::max(0.0f, 1.0f - sun_elevation_sin * sun_elevation_sin));
264 const math::vec3 sun(sun_cos, sun_elevation_sin, 0.0f);
265 const math::vec3 origin(0.0f, ad::bottom_radius_km + observer_altitude_km, 0.0f);
266 const float d_zenith = 0.5f * pi / float(irradiance_zenith_steps);
267 const float d_azimuth = 2.0f * pi / float(irradiance_azimuth_steps);
268 math::vec3 irradiance(0.0f);
269 for(int z = 0; z < irradiance_zenith_steps; ++z)
270 {
271 const float zenith = (float(z) + 0.5f) * d_zenith;
272 for(int a = 0; a < irradiance_azimuth_steps; ++a)
273 {
274 const float azimuth = (float(a) + 0.5f) * d_azimuth;
275 const math::vec3 direction(std::sin(zenith) * std::cos(azimuth), std::cos(zenith), std::sin(zenith) * std::sin(azimuth));
276 irradiance += integrate_sky_radiance(origin, direction, sun) * (std::cos(zenith) * std::sin(zenith) * d_zenith * d_azimuth);
277 }
278 }
279 return math::dot(irradiance, luminance_weights);
280 }

282 void build_ratios()
283 {
284 for(int i = 0; i < ratio_entries; ++i)
285 {
286 // The horizon entry samples just above it: a sun exactly on the horizon is half below the planet's edge.
287 const float elevation_sin = std::max(float(i) / float(ratio_entries - 1), 0.01f);
288 ratios_[i] = sky_irradiance_calibration * integrate_horizontal_irradiance(elevation_sin);

292 static auto fibonacci_sphere(int count) -> std::vector<math::vec3>
293 {
294 std::vector<math::vec3> directions;
295 directions.reserve(size_t(count));
296 const float golden_angle = pi * (3.0f - std::sqrt(5.0f));
297 for(int i = 0; i < count; ++i)
298 {
299 const float y = 1.0f - 2.0f * (float(i) + 0.5f) / float(count);
300 const float ring = std::sqrt(std::max(0.0f, 1.0f - y * y));
301 const float angle = golden_angle * float(i);
302 directions.emplace_back(std::cos(angle) * ring, y, std::sin(angle) * ring);
303 }
304 return directions;
305 }

308 static auto sample_table(const std::vector<math::vec3>& table, int rows, int columns, float u, float v, bool is_cell_centred)
309 -> math::vec3
310 {
311 const float offset = is_cell_centred ? 0.5f : 0.0f;
312 const float scale_u = is_cell_centred ? float(rows) : float(rows - 1);
313 const float scale_v = is_cell_centred ? float(columns) : float(columns - 1);
314 const float x = std::clamp(u * scale_u - offset, 0.0f, float(rows - 1) - 1e-3f);
315 const float y = std::clamp(v * scale_v - offset, 0.0f, float(columns - 1) - 1e-3f);
316 const int x0 = int(x);
317 const int y0 = int(y);
318 const int x1 = std::min(x0 + 1, rows - 1);
319 const int y1 = std::min(y0 + 1, columns - 1);
320 const float fx = x - float(x0);
321 const float fy = y - float(y0);
322 const auto at = [&](int r, int c) { return table[size_t(r) * columns + c]; };
323 return (at(x0, y0) * (1.0f - fx) + at(x1, y0) * fx) * (1.0f - fy) + (at(x0, y1) * (1.0f - fx) + at(x1, y1) * fx) * fy;
324 }
