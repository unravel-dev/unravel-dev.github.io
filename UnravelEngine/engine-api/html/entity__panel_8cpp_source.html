120auto entity_panel::get_entity_prefab_role(entt::handle entity) -> prefab_role
121{
122 if(!entity)
123 {
124 return prefab_role::none;
125 }
126 // The root of the document being edited is the document, not an instance of it.
127 if(is_authoring_root(entity))
128 {
129 return prefab_role::none;
130 }
131
132 const auto* prefab_comp = entity.try_get<prefab_component>();
133
134 // Whatever instance contains this one, if any. Nothing without one is local *to* anything
135 // - an instance placed straight into a scene is just an instance, and the most common
136 // thing on screen, so it must not be flagged as an exception.
137 const auto* trans_comp = entity.try_get<transform_component>();
138 auto current = trans_comp != nullptr ? trans_comp->get_parent() : entt::handle{};
139 while(current && !current.all_of<prefab_component>())
140 {
141 const auto* parent_trans = current.try_get<transform_component>();
142 current = parent_trans != nullptr ? parent_trans->get_parent() : entt::handle{};
143 }
144 const bool is_nested = static_cast<bool>(current);
145
146 if(prefab_comp != nullptr)
147 {
148 if(!is_nested)
149 {
150 return prefab_role::instance;
151 }
152
153 // A nil instance id is the whole distinction: an instance the containing prefab
154 // supplies is a slot in that file and carries its id, while one added or cloned here
155 // is nobody's slot and never gets one until this subtree is itself saved as a prefab.
156 return prefab_comp->instance_id.is_nil() ? prefab_role::local_instance : prefab_role::linked_instance;
157 }
158
159 if(!is_nested)
160 {
161 return prefab_role::none;
162 }
163
164 // Inside an instance: the asset's entities arrived with a prefab id, anything else was
165 // added here.
166 return entity.all_of<prefab_id_component>() ? prefab_role::asset_content : prefab_role::local_content;
167}

169auto entity_panel::describe_prefab_role(prefab_role role) -> const char*
170{
171 switch(role)
172 {
173 case prefab_role::instance:
174 return "An instance of a prefab.";
175 case prefab_role::linked_instance:
176 return "Comes from the prefab that contains it.\n"
177 "Edits to that prefab reach it, and it goes if the prefab drops it.";
178 case prefab_role::local_instance:
179 return "Added here, not part of any prefab file.\n"
180 "No prefab above it will replace or remove it.";
181 case prefab_role::asset_content:
182 return "Comes from this instance's prefab.";
183 case prefab_role::local_content:
184 return "Added inside this instance. Its prefab knows nothing about it.";
185 case prefab_role::none:
186 break;
187 }
188 return "";
189}

191auto entity_panel::get_entity_display_color(entt::handle entity) -> ImVec4
192{
193 auto& trans_comp = entity.get<transform_component>();
194 bool is_bone = entity.all_of<bone_component>();
195 bool is_submesh = entity.all_of<submesh_component>();
196 bool is_active_global = trans_comp.is_active_global();
197 bool has_source = entity.any_of<prefab_component, prefab_id_component>();
198 bool has_broken_source = false;
199
200 if(auto pfb = entity.try_get<prefab_component>())
201 {
202 if(!pfb->source)
203 {
204 has_source = false;
205 has_broken_source = true;
206 }
207 }
208
209 auto col = ImGui::GetStyleColorVec4(ImGuiCol_Text);
210
211 // Blue for anything a prefab supplies, amber for anything added where it stands. The
212 // second is the one worth spotting: it is what no prefab above will ever update, and what
213 // used to be indistinguishable from the rest of a prefab subtree.
214 const auto role = get_entity_prefab_role(entity);
215 const bool is_local = role == prefab_role::local_instance || role == prefab_role::local_content;
216
217 col = ImLerp(col, ImVec4(0.5f, 0.85f, 1.0f, 1.0f), float(has_source && !is_local) * 0.5f);
218 col = ImLerp(col, ImVec4(1.0f, 0.78f, 0.35f, 1.0f), float(is_local) * 0.55f);
219 col = ImLerp(col, ImVec4(1.0f, 0.0f, 0.0f, 1.0f), float(has_broken_source) * 0.5f);
220 col = ImLerp(col, ImVec4(0.5f, 0.85f, 1.0f, 1.0f), float(is_bone) * 0.5f);
221 col = ImLerp(col, ImVec4(0.8f, 0.4f, 0.4f, 1.0f), float(is_submesh) * 0.5f);
222 col = ImLerp(col, ImVec4(col.x * 0.75f, col.y * 0.75f, col.z * 0.75f, col.w * 0.75f), float(!is_active_global));
223
224 return col;
225}
