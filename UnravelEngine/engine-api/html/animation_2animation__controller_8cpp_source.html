24auto is_condition_compatible(animation_condition_mode mode, animation_parameter_type type) -> bool
25{
26 switch(mode)
27 {
28 case animation_condition_mode::greater:
29 case animation_condition_mode::less:
30 case animation_condition_mode::equals:
31 case animation_condition_mode::not_equals:
32 return is_numeric(type);
33 case animation_condition_mode::is_true:
34 return is_boolean(type);
35 case animation_condition_mode::is_false:
36 return type == animation_parameter_type::bool_value;

93void validate_state(const animation_controller& controller,
94 const animation_state_def& state,
95 const std::string& where,
96 bool is_base,
97 std::vector<std::string>& problems)
98{
99 switch(state.motion)
100 {
101 case animation_motion_type::clip:
102 // An empty state on a layer above the base is how a layer rests: it fades out and
103 // leaves the layers below it alone. The base layer has nothing below it.
104 if(!state.clip && is_base)
105 {
106 problems.push_back(fmt::format("{} has no clip", where));
107 }
108 break;
109 case animation_motion_type::blend_1d:
110 case animation_motion_type::blend_2d_cartesian:
111 case animation_motion_type::blend_2d_directional:
112 {
113 if(state.points.empty())
114 {
115 problems.push_back(fmt::format("{} has no blend points", where));
116 }
117 for(const auto& point : state.points)
118 {
119 if(!point.clip)
120 {
121 problems.push_back(fmt::format("{} has a blend point without a clip", where));
122 break;
123 }
124 }
125 if(state.parameter_x.empty())
126 {
127 problems.push_back(fmt::format("{} has no X parameter", where));
128 }
129 validate_float_parameter(controller, state.parameter_x, where, problems);
130 if(state.motion != animation_motion_type::blend_1d)
131 {
132 if(state.parameter_y.empty())
133 {
134 problems.push_back(fmt::format("{} has no Y parameter", where));
135 }
136 validate_float_parameter(controller, state.parameter_y, where, problems);
137 }
138 break;
139 }
140 }
141 validate_float_parameter(controller, state.speed_parameter, where, problems);
142}

144void validate_transition(const animation_controller& controller,
145 const animation_layer_def& layer,
146 const animation_transition_def& transition,
147 const std::string& layer_label,
148 std::vector<std::string>& problems)
149{
150 const std::string where =
151 fmt::format("{} transition {} -> {}", layer_label, describe_source(transition), transition.destination);
152 const int source = transition.source.empty() ? -1 : find_animation_state(layer, transition.source);
153 if(!transition.source.empty() && source < 0)
154 {
155 problems.push_back(fmt::format("{}: unknown source state", where));
156 }
157 if(find_animation_state(layer, transition.destination) < 0)
158 {
159 problems.push_back(fmt::format("{}: unknown destination state", where));
160 }
161 if(!transition.has_exit_time && transition.conditions.empty())
162 {
163 problems.push_back(fmt::format("{}: no conditions and no exit time, it can never fire", where));
164 }
165 if(transition.has_exit_time && transition.exit_time > 1.0f && source >= 0 && !layer.states[size_t(source)].loop)
166 {
167 // A state that plays once stops at its end: its time never passes 1.
168 problems.push_back(fmt::format("{}: exit time above 1 on a state that does not loop, it can never fire", where));
169 }
170 if(transition.source.empty() && transition.can_transition_to_self && !has_trigger_condition(controller, transition))
171 {
172 problems.push_back(fmt::format(
173 "{}: Any State to itself without a trigger restarts the state on every frame its conditions hold", where));
174 }
175 if(!transition.source.empty() && transition.source == transition.destination && !transition.has_exit_time &&
176 !has_trigger_condition(controller, transition))
177 {
178 problems.push_back(fmt::format(
179 "{}: a state to itself without an exit time or a trigger restarts it on every frame its conditions hold",
180 where));
181 }
182 for(const auto& condition : transition.conditions)
183 {
184 const int index = controller.find_parameter(condition.parameter);
185 if(index < 0)
186 {
187 problems.push_back(fmt::format("{}: condition on unknown parameter '{}'", where, condition.parameter));
188 continue;
189 }
190 if(!is_condition_compatible(condition.mode, controller.parameters[size_t(index)].type))
191 {
192 problems.push_back(fmt::format("{}: condition mode does not suit parameter '{}'", where, condition.parameter));
193 }
194 }
195}

197void validate_layer_settings(const animation_layer_def& layer,
198 const std::string& layer_label,
199 bool is_base,
200 std::vector<std::string>& problems)
201{
202 if(is_base)
203 {
204 // The base layer has nothing below it to blend with or mask against.
205 if(!layer.mask.empty())
206 {
207 problems.push_back(fmt::format("{} is the base layer: its bone mask is ignored", layer_label));
208 }
209 if(layer.weight != 1.0f)
210 {
211 problems.push_back(fmt::format("{} is the base layer: it always plays at full weight", layer_label));
212 }
213 if(layer.blend_mode == animation_layer_blend_mode::additive)
214 {
215 problems.push_back(fmt::format("{} is the base layer: it cannot be additive", layer_label));
216 }
217 }
218 std::unordered_set<std::string> mask_bones;
219 for(const auto& entry : layer.mask)
220 {
221 if(!mask_bones.insert(entry.bone).second)
222 {
223 problems.push_back(fmt::format("{} masks bone '{}' twice; the last entry counts", layer_label, entry.bone));
224 }

228void validate_layer(const animation_controller& controller,
229 const animation_layer_def& layer,
230 bool is_base,
231 std::vector<std::string>& problems)
232{
233 const std::string layer_label = fmt::format("layer '{}'", layer.name);
234 if(layer.states.empty())
235 {
236 problems.push_back(fmt::format("{} has no states", layer_label));
237 }
238 validate_layer_settings(layer, layer_label, is_base, problems);
239 std::unordered_set<std::string> state_names;
240 for(const auto& state : layer.states)
241 {
242 if(state.name.empty())
243 {
244 problems.push_back(fmt::format("{} has a state without a name", layer_label));
245 }
246 else if(!state_names.insert(state.name).second)
247 {
248 problems.push_back(fmt::format("{} has two states named '{}'", layer_label, state.name));
249 }
250 validate_state(controller, state, fmt::format("{} state '{}'", layer_label, state.name), is_base, problems);
251 }
252
253 if(!layer.default_state.empty() && find_animation_state(layer, layer.default_state) < 0)
254 {
255 problems.push_back(fmt::format("{} default state '{}' does not exist", layer_label, layer.default_state));
256 }
257
258 for(const auto& transition : layer.transitions)
259 {
260 validate_transition(controller, layer, transition, layer_label, problems);
261 }

280auto validate_animation_controller(const animation_controller& controller) -> std::vector<std::string>
281{
282 std::vector<std::string> problems;
283 std::unordered_set<std::string> parameter_names;
284 for(const auto& parameter : controller.parameters)
285 {
286 if(parameter.name.empty())
287 {
288 problems.push_back("a parameter has no name");
289 }
290 else if(!parameter_names.insert(parameter.name).second)
291 {
292 problems.push_back(fmt::format("two parameters are named '{}'", parameter.name));
293 }
294 }
295 if(controller.layers.empty())
296 {
297 problems.push_back("the controller has no layers");
298 }
299 std::unordered_set<std::string> layer_names;
300 for(size_t index = 0; index < controller.layers.size(); ++index)
301 {
302 const auto& layer = controller.layers[index];
303 if(layer.name.empty())
304 {
305 // Running controllers keep each layer's state by name across edits.
306 problems.push_back("a layer has no name");
307 }
308 else if(!layer_names.insert(layer.name).second)
309 {
310 problems.push_back(fmt::format("two layers are named '{}'", layer.name));
311 }
312 validate_layer(controller, layer, index == 0, problems);
313 }
314 return problems;
315}

343{

346 return;
347 }
348 const std::string name = layer.states[state_index].name;

396auto compute_animation_mask_weights(const std::vector<animation_mask_entry>& mask,
397 const std::vector<std::string>& names,
398 const std::vector<int>& parents) -> std::vector<float>
399{
400 std::unordered_map<std::string, const animation_mask_entry*> entries;
401 for(const auto& entry : mask)
402 {
403 entries[entry.bone] = &entry;
404 }
405 std::vector<float> weights(names.size(), 0.0f);
406 for(size_t bone = 0; bone < names.size(); ++bone)
407 {
408 int current = int(bone);
409 // Bounded walk: a malformed parent list must not loop forever.
410 for(size_t depth = 0; current >= 0 && size_t(current) < names.size() && depth <= names.size(); ++depth)
411 {
412 const auto it = entries.find(names[size_t(current)]);
413 if(it != entries.end() && (size_t(current) == bone || it->second->include_children))
414 {
415 weights[bone] = math::clamp(it->second->weight, 0.0f, 1.0f);
416 break;
417 }
418 current = size_t(current) < parents.size() ? parents[size_t(current)] : -1;
419 }
420 }
421 return weights;
422}
