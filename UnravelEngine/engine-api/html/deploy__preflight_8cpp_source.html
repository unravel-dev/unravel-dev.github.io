25auto get_folder_problem(const fs::path& location, const fs::path& output, bool is_android) -> std::string
26{
27 fs::error_code ec;
28 if(location.empty())
29 {
30 return fmt::format("Choose a folder to deploy to, such as {} in the project.", DEFAULT_DEPLOY_LOCATION);
31 }
32 if(is_inside(get_project_root(), location))
33 {
34 return fmt::format("The deploy folder contains the project. Choose another one, such as {} in the project.",
35 DEFAULT_DEPLOY_LOCATION);
36 }
37 const auto managed = get_managed_project_folder(location, get_project_root());
38 if(!managed.empty())
39 {
40 return fmt::format("The deploy folder is in the project's {} folder, which the editor manages. Choose "
41 "another one, such as {} in the project.",
42 managed,
43 DEFAULT_DEPLOY_LOCATION);
44 }
45 if(fs::exists(location, ec) && !fs::is_directory(location, ec))
46 {
47 return fmt::format("The deploy location {} is a file, not a folder.", location.generic_string());
48 }
49 if(!fs::exists(location, ec) && !fs::is_directory(location.parent_path(), ec))
50 {
51 return fmt::format("The deploy folder's parent {} does not exist.", location.parent_path().generic_string());
52 }
53 if(!fs::exists(output, ec))
54 {
55 return {};
56 }
57 const auto folder = output.filename().generic_string();
58 if(is_android)
59 {
60 return is_android_package_folder(output)
61 ? std::string{}
62 : fmt::format("The {} folder in the deploy folder holds other files. A deploy replaces the package "
63 "in it. Empty it, or choose another deploy folder.",
64 folder);
65 }
66 if(!fs::is_empty(output, ec) && !is_previous_deploy(output))
67 {
68 return fmt::format("The {} folder in the deploy folder holds files of no earlier build. A deploy replaces "
69 "and removes files in it. Empty it, or choose another deploy folder.",
70 folder);
71 }
72 return {};
73}

131auto get_pending_compile_reason(rtti::context& ctx) -> std::string
132{
133 // The deploy reads the compiled cache as it is: until the compiler is idle that is a mix of old and
134 // new files. A script library build lands only when its domain reloads.
135 const size_t compile_jobs = ctx.get_cached<asset_watcher>().get_pending_compile_jobs_count(ctx);
136 if(compile_jobs > 0)
137 {
138 return fmt::format("Assets are still compiling ({} job(s) left). The deploy can start once they finish.", compile_jobs);
139 }
140 auto& scripts = ctx.get_cached<script_system>();
141 if(scripts.is_build_pending(ENGINE_PROTOCOL) || scripts.is_build_pending(CONTENT_PROTOCOL))
142 {
143 return "Scripts are still compiling. The deploy can start once they finish.";
144 }
145 return {};
146}

148auto get_problems(rtti::context& ctx, const deploy_settings& params) -> std::vector<std::string>
149{
150 std::vector<std::string> problems;
151 auto& pm = ctx.get_cached<project_manager>();
152 if(pm.get_profiles().find(params.profile) == nullptr)
153 {
154 problems.push_back("Choose the profile to deploy.");
155 return problems;
156 }
157 const bool is_android = params.deploy_target == deploy_platform::android;
158 const auto folder_problem = get_folder_problem(get_deploy_folder(params), get_output_location(params), is_android);
159 add_problem(problems, folder_problem);
160 if(folder_problem.empty() && !is_android && is_executable_in_use(get_deployed_executable(ctx, params)))
161 {
162 add_problem(problems, "The game in the deploy folder is running. Close it first.");
163 }
164 if(!pm.get_profile_settings(params.profile).standalone.startup_scene.is_valid())
165 {
166 add_problem(problems, "Choose the scene the game starts with.");
167 }
168 if(params.deploy_cooked_assets && ctx.get_cached<play_mode>().is_active())
169 {
170 add_problem(problems, "Stop play mode first: prefabs cannot be cooked while the game runs.");
171 }
172 // An Android game carries the runtime of the editor's Android player, not the editor's own.
173 if(is_android)
174 {
175 for(auto& problem : get_android_problems(ctx, params))
176 {
177 add_problem(problems, std::move(problem));
178 }
179 }
180 else
181 {
182 add_problem(problems, get_runtime_problem());
183 // An Android package carries every pack whole, so only desktop deploys patch.
184 add_problem(problems, get_patch_problem(params));
185 }
186 add_problem(problems, get_compile_problem(ctx));
187 return problems;
188}

195auto problems_cache::get(rtti::context& ctx, const deploy_settings& params) -> const std::vector<std::string>&
196{
197 const auto signature = fmt::format("{}|{}|{}|{}|{}|{}|{}|{}|{}|{}|{}|{}|{}|{}",
198 params.profile,
199 params.deploy_location.generic_string(),
200 params.deploy_patch_release,
201 params.deploy_packed_assets,
202 params.deploy_cooked_assets,
203 static_cast<int>(params.deploy_target),
204 params.deploy_and_run,
205 params.android_device,
206 params.android.application_id,
207 params.android.release,
208 params.android.keystore.generic_string(),
209 params.android.key_alias,
210 params.android.icon.id(),
211 ctx.get_cached<project_manager>().get_profiles().get_revision());
212 const auto now = std::chrono::steady_clock::now();
213 if(signature != signature_ || now - checked_at_ >= PROBLEMS_REFRESH_INTERVAL)
214 {
215 problems_ = get_problems(ctx, params);
216 signature_ = signature;
217 checked_at_ = now;
218 }
219 return problems_;
220}
