183auto get_curve_range(const std::vector<animation_curve>& curves) -> ImVec2
184{
185 ImVec2 range(0.0f, 1.0f);
186 for(const auto& curve : curves)
187 {
188 for(int i = 0; i <= CURVE_SAMPLES; ++i)
189 {
190 const float value = curve.curve.sample(float(i) / float(CURVE_SAMPLES));
191 range.x = std::min(range.x, value);
192 range.y = std::max(range.y, value);
193 }
194 }
195 return range;

222void draw_curves_overlay(ImDrawList* draw_list, const ImRect& lane, const std::vector<animation_curve>& curves)
223{
224 if(curves.empty())
225 {
226 return;
227 }
228 const ImVec2 range = get_curve_range(curves);
229 const float span = std::max(range.y - range.x, MIN_CURVE_SPAN);
230 const float drawn_height = lane.GetHeight() - CURVE_LANE_PADDING * 2.0f;
231 std::vector<ImVec2> points(CURVE_SAMPLES + 1);
232 for(size_t c = 0; c < curves.size(); ++c)
233 {
234 for(int i = 0; i <= CURVE_SAMPLES; ++i)
235 {
236 const float progress = float(i) / float(CURVE_SAMPLES);
237 const float value = (curves[c].curve.sample(progress) - range.x) / span;
238 points[size_t(i)] =
239 ImVec2(lane.Min.x + progress * lane.GetWidth(), lane.Max.y - CURVE_LANE_PADDING - value * drawn_height);
240 }
241 const ImU32 color = with_alpha(CURVE_COLORS[c % CURVE_COLORS.size()], CURVE_ALPHA);
242 draw_list->AddPolyline(points.data(), int(points.size()), color, ImDrawFlags_None, CURVE_THICKNESS);
243 }

260auto find_marker(const std::vector<animation_event>& events, const ImRect& lane, float mouse_x) -> int
261{
262 int best = -1;
263 float best_distance = MARKER_HALF_SIZE + MARKER_PICK_SLACK;
264 for(size_t i = 0; i < events.size(); ++i)
265 {
266 const float x = lane.Min.x + events[i].position * lane.GetWidth();
267 const float distance = std::fabs(mouse_x - x);
268 if(distance <= best_distance)
269 {
270 best_distance = distance;
271 best = int(i);
272 }
273 }
274 return best;

327void animation_clip_timeline::advance_playback(const animation_importer_meta& importer, float duration)
328{
329 if(!playing_)
330 {
331 return;
332 }
333 const float previous = playhead_;
334 float next = playhead_ + ImGui::GetIO().DeltaTime / duration;
335 const bool wrapped = next >= 1.0f;
336 next = wrapped ? next - std::floor(next) : next;
337 flash_until_.resize(importer.events.size(), 0.0);
338 const double now = ImGui::GetTime();
339 for(size_t i = 0; i < importer.events.size(); ++i)
340 {
341 const float position = importer.events[i].position;
342 const bool crossed = wrapped ? (position > previous || position <= next) : (position > previous && position <= next);
343 if(crossed)
344 {
345 flash_until_[i] = now + FLASH_SECONDS;
346 }
347 }
348 set_playhead(next);
349}

429{

431 auto& events = importer.events;

433 const float width = std::max(ImGui::GetContentRegionAvail().x, MIN_TRACK_WIDTH);
434 const ImRect rect(origin, ImVec2(origin.x + width, origin.y + TRACK_HEIGHT));
435 const ImRect lane(ImVec2(rect.Min.x, rect.Min.y + RULER_HEIGHT), rect.Max);

437 const bool hovered = ImGui::IsItemHovered();
438 const bool active = ImGui::IsItemActive();

441 const int hovered_marker = hovered && lane.Contains(mouse) ? find_marker(events, lane, mouse.x) : -1;
442
443 // An undo during a drag puts other settings in place: the drag ends there, without an edit.
444 const bool is_drag_target_gone = dragging_event_ >= int(events.size()) ||
445 (dragging_event_ >= 0 && events[size_t(dragging_event_)].position != drag_position_);
446 if(is_drag_target_gone)

449 }
450 // Interaction: markers select and drag, empty space scrubs, a double click adds an event.
451 if(ImGui::IsItemClicked(ImGuiMouseButton_Left))
452 {
453 if(edit_events && hovered_marker >= 0)
454 {

456 dragging_event_ = hovered_marker;

464 }
465 }

494 }

512 flash_until_.resize(events.size(), 0.0);

514 // Labels of markers near the end stop at the track's edge.
515 const ImVec4 label_clip(rect.Min.x, rect.Min.y, rect.Max.x, rect.Max.y);
516 for(size_t i = 0; i < events.size(); ++i)
517 {
518 const float x = lane.Min.x + events[i].position * lane.GetWidth();
519 const bool selected = edit_events && int(i) == selected_event_;

521 if(!edit_events)
522 {
523 color = with_alpha(color, MUTED_MARKER_ALPHA);
524 }
525 draw_list->AddLine(ImVec2(x, marker_y), ImVec2(x, lane.Max.y), with_alpha(color, MARKER_LINE_ALPHA));

541 draw_list->AddTriangleFilled(ImVec2(playhead_x - PLAYHEAD_HANDLE_HALF_WIDTH, rect.Min.y),
542 ImVec2(playhead_x + PLAYHEAD_HANDLE_HALF_WIDTH, rect.Min.y),

544 COL_PLAYHEAD_LINE);
545 if(hovered_marker >= 0 && dragging_event_ < 0)
546 {
547 const auto& event = events[size_t(hovered_marker)];

566 }

571 ImGui::EndPopup();
572 }
573 // Delete belongs to the focused window: with the content browser focused it deletes assets there.
574 const bool has_focus = ImGui::IsWindowFocused(ImGuiFocusedFlags_RootAndChildWindows);
575 if(edit_events && has_focus && selected_event_ >= 0 && selected_event_ < int(events.size()) && hovered &&
576 !ImGui::GetIO().WantTextInput && ImGui::IsKeyPressed(shortcuts::delete_item, false))
577 {
578 erase_event(importer, selected_event_, result);
579 }
580 return result;
