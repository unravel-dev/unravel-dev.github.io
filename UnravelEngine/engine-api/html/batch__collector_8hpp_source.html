31template<typename Key>
32void batch_group_t<Key>::calculate_camera_distance(const math::vec3& camera_pos)
33{
34 if(instances.empty())
35 {
36 camera_distance = 0.0f;
37 return;
38 }
39
40 math::vec3 average_position(0.0f);
41 for(const auto& instance : instances)
42 {
43 if(!instance.world_transform_ptr)
44 {
45 continue;
46 }
47 const auto& transform = *instance.world_transform_ptr;
48 average_position += math::vec3(transform[3][0], transform[3][1], transform[3][2]);
49 }
50
51 average_position /= static_cast<float>(instances.size());
52 camera_distance = math::length(average_position - camera_pos);
53}

203{
204 if constexpr(std::is_same_v<Key, batch_key>)

206 std::sort(prepared_batches_.begin(),
207 prepared_batches_.end(),
208 [&context](const batch_group_t<Key>* a, const batch_group_t<Key>* b) -> bool
209 {
210 if(a->key.material_ptr.get() != b->key.material_ptr.get())
211 {
212 return a->key.material_ptr.get() < b->key.material_ptr.get();
213 }
214 if(a->key.mesh_ptr.get() != b->key.mesh_ptr.get())
215 {
216 return a->key.mesh_ptr.get() < b->key.mesh_ptr.get();

218 if(a->key.lod_index != b->key.lod_index)

226 if(context.enable_distance_sorting)

231 });

291template<typename Key>
292auto batch_collector_t<Key>::get_or_create_batch_group(const Key& key) -> batch_group_t<Key>&
293{
294 if((group_count_ + 1) * BUCKETS_PER_GROUP > buckets_.size())
295 {
296 grow_buckets();
297 }
298 const size_t key_hash = batch_collector_detail::mix_hash(key.hash());
299 const size_t mask = buckets_.size() - 1;
300 size_t bucket = key_hash & mask;
301 while(buckets_[bucket] != EMPTY_BUCKET)
302 {
303 auto& group = groups_[buckets_[bucket]];
304 if(group.key_hash == key_hash && group.key == key)
305 {
306 return group;
307 }
308 bucket = (bucket + 1) & mask;
309 }
310 // A new group: the slots may move now, and the prepared list would miss it anyway.
311 prepared_batches_.clear();
312 if(group_count_ == groups_.size())
313 {
314 groups_.emplace_back();
315 }
316 auto& group = groups_[group_count_];
317 group.key = key;
318 group.key_hash = key_hash;
319 buckets_[bucket] = static_cast<uint32_t>(group_count_);
320 ++group_count_;
321 return group;
322}

324template<typename Key>
325void batch_collector_t<Key>::grow_buckets()
326{
327 const size_t bucket_count = std::max(MIN_BUCKET_COUNT, buckets_.size() * 2);
328 buckets_.assign(bucket_count, EMPTY_BUCKET);
329 const size_t mask = bucket_count - 1;
330 for(size_t index = 0; index < group_count_; ++index)
331 {
332 size_t bucket = groups_[index].key_hash & mask;
333 while(buckets_[bucket] != EMPTY_BUCKET)
334 {
335 bucket = (bucket + 1) & mask;
336 }
337 buckets_[bucket] = static_cast<uint32_t>(index);
338 }
339}
