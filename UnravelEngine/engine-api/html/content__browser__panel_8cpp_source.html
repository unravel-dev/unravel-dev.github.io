135 if(payload != nullptr)

418void apply_entry_action(const content_browser_item& item, entry_action action, const std::string& typed_name)
419{
420 if(action != entry_action::none)
421 {
422 pending_rename.clear();
423 }
424 const auto run = [](const content_browser_item::on_action_t& callback)
425 {
426 if(callback)
427 {
428 callback();
429 }
430 };
431 switch(action)
432 {
433 case entry_action::clicked:
434 run(item.on_click);
435 break;
436 case entry_action::double_clicked:
437 run(item.on_double_click);
438 break;
439 case entry_action::renamed:
440 if(item.on_rename && !typed_name.empty() && typed_name != item.entry.stem)
441 {
442 item.on_rename(typed_name);
443 }
444 break;
445 case entry_action::deleted:
446 run(item.on_delete);
447 break;
448 case entry_action::duplicate:
449 asset_actions::duplicate_entry(item.entry.entry.path());
450 break;
451 case entry_action::canceled:
452 run(item.on_cancel);
453 break;
454 default:
455 break;
456 }
