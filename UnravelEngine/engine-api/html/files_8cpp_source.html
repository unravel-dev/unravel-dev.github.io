152auto collect_pack_inputs(const fs::path& compiled,
153 const cooked_folders& cooked,
154 const fs::path& index,
155 const compiled_file_filter& is_shipped) -> std::vector<fs::pack_input>
156{
157 std::vector<fs::pack_input> inputs;
158 fs::error_code ec;
159 const auto compiled_folder = ex::get_compiled_directory_no_slash();
160 for(fs::recursive_directory_iterator it(compiled, ec), end; !ec && it != end; it.increment(ec))
161 {
162 const auto& file = it->path();
163 const auto relative = file.lexically_relative(compiled);
164 if(!it->is_regular_file(ec) || is_loose_only_file(file) || !is_shipped(file, relative))
165 {
166 continue;
167 }
168 inputs.push_back({compiled_folder + "/" + relative.generic_string(), get_shipped_variant(cooked, relative, file)});
169 }
170 if(ec)
171 {
172 throw std::runtime_error(fmt::format("Listing {} failed: {}", compiled.generic_string(), ec.message()));
173 }
174 inputs.push_back({ex::get_asset_index_file_name(), index});
175 std::sort(inputs.begin(),
176 inputs.end(),
177 [](const fs::pack_input& lhs, const fs::pack_input& rhs)
178 {
179 return lhs.name < rhs.name;
180 });
181 return inputs;
182}

253auto is_binary_file(const fs::path& file) -> bool
254{
255#ifdef _WIN32
256 const auto extension = string_utils::to_lower(file.extension().generic_string());
257 return extension == ".dll" || extension == ".exe";
258#else
259 // Shared libraries carry ".so", versioned or not; executables carry no extension and the exec bit.
260 const auto name = file.filename().generic_string();
261 if(name.ends_with(".so") || name.find(".so.") != std::string::npos)
262 {
263 return true;
264 }
265 fs::error_code ec;
266 const auto permissions = fs::status(file, ec).permissions();
267 return !file.has_extension() && !ec && (permissions & fs::perms::owner_exec) != fs::perms::none;
268#endif
269}
