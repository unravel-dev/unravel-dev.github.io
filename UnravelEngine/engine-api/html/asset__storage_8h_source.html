472 void rename_asset(const std::string& key, const std::string& new_key)
473 {
474 std::lock_guard<std::mutex> lock(asset_mutex_);
475 // Nothing at key: the entry has moved already (the meta syncer and the compiled-asset watcher
476 // both report a rename), and whatever is at new_key is it.
477 if(key == new_key || find_entry_by_location_unlocked(key) == asset_meta_.end())
478 {
479 return;
480 }
481 // An entry already at new_key was registered by a load of the new path that ran before this
482 // rename arrived. The renamed entry is the asset (its .meta carries that id), so the early one
483 // goes rather than alias the same location.
484 std::erase_if(asset_meta_,
485 [&new_key](const auto& kvp)
486 {
487 return kvp.second.location == new_key;
488 });
489 for(auto& kvp : asset_meta_)
490 {
491 auto& uid = kvp.first;
492 auto& metainfo = kvp.second;
493 if(metainfo.location == key)
494 {
495 APPLOG_TRACE("{}::{} - {} -> {}", __func__, hpp::to_string(uid), key, new_key);
496
497 metainfo.location = new_key;
498 }
499 }
500 }
501
506 void remove_asset(const std::string& key)
507 {
508 std::lock_guard<std::mutex> lock(asset_mutex_);
509 for(auto& kvp : asset_meta_)
510 {
511 auto& uid = kvp.first;
512 auto& metainfo = kvp.second;
513 if(metainfo.location == key)
514 {

755 void evict_unused(tpp::thread_pool& pool, const std::string& group, std::chrono::steady_clock::duration max_idle) final
756 {
757 auto now = std::chrono::steady_clock::now();
758 std::lock_guard<std::recursive_mutex> lock(container_mutex);
759 for(auto& [key, handle] : container)
760 {
761 const auto& id = handle.id();
762 hpp::string_view id_view(id);
763 if(!id_view.starts_with(group))
764 {
765 continue;
766 }
767 if(!handle.is_ready())
768 {
769 continue;
770 }
771 bool is_idle = (now - handle.last_access()) > max_idle;
772 if(!is_idle)
773 {
774 continue;
775 }
776 pool.stop(handle.task_id());
777 handle.demote_to_deferred();
778 load_from_file(pool, handle, key, load_mode::deferred);
779 }
780 }
