48void format_count(uint64_t value, char* buffer, size_t buffer_size)
49{
50 std::array<char, 32> digits{};
51 const int written = std::snprintf(digits.data(), digits.size(), "%llu", static_cast<unsigned long long>(value));
52 if(written <= 0)
53 {
54 std::snprintf(buffer, buffer_size, "0");
55 return;
56 }
57 size_t out = 0;
58 for(int read = 0; read < written && out + 1 < buffer_size; ++read)
59 {
60 if(read > 0 && ((written - read) % 3) == 0)
61 {
62 buffer[out++] = ',';
63 }
64 if(out + 1 < buffer_size)
65 {
66 buffer[out++] = digits[static_cast<size_t>(read)];
67 }
68 }
69 buffer[out] = '\0';
70}
