73auto internal_m2n_screen_get_safe_area() -> script_rect
74{
75 auto& ctx = engine::context();
76 const auto& size = ctx.get_cached<rendering_system>().get_game_view_size();
77 script_rect area{0.0f, 0.0f, float(size.width), float(size.height)};
78 auto* window = ctx.has<renderer>() ? ctx.get_cached<renderer>().get_main_window() : nullptr;
79 if(window == nullptr)
80 {
81 return area;
82 }
83 // In the player the game view is the window. The editor's Game panel is a part of its window,
84 // and nothing covers it.
85 const auto window_size = window->get_window().get_size();
86 if(window_size.w != size.width || window_size.h != size.height)
87 {
88 return area;
89 }
90 const auto insets = window->get_window().get_safe_area_insets();
91 area.x = float(insets.left);
92 area.y = float(insets.top);
93 area.width = float(int(size.width) - insets.left - insets.right);
94 area.height = float(int(size.height) - insets.top - insets.bottom);
95 return area;
96}
