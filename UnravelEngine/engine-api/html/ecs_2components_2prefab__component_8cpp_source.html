13auto is_parent_path(const std::string& parent, const std::string& child) -> bool
14{
15 if(child.length() <= parent.length())
16 {
17 return false;
18 }
19 if(child.compare(0, parent.length(), parent) != 0)
20 {
21 return false;
22 }
23 const char next_char = child[parent.length()];
24 return next_char == '/' || next_char == '[';
25}

190auto prefab_statements::has_override_touching(const std::vector<hpp::uuid>& instance_path,
191 const hpp::uuid& entity_uuid,
192 hpp::string_view component_path) const -> bool
193{
194 if(overrides.empty())
195 {
196 return false;
197 }
198
199 // Below: the first entry at or after the path, for the same entity, that extends it.
200 const auto it = overrides.lower_bound(
201 prefab_property_override_data{instance_path, entity_uuid, std::string(component_path), std::string{}});
202 if(it != overrides.end() && it->instance_path == instance_path && it->entity_uuid == entity_uuid)
203 {
204 const hpp::string_view candidate(it->component_path);
205 if(candidate.starts_with(component_path) &&
206 (candidate.size() == component_path.size() || candidate[component_path.size()] == '/' ||
207 candidate[component_path.size()] == '['))
208 {
209 return true;
210 }
211 }
212
213 // Above: every ancestor of the path.
214 auto ancestor = component_path;
215 while(true)
216 {
217 const auto separator = ancestor.rfind('/');
218 if(separator == hpp::string_view::npos)
219 {
220 return false;
221 }
222 ancestor = ancestor.substr(0, separator);
223 if(has_override(instance_path, entity_uuid, std::string(ancestor)))
224 {
225 return true;
226 }
227 }
228}

383auto prefab_component::has_serialization_override(const std::string& serialization_path) const -> bool
384{
385 if(local.overrides.empty())
386 {
387 return false;
388 }
389 // Expected format: "entities/<entity_uuid>/components/<component_type>/<property_path>"
390 const hpp::string_view path(serialization_path);
391 const auto first = path.find('/');
392 if(first == hpp::string_view::npos)
393 {
394 return false;
395 }
396 const auto second = path.find('/', first + 1);
397 if(second == hpp::string_view::npos)
398 {
399 return false;
400 }
401 const auto third = path.find('/', second + 1);
402 if(third == hpp::string_view::npos || third + 1 >= path.size())
403 {
404 return false;
405 }
406 if(!path.substr(0, first).starts_with("entities"))
407 {
408 return false;
409 }
410 if(!path.substr(second + 1, third - second - 1).starts_with("components"))
411 {
412 return false;
413 }
414 const auto uuid_opt = hpp::uuid::from_string(path.substr(first + 1, second - first - 1));
415 if(!uuid_opt.has_value())
416 {
417 return false;
418 }
419 return has_override(uuid_opt.value(), std::string(path.substr(third + 1)));
420}
