103void layout_manager::load_presets_from_disk()
104{
105 fs::error_code err;
106 if(!fs::exists(layouts_directory_, err))
107 {
108 return;
109 }
110
111 for(const auto& entry : fs::directory_iterator(layouts_directory_, err))
112 {
113 if(!entry.is_regular_file())
114 {
115 continue;
116 }
117
118 const auto& path = entry.path();
119 if(path.extension() != ".ini")
120 {
121 continue;
122 }
123
124 std::ifstream file(path.string(), std::ios::binary);
125 if(!file.is_open())
126 {
127 continue;
128 }
129
130 std::ostringstream ss;
131 ss << file.rdbuf();
132 std::string data = ss.str();
133
134 if(!data.empty())
135 {
136 auto name = path.stem().string();
137 presets_[name] = std::move(data);
138 }
139 }
