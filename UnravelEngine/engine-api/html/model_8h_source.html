102 auto add_transform(const SubmeshIndices& submesh_indices, const math::mat4& transform, bool active) -> uint32_t
103 {
104 // Add the transform to the pool
105 uint32_t trans_index = add_transform(transform);
106
107 // Map all submesh indices to this transform
108 for(uint32_t submesh_index : submesh_indices)

135 void map_submesh(uint32_t submesh_index, uint32_t trans_index, bool active, bool casts_shadow)
