34auto parse_version_name(const std::string& name) -> std::vector<int>
35{
36 std::vector<int> parts;
37 std::string current;
38 for(const char c : name)
39 {
40 if(c >= '0' && c <= '9')
41 {
42 current += c;
43 continue;
44 }
45 if(c != '.')
46 {
47 break;
48 }
49 parts.push_back(current.empty() ? 0 : std::atoi(current.c_str()));
50 current.clear();
51 }
52 if(!current.empty())
53 {
54 parts.push_back(std::atoi(current.c_str()));
55 }
56 return parts;
57}

61auto pick_highest_version_folder(const fs::path& base, int preferred_major) -> fs::path
62{
63 fs::path best;
64 std::vector<int> best_version;
65 bool best_matches_major = false;
66 fs::error_code ec;
67 for(const auto& entry : fs::directory_iterator(base, ec))
68 {
69 const auto version = entry.is_directory(ec) ? parse_version_name(entry.path().filename().string()) : std::vector<int>{};
70 if(version.empty())
71 {
72 continue;
73 }
74 const bool matches_major = preferred_major >= 0 && version.front() == preferred_major;
75 const bool is_better = best.empty() || (matches_major != best_matches_major
76 ? matches_major
77 : std::lexicographical_compare(best_version.begin(),
78 best_version.end(),
79 version.begin(),
80 version.end()));
81 if(is_better)
82 {
83 best = entry.path();
84 best_version = version;
85 best_matches_major = matches_major;
86 }
87 }
88 return best;
89}

132 plan.config_destination = mono_root / "etc" / MONO_FOLDER;

136void deploy_runtime(const runtime_plan& plan)
137{
138 for(const auto& name : plan.library_names)
139 {
140 fs::error_code ec;
141 if(fs::exists(plan.host_source / name, ec))
142 {
143 copy_file_or_throw(plan.host_source / name, plan.host_destination / name);
144 }
145 }
146 mirror_or_throw(plan.managed_source, plan.managed_destination);
147 mirror_or_throw(plan.framework_source, plan.framework_destination);
148 // An older class library version beside the current one goes, the cache next to it stays.
149 const auto libraries_root = plan.managed_destination.parent_path();
150 fs::error_code ec;
151 for(const auto& entry : fs::directory_iterator(libraries_root, ec))
152 {
153 const auto name = entry.path().filename();
154 if(name != plan.managed_destination.filename() && name != plan.framework_destination.filename())
155 {
156 fs::error_code remove_ec;
157 fs::remove_all(entry.path(), remove_ec);
158 }
159 }
160 mirror_or_throw(plan.config_source, plan.config_destination);
161}

170auto make_runtime_plan(rtti::context&, const deploy_settings& params) -> runtime_plan
171{
172 const auto engine_root = get_engine_root(params);
173 runtime_plan plan;
174 plan.step_name = ".NET";
175 plan.with_debug_symbols = params.deploy_debug_symbols;
176 // The managed bridge payload (Clrpp.Managed.dll, runtimeconfig, NuGet deps) is one folder; the game
177 // passes its deployed location to the runtime at init.
178 const std::string managed_folder = dotnet::managed_runtime_dir();
179 plan.managed_source = fs::resolve_protocol("binary:/" + managed_folder);
180 plan.managed_destination = engine_root / managed_folder;
181 // A pruned dotnet root (hostfxr and the shared framework) lets the game run without an installed
182 // .NET. It carries the runtime the editor runs on, which get_runtime_problem requires to be the
183 // preferred one, and the hostfxr of that runtime's installation.
184 const fs::path dotnet_root = dotnet::get_core_assembly_path();
185 const int runtime_major = get_version_major(dotnet::get_runtime_version());
186 plan.host_source = pick_highest_version_folder(dotnet_root / HOST_FXR_FOLDER, runtime_major);
187 plan.framework_source = dotnet::get_runtime_framework_dir();
188 plan.host_destination = engine_root / DOTNET_FOLDER / HOST_FXR_FOLDER / plan.host_source.filename();
189 plan.framework_destination = engine_root / DOTNET_FOLDER / SHARED_FRAMEWORK_FOLDER / plan.framework_source.filename();
190 if(plan.framework_source.empty())
191 {
192 APPLOG_WARNING("Deploy: the folder of the editor's .NET runtime is unknown, so the deployed game will need an "
193 "installed .NET runtime.");
194 plan.host_source.clear();
195 }
196 else if(plan.host_source.empty())
197 {
198 APPLOG_INFO("Deploy: no hostfxr under {}; the game starts the bundled framework through the coreclr host",
199 dotnet_root.generic_string());
200 }
201 return plan;
202}

204void deploy_runtime(const runtime_plan& plan)
205{
206 const fs::mirror_filter include = [&plan](const fs::path& file)

212 if(plan.framework_source.empty())
213 {
214 return;

216 // The runtime is the largest part of a build and rarely changes: only an updated version is copied,
217 // and the one it replaces removed.
218 if(!plan.host_source.empty())
219 {
220 mirror_or_throw(plan.host_source, plan.host_destination);
221 remove_siblings(plan.host_destination);
222 }
223 mirror_or_throw(plan.framework_source, plan.framework_destination);
224 remove_siblings(plan.framework_destination);
225}
