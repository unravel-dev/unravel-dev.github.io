64auto make_range(float lo, float hi, bool include_zero) -> axis_range
65{
66 if(include_zero)
67 {
68 lo = std::min(lo, 0.0f);
69 hi = std::max(hi, 0.0f);
70 }
71 if(hi - lo < 1e-3f)
72 {
73 lo -= 1.0f;
74 hi += 1.0f;
75 }
76 const float margin = (hi - lo) * RANGE_MARGIN;
77 return {lo - margin, hi + margin};

87auto make_point_labels(const animation_state_def& state) -> std::vector<std::string>
88{
89 std::vector<std::string> labels;
90 for(const auto& point : state.points)
91 {
92 labels.push_back(point.clip ? fs::path(point.clip.id()).stem().string() : std::string("(none)"));
93 }
94 if(labels.size() < 2)
95 {
96 return labels;
97 }
98 size_t prefix = labels.front().size();
99 for(const auto& label : labels)
100 {
101 prefix = std::min(prefix, label.size());
102 for(size_t i = 0; i < prefix; ++i)
103 {
104 if(label[i] != labels.front()[i])
105 {
106 prefix = i;
107 break;
108 }
109 }
110 }
111 while(prefix > 0 && !is_name_separator(labels.front()[prefix - 1]))
112 {
113 --prefix;
114 }
115 for(auto& label : labels)
116 {
117 if(prefix < label.size())
118 {
119 label = label.substr(prefix);
120 }
121 }
122 return labels;
123}
