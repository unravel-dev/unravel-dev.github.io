32auto hex_to_xxh128(const std::string& hex) -> XXH128_hash_t
33{
34 XXH128_hash_t hash{};

39 auto parse_nibble = [](char c) -> uint64_t

41 if(c >= '0' && c <= '9')
42 {
43 return static_cast<uint64_t>(c - '0');
44 }
45 if(c >= 'a' && c <= 'f')
