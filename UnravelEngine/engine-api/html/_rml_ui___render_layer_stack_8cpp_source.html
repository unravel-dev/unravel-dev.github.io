29auto RmlUi_RenderLayerStack::push_layer() -> Rml::LayerHandle
30{
31 RMLUI_ASSERT(layers_size_ <= static_cast<int>(fb_layers_.size()));
32
33 if(layers_size_ == static_cast<int>(fb_layers_.size()))
34 {
35 gfx::texture::ptr shared_depth_stencil = nullptr;
36 if(!fb_layers_.empty())
37 {
38 shared_depth_stencil = fb_layers_.front().get_depth_texture();
39 }
40
41 fb_layers_.push_back(
42 create_layer_framebuffer(width_, height_, RMLUI_NUM_MSAA_SAMPLES, true, shared_depth_stencil));
43 }
44
45 layers_size_++;
46 auto layer_handle = get_top_layer_handle();
47 auto& layer = get_layer(layer_handle);
48 layer.layer_id = next_layer_id_++;
49 layer.needs_rebind = true;
50
51 gfx::render_pass::push_scope(fmt::format("Layer {}", layer.layer_id).c_str());
52 gfx::render_pass layer_pass("Layer Clear Pass");
53 layer_pass.bind(layer.framebuffer.get());
54 auto view = Rml::Matrix4f::Identity();
55 auto proj = Rml::Matrix4f::Identity();
56 layer_pass.set_view_proj(view.Transpose().data(), proj.Transpose().data());
57 layer_pass.clear(BGFX_CLEAR_COLOR, 0x00000000, 1.0f, 0);
58
59 return layer_handle;
60}

165auto RmlUi_RenderLayerStack::create_layer_framebuffer(int width,
166 int height,
167 int samples,
168 bool with_depth_stencil,
169 gfx::texture::ptr depth_texture)
170 -> RmlUi_LayerFramebuffer
171{
172 RmlUi_LayerFramebuffer fb{};
173 fb.samples = samples;
174
175 uint64_t texture_flags = BGFX_TEXTURE_RT | BGFX_TEXTURE_BLIT_DST;
176 if(samples > 1)
177 {
178 texture_flags |= BGFX_TEXTURE_RT_MSAA_X2;
179 if(samples >= 4)
180 texture_flags |= BGFX_TEXTURE_RT_MSAA_X4;
181 if(samples >= 8)
182 texture_flags |= BGFX_TEXTURE_RT_MSAA_X8;
183 if(samples >= 16)
184 texture_flags |= BGFX_TEXTURE_RT_MSAA_X16;
185 }
186
187 uint64_t depth_texture_flags = texture_flags;
188 if(samples > 1)
189 {
190 depth_texture_flags |= BGFX_TEXTURE_RT_WRITE_ONLY;
191 }
192
193 auto color_texture = std::make_shared<gfx::texture>(static_cast<uint16_t>(width),
194 static_cast<uint16_t>(height),
195 false,
196 1,
197 gfx::texture_format::RGBA8,
198 texture_flags);
199
200 if(!color_texture->is_valid())
201 {
202 APPLOG_ERROR("Failed to create color texture");
203 return fb;
204 }
205
206 std::vector<gfx::texture::ptr> textures;
207 textures.push_back(color_texture);
208
209 if(with_depth_stencil)
210 {
211 if(!depth_texture)
212 {
213 depth_texture = std::make_shared<gfx::texture>(static_cast<uint16_t>(width),
214 static_cast<uint16_t>(height),
215 false,
216 1,
217 gfx::texture_format::D24S8,
218 depth_texture_flags);
219
220 if(!depth_texture->is_valid())
221 {
222 APPLOG_ERROR("Failed to create depth/stencil texture");
223 return fb;
224 }
225 }
226 textures.push_back(depth_texture);
227 }
228
229 fb.framebuffer = std::make_shared<gfx::frame_buffer>(textures);
230
231 if(!fb.framebuffer->is_valid())
232 {
233 APPLOG_ERROR("Failed to create framebuffer");
234 return fb;
235 }
236
237 return fb;
238}
