335auto inspector_animation_condition_def::inspect(rtti::context& ctx,

347 const auto& controller = *scope->controller;

350 const auto type = parameter_index >= 0 ? controller.parameters[size_t(parameter_index)].type
351 : animation_parameter_type::float_value;
352 const auto modes = get_condition_modes(type);
353 const bool needs_threshold = is_numeric(type);
354 // A trigger has one comparison only: the row shows just its parameter.
355 const bool needs_mode = modes.size() > 1;
356 const float spacing = ImGui::GetStyle().ItemSpacing.x;
357 const float mode_width = needs_mode ? get_mode_combo_width(type) : 0.0f;
358 const float gaps = spacing * float(int(needs_mode) + int(needs_threshold));
359 const float rest = ImGui::GetContentRegionAvail().x - mode_width - gaps;
360 ImGui::SetNextItemWidth(needs_threshold ? rest * CONDITION_PARAMETER_SHARE : rest);
361 if(draw_parameter_picker("##parameter", controller, condition.parameter, false, false))
362 {

365 if(std::find(picked_modes.begin(), picked_modes.end(), condition.mode) == picked_modes.end())
366 {
367 condition.mode = picked_modes.front();
368 }
369 result.changed = true;
370 result.edit_finished = true;
371 }
372 if(needs_mode)
373 {
374 ImGui::SameLine();
