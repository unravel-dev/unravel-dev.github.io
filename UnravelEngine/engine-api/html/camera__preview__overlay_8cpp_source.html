53auto calc_aspect(const preview_desc& desc) -> float
54{
55 if(desc.resolution.x <= 0.0f || desc.resolution.y <= 0.0f)

57 return DEFAULT_ASPECT;
58 }

62auto calc_card_layout(const preview_desc& desc, const ImRect& view_rect, bool is_collapsed) -> card_layout
63{
64 const float margin = panel_toolbar::get_bar_margin();
65 const float padding = panel_toolbar::get_overlay_padding();
66 const float width = ImClamp(view_rect.GetWidth() * CARD_WIDTH_SHARE, to_pixels(CARD_MIN_WIDTH), to_pixels(CARD_MAX_WIDTH));
67 card_layout layout{};
68 if(!is_collapsed)
69 {
70 const float image_width = width - 2.0f * padding;
71 layout.image_height = ImFloor(ImMin(image_width / calc_aspect(desc), view_rect.GetHeight() * IMAGE_HEIGHT_SHARE));
72 }
73 const float image_extent = is_collapsed ? 0.0f : to_pixels(CONTENT_GAP) + layout.image_height;
74 const float height = 2.0f * padding + to_pixels(HEADER_HEIGHT) + image_extent;
75 layout.is_visible = width + 2.0f * margin <= view_rect.GetWidth() && height + 2.0f * margin <= view_rect.GetHeight();
76 const ImVec2 card_max(view_rect.Max.x - margin, view_rect.Max.y - margin);
77 layout.card = ImRect(card_max - ImVec2(width, height), card_max);
78 return layout;
79}

116void draw_badge(ImDrawList* draw_list, const preview_desc& desc, const ImRect& image_rect)
117{
118 std::string text;
119 if(desc.resolution.x > 0.0f && desc.resolution.y > 0.0f)
120 {
121 text = fmt::format("{:.0f}x{:.0f}", desc.resolution.x, desc.resolution.y);
122 }
123 if(desc.projection != nullptr)
124 {
125 text.append(text.empty() ? "" : " ").append(desc.projection);
126 }
127 if(text.empty())
128 {
129 return;
130 }
131 const ImVec2 padding(to_pixels(BADGE_PADDING_X), to_pixels(BADGE_PADDING_Y));
132 const ImVec2 text_size = ImGui::CalcTextSize(text.c_str());
133 const float margin = to_pixels(BADGE_MARGIN);
134 const ImVec2 badge_min(image_rect.Min.x + margin, image_rect.Max.y - margin - text_size.y - 2.0f * padding.y);
135 const ImVec2 badge_max = badge_min + text_size + 2.0f * padding;
136 // A badge wider than the image would cover what it describes.
137 if(badge_max.x > image_rect.Max.x - margin || badge_min.y < image_rect.Min.y + margin)
138 {
139 return;
140 }
141 draw_list->AddRectFilled(badge_min, badge_max, BADGE_BG_COLOR, to_pixels(BADGE_ROUNDING));
142 draw_list->AddText(badge_min + padding, ImGui::GetColorU32(ImGuiCol_Text, BADGE_TEXT_ALPHA), text.c_str());
143}

153 const ImVec2 image_min(ImFloor(area_min.x + (area_size.x - image_size.x) * 0.5f), area_min.y);
154 const ImRect image_rect(image_min, image_min + image_size);

156 ImDrawList* draw_list = ImGui::GetWindowDrawList();

160 const char* placeholder = "No camera output";
161 const ImVec2 text_size = ImGui::CalcTextSize(placeholder);
162 draw_list->AddText(ImFloor(image_rect.GetCenter() - text_size * 0.5f),
163 ImGui::GetColorU32(ImGuiCol_Text, PLACEHOLDER_TEXT_ALPHA),

182auto draw(const char* id, const preview_desc& desc, const ImRect& view_rect, state& preview_state) -> request
183{
184 request result{};
185 ImGuiWindow* window = ImGui::GetCurrentWindow();
186 if(window == nullptr || window->SkipItems)
187 {
188 return result;
189 }
190 const card_layout layout = calc_card_layout(desc, view_rect, preview_state.is_collapsed);
191 if(!layout.is_visible)
192 {
193 return result;
194 }
195 if(panel_toolbar::begin_overlay(id, layout.card))
196 {
197 ImGui::PushStyleVar(ImGuiStyleVar_ItemSpacing, ImVec2(0.0f, to_pixels(CONTENT_GAP)));
198 const header_request header = draw_header(desc, preview_state.is_collapsed);
199 if(!preview_state.is_collapsed)
200 {
201 draw_image(desc, layout.image_height);
202 }
203 ImGui::PopStyleVar();
204 result.is_align_pressed = header.is_align_pressed;
205 if(header.is_collapse_pressed)
206 {
207 preview_state.is_collapsed = !preview_state.is_collapsed;
208 }
209 }
210 panel_toolbar::end_overlay();
211 return result;
212}
