26// Performs piecewise linear interpolation of a Color parameter.
27class dynamic_value_controller
28{
29 using value_type = Color;
30 using key_map = std::map<float, value_type>;
31
32public:
33 dynamic_value_controller(const key_map& keymap) : key_map_(keymap)
34 {
35 }
36
37 value_type get_value(float time) const
38 {
39 auto itUpper = key_map_.upper_bound(time + 1e-6f);
40 auto itLower = itUpper;
41 --itLower;
42
43 if(itLower == key_map_.end())
44 {
45 return itUpper->second;
46 }
47
48 if(itUpper == key_map_.end())
49 {
50 return itLower->second;
51 }
52
53 float lowerTime = itLower->first;
54 const auto& lowerVal = itLower->second;
55 float upperTime = itUpper->first;
56 const auto& upperVal = itUpper->second;
57
58 if(lowerTime == upperTime)
59 {
60 return lowerVal;
61 }
62
63 return interpolate(lowerTime, lowerVal, upperTime, upperVal, time);
64 };
65
66private:
67 value_type interpolate(float lowerTime,
68 const value_type& lowerVal,
69 float upperTime,
70 const value_type& upperVal,
71 float time) const
72 {
73 const float tt = (time - lowerTime) / (upperTime - lowerTime);
74 const auto result = bx::lerp(lowerVal, upperVal, tt);
75 return result;
76 };
77
78 const key_map& key_map_;
79};

172void compute_perez_coeff(float _turbidity, float* _outPerezCoeff)
173{
174 const bx::Vec3 turbidity = {_turbidity, _turbidity, _turbidity};
175 for(uint32_t ii = 0; ii < 5; ++ii)
176 {
177 const bx::Vec3 tmp = bx::mad(ABCDE_t[ii], turbidity, ABCDE[ii]);
178 float* out = _outPerezCoeff + 4 * ii;
179 bx::store(out, tmp);
180 out[3] = 0.0f;
181 }

184float hour_of_day(math::vec3 sun_dir)
185{
186 // Define the ground normal vector (assuming flat and horizontal ground)
187 math::vec3 normal(0.0, -1.0, 0.0);
188
189 auto v1 = sun_dir;
190 auto v2 = normal;
191 auto ref = math::vec3(-1.0f, 0.0f, 0.0f);
192
193 float angle = math::orientedAngle(v1, v2, ref); // angle in [-pi, pi]
194 angle = math::mod(angle, 2 * math::pi<float>()); // angle in [0, 2pi]
195 angle = math::degrees(angle);
196 // The hour angle is 0 at 6:00, 90 at 12:00, and 180 at 18:00
197 // Therefore, we can use a simple linear formula to map the hour angle to the hour of day
198 float hour_of_day = angle / 15;
199
200 // Return the hour of day
201 return hour_of_day;
202}

347 gfx::render_view& rview,
348 const run_params& params) -> cloud_shadow_result
349{
350 cloud_shadow_result result;
351 if(params.cloud_mode == 0 || !cloud_shadow_program_.program || !cloud_shadow_program_.program->is_valid())
352 {
353 return result;
354 }
355 const math::vec3 sun_dir = math::normalize(-params.light_direction);
356 if(sun_dir.y <= cloud_shadow_min_sun_elevation)
357 {
358 return result;
359 }
360
361 const auto& cam_pos = camera.get_position();
362 result.extent = cloud_shadow_extent_sizes * params.cloud_size;
363 result.base_world_y = layer_base_world_y(params, camera);
364 result.opacity = params.cloud_shadow_opacity;
365 result.apply_to_lights = params.cloud_shadows;
366 // Centre the map where the sun ray through the camera meets the layer base, snapped to
367 // the texel grid so the texels do not swim as the camera moves.
368 const float texel = result.extent / float(cloud_shadow_resolution);
369 const float to_base = std::max(result.base_world_y - cam_pos.y, 0.0f) / sun_dir.y;
370 math::vec2 origin(cam_pos.x + sun_dir.x * to_base, cam_pos.z + sun_dir.z * to_base);
371 origin.x = std::floor(origin.x / texel) * texel;
372 origin.y = std::floor(origin.y / texel) * texel;
373 result.origin = origin;
374
375 constexpr uint64_t flags = BGFX_TEXTURE_RT | BGFX_SAMPLER_U_CLAMP | BGFX_SAMPLER_V_CLAMP;
376 const usize32_t size{cloud_shadow_resolution, cloud_shadow_resolution};
377 // Mips: the lowest level is the mean transmittance the irradiance bake reads.
378 const auto& tex = rview.tex_acquire(
379 cloud_shadow_tex_key,
380 {.size = size, .format = bgfx::TextureFormat::R8, .flags = flags, .has_mips = true});

392

394 pass.bind(fbo.get());
395 pass.set_view_proj(camera.get_view_relative(), camera.get_projection());
396 pass.clear(BGFX_CLEAR_COLOR, 0xFFFFFFFF, 0.0f, 0);

398 cloud_shadow_program_.program->begin();

412 }

414 {
415 gfx::set_texture(cloud_shadow_program_.s_cloudNoise2D, 1, cloud_noise.flat_noise.get());
416 }

427 result.valid = true;
428 // FNV-1a over the uniforms the map renders from and its placement.
429 const auto hash_floats = [](uint64_t hash, const float* values, size_t count)
430 {
431 for(size_t i = 0; i < count; ++i)
432 {
433 uint32_t bits = 0;
434 std::memcpy(&bits, &values[i], sizeof(bits));
435 hash = (hash ^ uint64_t(bits)) * 0x100000001b3ull;

438 };

440 result.origin.y,
441 result.extent,
442 result.base_world_y,
443 result.opacity,
444 result.apply_to_lights ? 1.0f : 0.0f,
445 float(cloud_shadow_resolution)};
446 uint64_t signature = hash_floats(0xcbf29ce484222325ull, placement, std::size(placement));
447 signature = hash_floats(signature, &perez.sun_direction.x, 3);
448 signature = hash_floats(signature, &uniforms.exposition[0], sizeof(uniforms) / sizeof(float));
449 result.signature = signature;
450 return result;

468auto atmospheric_pass_perez::run_cloud_prepass(const camera& camera,
469 gfx::render_view& rview,
470 const usize32_t& output_size,
471 const gfx::texture::ptr& depth,
472 const irradiance_perez_params& perez,
473 const cloud_uniform_block& uniforms,
474 const run_params& params) -> cloud_prepass_result
475{
476 const auto& view = camera.get_view_relative();
477 const auto& proj = camera.get_projection();
478
479 const uint32_t half_w = std::max(1u, output_size.width / 2);
480 const uint32_t half_h = std::max(1u, output_size.height / 2);
481 const usize32_t half_size{half_w, half_h};
482
483 constexpr uint64_t cloud_tex_flags = BGFX_TEXTURE_RT | BGFX_SAMPLER_U_CLAMP | BGFX_SAMPLER_V_CLAMP;
484
485 auto& state = rview.data().get_or_emplace<cloud_view_state>(cloud_view_state::view_key);
486 auto& cloud_frame_count = state.frame_count;
487
488 // The wind offset wraps to the noise tile period; the per-frame advance (unwrapped)
489 // drives the history reprojection.
490 constexpr float wind_period = float(cloud_noise_textures::tile_period);
491 float wind_delta[2] = {0.0f, 0.0f};
492 for(int i = 0; i < 2; ++i)
493 {
494 const float cur = params.cloud_wind_offset[i];
495 float delta = cur - state.prev_wind_offset[i];
496 if(delta > 0.5f * wind_period)
497 {
498 delta -= wind_period;
499 }
500 else if(delta < -0.5f * wind_period)
501 {
502 delta += wind_period;
503 }
504 wind_delta[i] = delta;
505 state.prev_wind_offset[i] = cur;
506 }
507
508 bool recreated = false;
509 const auto mark_recreated = [&](const gfx::texture::ptr& /*tex*/)
510 {
511 recreated = true;
512 };
513 gfx::texture::ptr cloud_tex[2];
514 gfx::texture::ptr cloud_aux[2];
515 for(int i = 0; i < 2; ++i)
516 {
517 cloud_tex[i] = rview.tex_acquire(
518 cloud_tex_keys[i],
519 {.size = half_size, .format = bgfx::TextureFormat::RGBA16F, .flags = cloud_tex_flags},
520 mark_recreated);
521 // R = history sample count / max, G = scene distance (km) for the depth-aware composite.
522 cloud_aux[i] = rview.tex_acquire(
523 cloud_aux_keys[i],

665void atmospheric_pass_perez::run(gfx::frame_buffer::ptr input,
666 gfx::frame_buffer::ptr composite_target,
667 const camera& camera,
668 gfx::render_view& rview,
669 delta_t dt,
670 const run_params& params)
671{
672 const auto& view = camera.get_view_relative();
673 const auto& proj = camera.get_projection();
674
675 const auto surface = input.get();
676 const auto output_size = surface->get_size();
677
678 irradiance_perez_params perez;
679 compute_irradiance_perez_params(params.light_direction, params.turbidity, params.sun_intensity, perez);
680 // The visible sky and the clouds are scene lighting: written with the view's pre-exposure.
681 perez.exposition *= params.sky_brightness * params.pre_exposure.value;
682
683 const float hour = ANONYMOUS::hour_of_day(-params.light_direction);
684 const cloud_uniform_block uniforms = make_cloud_uniforms(params, camera, perez.exposition, hour);
685
686 // === Pass 1: cloud pre-pass at half resolution with temporal accumulation ===
687 // The input frame buffer carries the scene depth as its second attachment.
688 const gfx::texture::ptr depth = surface->get_attachment_count() > 1 ? surface->get_texture(1) : gfx::texture::ptr{};
689 cloud_prepass_result prepass;
690 const bool volumetric = params.cloud_mode == cloud_mode_volumetric;
691 if(volumetric && cloud_program_.program && cloud_program_.program->is_valid())
692 {
693 prepass = run_cloud_prepass(camera, rview, output_size, depth, perez, uniforms, params);
694 }
695 else
696 {
697 release_cloud_resources(rview);
698 }
699
700 // === Pass 2: sky pass (full resolution, composites the half-res clouds) ===
701 gfx::render_pass pass("Atmospherics/Sky Pass");
702 pass.bind(surface);
703 pass.set_view_proj(view, proj);
704
705 if(atmospheric_program_.program->is_valid())
706 {
707 atmospheric_program_.program->begin();
708
709 gfx::set_uniform(atmospheric_program_.u_sunLuminance, perez.sun_luminance_rgb);
710 gfx::set_uniform(atmospheric_program_.u_skyLuminanceXYZ, perez.sky_luminance_xyz);
711 gfx::set_uniform(atmospheric_program_.u_skyLuminance, perez.sky_luminance_rgb);
712 gfx::set_uniform(atmospheric_program_.u_sunDirection, perez.sun_direction);
713 gfx::set_uniform(atmospheric_program_.u_parameters, uniforms.exposition);
714 gfx::set_uniform(atmospheric_program_.u_perezCoeff, &perez.perez_coeff[0][0], 5);
715 gfx::set_uniform(atmospheric_program_.u_cloudParams, uniforms.cloud_params);
716 gfx::set_uniform(atmospheric_program_.u_cloudParams2, uniforms.cloud_params2);
717 gfx::set_uniform(atmospheric_program_.u_cloudParams3, uniforms.cloud_params3);
718 gfx::set_uniform(atmospheric_program_.u_cloudParams4, uniforms.cloud_params4);
719 gfx::set_uniform(atmospheric_program_.u_cloudCamera, uniforms.camera);
720
721 // Layout of pre_exposure.sh (the dome's constant terms take the value from here).
722 gfx::set_uniform(atmospheric_program_.u_pre_exposure, params.pre_exposure.to_uniform().data());
723 params.pre_exposure.bind();
724
725 auto& cloud_noise = default_textures::get().cloud_noise();
726 if(cloud_noise.flat_noise)
727 {
728 gfx::set_texture(atmospheric_program_.s_cloudNoise2D, 1, cloud_noise.flat_noise.get());
729 }
730
731 irect32_t rect(0, 0, irect32_t::value_type(output_size.width), irect32_t::value_type(output_size.height));
732 bgfx::setScissor(rect.left, rect.top, rect.width(), rect.height());
733
734 bgfx::setState(BGFX_STATE_WRITE_RGB | BGFX_STATE_WRITE_A | BGFX_STATE_DEPTH_TEST_EQUAL);
735 bgfx::setIndexBuffer(ib_->native_handle());
736 bgfx::setVertexBuffer(0, vb_->native_handle());
737 bgfx::submit(pass.id, atmospheric_program_.program->native_handle());
738
739 bgfx::setState(BGFX_STATE_DEFAULT);
740 atmospheric_program_.program->end();
741 }
742
743 // === Pass 3: volumetric composite over the whole frame (sky and geometry) ===
744 if(volumetric)
745 {
746 run_cloud_composite(composite_target.get(), camera, output_size, depth, prepass);
747 }
748
749 bgfx::discard();
750}

768constexpr int k_perez_bake_samples = 64;
771constexpr float k_perez_min_unit_irradiance = 1e-3f;
772
773auto radical_inverse_vdc(uint32_t bits) -> float
774{
775 bits = (bits << 16u) | (bits >> 16u);
776 bits = ((bits & 0x55555555u) << 1u) | ((bits & 0xAAAAAAAAu) >> 1u);
777 bits = ((bits & 0x33333333u) << 2u) | ((bits & 0xCCCCCCCCu) >> 2u);
778 bits = ((bits & 0x0F0F0F0Fu) << 4u) | ((bits & 0xF0F0F0F0u) >> 4u);
779 bits = ((bits & 0x00FF00FFu) << 8u) | ((bits & 0xFF00FF00u) >> 8u);
780 return float(bits) * 2.3283064365386963e-10f;
781}

784auto perez_channel(float a, float b, float c, float d, float e, float costeta, float cosgamma) -> float
785{
786 const float inv_costeta = 1.0f / std::max(costeta, 0.01f);
787 return (1.0f + a * std::exp(b * inv_costeta)) *
788 (1.0f + c * std::exp(d * std::acos(cosgamma)) + e * cosgamma * cosgamma);
789}
790} // namespace

792auto compute_perez_horizontal_irradiance(const irradiance_perez_params& perez) -> float
793{
794 // The shader's chain, per channel of the xyY coefficients (perez_coeff[row][channel]).
795 const float* A = perez.perez_coeff[0];
796 const float* B = perez.perez_coeff[1];
797 const float* C = perez.perez_coeff[2];
798 const float* D = perez.perez_coeff[3];
799 const float* E = perez.perez_coeff[4];
800 const math::vec3 sun_dir = math::normalize(perez.sun_direction);
801 const float cosgammas = std::clamp(sun_dir.y, -0.9999f, 0.9999f);
802 float p0_inv[3];
803 for(int c = 0; c < 3; ++c)
804 {
805 p0_inv[c] = 1.0f / std::max(perez_channel(A[c], B[c], C[c], D[c], E[c], 1.0f, cosgammas), 0.0001f);
806 }
807 const math::vec3& xyz = perez.sky_luminance_xyz;
808 const float denom = std::max(xyz.x + xyz.y + xyz.z, 0.0001f);
809 const float sky_xyY[3] = {xyz.x / denom, xyz.y / denom, xyz.y};
810 // Only the SH bands that survive at the zenith: Y00, Y1(y), Y2(3z^2 - 1), Y2(x^2 - y^2).
811 float sh0 = 0.0f;
812 float sh1 = 0.0f;
813 float sh7 = 0.0f;
814 float sh8 = 0.0f;
815 const float d_omega = 2.0f * math::pi<float>() / float(k_perez_bake_samples);
816 for(int i = 0; i < k_perez_bake_samples; ++i)
817 {
818 const float e_x = float(i) / float(k_perez_bake_samples);
819 const float e_y = radical_inverse_vdc(uint32_t(i));
820 const float phi = 2.0f * math::pi<float>() * e_x;
821 const float cos_theta = 1.0f - e_y;
822 const float sin_theta = std::sqrt(std::max(0.0f, 1.0f - cos_theta * cos_theta));
823 const math::vec3 dir(sin_theta * std::cos(phi), cos_theta, sin_theta * std::sin(phi));
824 const float costeta = std::max(dir.y, 0.01f);
825 const float cosgamma = std::clamp(math::dot(dir, sun_dir), -0.9999f, 0.9999f);
826 float Yp[3];
827 for(int c = 0; c < 3; ++c)
828 {
829 Yp[c] = sky_xyY[c] * perez_channel(A[c], B[c], C[c], D[c], E[c], costeta, cosgamma) * p0_inv[c];
830 }
831 const float yp_y = std::max(Yp[1], 0.0001f);
832 const float X = Yp[0] * Yp[2] / yp_y;
833 const float Y = Yp[2];
834 const float Z = (1.0f - Yp[0] - Yp[1]) * Yp[2] / yp_y;
835 // irradiance_convertXYZ2RGB, clamped at zero per channel like the shader; the
836 // shader's saturation boost preserves this luma and so does not appear.
837 const float r = std::max(3.2404542f * X - 1.5371385f * Y - 0.4985314f * Z, 0.0f);
838 const float g = std::max(-0.9692660f * X + 1.8760108f * Y + 0.0415560f * Z, 0.0f);
839 const float b = std::max(0.0556434f * X - 0.2040259f * Y + 1.0572252f * Z, 0.0f);
840 const float luma = 0.2126f * r + 0.7152f * g + 0.0722f * b;
841 sh0 += luma * 0.282095f * d_omega;
842 sh1 += luma * 0.488603f * dir.y * d_omega;
843 sh7 += luma * 0.315392f * (3.0f * dir.z * dir.z - 1.0f) * d_omega;
844 sh8 += luma * 0.546274f * (dir.x - dir.y) * (dir.x + dir.y) * d_omega;
845 }
846 // eval_irradiance_sh at N = (0, 1, 0): lobes A0, A1 * y, A2 * (3z^2 - 1) = -A2, A2 * (x^2 - y^2) = -A2.
847 const float A0 = math::pi<float>();
848 const float A1 = 2.0f * math::pi<float>() / 3.0f;
849 const float A2 = math::pi<float>() * 0.25f;
850 const float irradiance = A0 * 0.282095f * sh0 + A1 * 0.488603f * sh1 - A2 * 0.315392f * sh7 - A2 * 0.546274f * sh8;
851 return std::max(irradiance, 0.0f);
852}

869void compute_irradiance_perez_params(const math::vec3& light_direction,
870 float turbidity,
871 float sun_intensity,
872 irradiance_perez_params& out)
873{
874 math::vec3 sun_dir(-light_direction.x, -light_direction.y, -light_direction.z);
875 sun_dir = math::normalize(sun_dir);
876
877 auto hour = ANONYMOUS::hour_of_day(-light_direction);
878 ANONYMOUS::dynamic_value_controller sun_luminance_dc(ANONYMOUS::sunLuminanceXYZTable);
879 ANONYMOUS::dynamic_value_controller sky_luminance_dc(ANONYMOUS::skyLuminanceXYZTable);
880 auto sunLuminanceXYZ = sun_luminance_dc.get_value(hour);
881 auto sunLuminanceRGB = ANONYMOUS::xyzToRgb(sunLuminanceXYZ);
882 out.sun_luminance_rgb = math::vec3(sunLuminanceRGB.x, sunLuminanceRGB.y, sunLuminanceRGB.z);
883
884 auto skyLuminanceXYZ = sky_luminance_dc.get_value(hour);
885 out.sky_luminance_xyz =
886 math::vec3(skyLuminanceXYZ.x, skyLuminanceXYZ.y, skyLuminanceXYZ.z);
887 auto skyLuminanceRGB = ANONYMOUS::xyzToRgb(skyLuminanceXYZ);
888 out.sky_luminance_rgb = math::vec3(skyLuminanceRGB.x, skyLuminanceRGB.y, skyLuminanceRGB.z);
889
890 out.sun_direction = sun_dir;
891
892 ANONYMOUS::compute_perez_coeff(turbidity, &out.perez_coeff[0][0]);
893 out.horizontal_irradiance_unit = compute_perez_horizontal_irradiance(out);
894
895 // The one shared Perez -> engine conversion (see perez_luminance.h): the sky dome,
896 // the irradiance bake and the flat ambient all inherit this value, so their ratios
897 // cannot drift apart. SUN-RELATIVE when a directional light exists: the exposition that
898 // puts the bake's horizontal irradiance at the atmosphere's sky-to-sun ratio for this sun
899 // elevation (compute_atmosphere_sky_to_sun_ratio) x the sun's luminous intensity, faded into the fixed conversion through the ambient's own day/night ramp -
900 // at the horizon the luminance tables collapse and a solved exposition would amplify
901 // their residual into a night-time sky.
902 const float fixed_exposition = compute_perez_exposition(sun_dir.y);
903 out.exposition = fixed_exposition;
904 out.sun_relative = false;
905 if(sun_intensity > 0.0f && out.horizontal_irradiance_unit > k_perez_min_unit_irradiance)
906 {
907 const float solved = compute_atmosphere_sky_to_sun_ratio(sun_dir.y) * sun_intensity / out.horizontal_irradiance_unit;
908 out.exposition = bx::lerp(fixed_exposition, solved, compute_perez_sun_weight(sun_dir.y));
909 out.sun_relative = true;
910 }
