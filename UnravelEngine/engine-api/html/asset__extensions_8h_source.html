186inline auto is_known_format(const std::string& extension) -> bool
187{
188 std::string lower = extension;
189 std::transform(lower.begin(), lower.end(), lower.begin(), [](unsigned char c) { return char(std::tolower(c)); });
190 for(const auto& formats : get_all_formats())
191 {
192 if(std::find(formats.begin(), formats.end(), lower) != formats.end())
193 {
194 return true;
195 }
196 }
197 return false;

223// Template specializations for format versions
224// Increment these versions when the binary serialization format changes for each asset type
225template<>
226inline auto get_format_version<unravel::mesh>() -> uint64_t
227{
228 // 8: per-submesh stable_id, per-bone bind-space bounds, per-LOD submesh bbox recompute.
229 // 9: mesh bounds are bind-pose only (import-time animation sampling removed; runtime
230 // pose bounds handle animation-driven expansion).
231 // 11: baked sparse signed distance field for GI tracing.
232 // 13: One SDF per SUBMESH instead of one per mesh. Submeshes are drawn at their own node
233 // transforms, so a single whole-mesh field cannot be placed correctly for a model whose
234 // submeshes differ -- it duplicated the entire model once per transform.
235 // 12: SDF adjacency computed on position-welded topology, open surfaces fall back to an
236 // unsigned shell.
237 // 14: A submesh's SDF covers that submesh alone. It was selected by data group, which is the
238 // MATERIAL index, so every submesh sharing a material baked that whole group -- each one
239 // carrying its siblings' geometry at its own transform.
240 // 15: fields are bounded by a TOTAL voxel budget, not only by a per-axis resolution cap. The
241 // per-axis cap alone permitted 16.7M voxels in one field, so a model split into many
242 // submeshes overran the atlas several times over and the bake cost hours of voxel work.
243 // 16: fields bake from LOD 2 by default rather than the base topology. A DEFAULT counts as a
244 // bake-algorithm change for every asset that never overrode it -- their .meta files carry
245 // no value to compare, so nothing else would notice the switch and they would keep the
246 // field baked from full detail.
247 // 17: unsigned SHELL fields classify bricks in the field's own space. The stored voxels are
248 // `unsigned - thickness`, but brick classification and the empty-brick distance used the
249 // raw unsigned value, so every empty entry over-reported by exactly the thickness -- an
250 // over-estimate, which is the one direction a conservative field may never err in, and a
251 // trace stepped that far too much and passed through the shell. It also stops the interior
252 // of a thick shell being stored as surface bricks, so a scene of open meshes uses far less
253 // atlas.
254 // 18: triangles carrying no surface -- zero-area slivers and non-finite positions -- no longer
255 // reach the bake or the bounds. The renderer discards them, so such a submesh looks empty in
256 // the viewport, but their corners still sized the field: measured extents of 7,000 to 10,000
257 // units on a scene whose buildings are tens across, giving voxels of 40+ units. Since an
258 // unsigned shell is floored at one voxel, those fields traced as solid blocks swallowing
259 // whole streets, sourced from geometry nobody could see.
260 // 19: a submesh whose geometry is scattered over more than 32x its largest connected piece is
261 // REFUSED a field rather than given an unusable one. The voxel comes from the bounds, so
262 // parts spread far apart each fall below one voxel and what the bake produced was a blob the
263 // size of the spread, tracing as solid geometry that is not there. Refusing costs the GI
264 // contribution of parts too small to have contributed usefully; the field cost a
265 // neighbourhood. The sliver threshold also moved from 1e-6 to 1e-3 in the same pass.
266 // 20: the closedness test now requires every welded edge to be shared by EXACTLY two faces,
267 // and triangles whose corners weld together no longer contribute topology. Double-sided
268 // sheet geometry (the engine's own plane primitive is two coincident, oppositely wound
269 // sheets) previously counted as closed, its cancelling pseudonormals made every voxel's
270 // sign floating-point noise, and the field baked as random brick-quantised walls and
271 // stairs. Such geometry now takes the unsigned-shell path. UV-sphere pole slivers also
272 // stop poisoning the pole pseudonormals.
273 // 21: skinned submeshes are refused a field. The bake reads bind-pose vertices and skinning
274 // rewrites the surface every frame, so the field could only ever occlude as a rigid
275 // bind-pose statue pinned to the entity's root transform. The runtime walk also refuses
276 // stale fields, so this bump is what reclaims the disk and atlas space they held.
277 // 22: fields carry the baked geometry's own box (mesh_sdf::surface_bounds). The grid rounds up to whole
278 // bricks on the max side only, so the box cannot be recovered from the field bounds, and the composed
279 // distance's per-axis box bound grew by that slack times the placement's scale.
280 // 23: a flat axis centres a sample layer on the sheet. Between two layers an unsigned sheet never read below
281 // half a voxel, so the GI trace's half-voxel surface expand let rays pass through scaled planes (floors).
282 // 24: a chain's coarser levels sign open surfaces with the finest level's vote reach. Each level's own reach
283 // (four voxel diagonals) signed a 1.6 m layer behind a vault's back faces inside at 0.4 m voxels, which
284 // the GI cascade's coarse-first distance composed above the floor slab another submesh holds.
285 // 25: mesh cards are built at compile time and stored in the mesh asset, one set per submesh for its imported
286 // material's sidedness, from LOD 1 by default, instead of on first use at runtime.
287 //
288 // NOTE: the compiled asset is a function of the BAKE ALGORITHM, not only of the source
289 // mesh. Any change to mesh_sdf_baker that alters its output needs a bump here, or existing
290 // projects silently keep the field produced by the previous code. The same holds for the card builder.
291 return 25;
292}

405inline auto has_filename_extension(const std::string& key) noexcept -> bool
406{
407 if(key.empty())
408 {
409 return false;
410 }
411 for(std::size_t i = key.size(); i-- > 0;)
412 {
413 const char c = key[i];
414 if(c == '/' || c == '\\')
415 {
416 return false;
417 }
418 if(c == '.')
419 {
420 const bool has_chars_after = (i + 1) < key.size();
421 const bool not_leading_dot =
422 (i > 0) && key[i - 1] != '/' && key[i - 1] != '\\';
423 return has_chars_after && not_leading_dot;
424 }
425 }
426 return false;
427}

450inline auto get_source_path_from_compiled(const fs::path& compiled) -> fs::path
451{
452 constexpr hpp::string_view compiled_suffix = ".asset";
453 const std::string name = compiled.filename().string();
454 const size_t suffix_pos = name.rfind(compiled_suffix);
455 if(suffix_pos != std::string::npos && suffix_pos > 0)
456 {
457 const hpp::string_view rest = hpp::string_view(name).substr(suffix_pos + compiled_suffix.size());
458 const bool is_compiled_suffix =
459 rest.empty() || (rest.front() == '.' && rest.find('.', 1) == hpp::string_view::npos);
460 if(is_compiled_suffix)
461 {
462 return compiled.parent_path() / name.substr(0, suffix_pos);
463 }
464 }
465 return fs::reduce_trailing_extensions(compiled);
466}

480template<typename T>
481inline auto resolve_key_missing_extension(const std::string& key) -> std::string
482{
483 fs::error_code err;
484 for(const auto& format : get_suported_formats<T>())
485 {
486 std::string candidate;
487 candidate.reserve(key.size() + format.size());
488 candidate.assign(key);
489 candidate.append(format);
490 if(fs::exists(fs::resolve_protocol(candidate), err))
491 {
492 return candidate;
493 }
494 }
495 return key + get_format<T>();
496}

651inline auto is_binary(const std::string& ex) -> bool
652{
653 // Binary formats (compiled/processed assets)
654 if(is_format<gfx::texture>(ex))
655 {
656 // Most texture formats are binary, except some like .hdr which can be text-based
657 // But for simplicity, we'll consider all texture formats as binary
658 return true;
659 }
660 if(is_format<unravel::mesh>(ex))
661 {
662 // .emesh is binary, but source formats like .obj, .gltf can be text
663 // Check for binary formats specifically
664 return ex == ".emesh" || ex == ".glb" || ex == ".fbx" || ex == ".FBX" ||
665 ex == ".blend" || ex == ".3ds" || ex == ".dae";
666 }
667 if(is_format<unravel::audio_clip>(ex))
668 {
669 // Audio formats are typically binary
670 return true;
671 }
672 if(is_format<gfx::shader>(ex))
673 {
674 // .sc shader files are text-based
675 return false;
676 }
677 if(is_format<unravel::material>(ex))
678 {
679 // .mat and .ematerial are typically text-based (JSON/XML)
680 return false;
681 }
682 if(is_format<unravel::animation_clip>(ex))
683 {
684 // .anim files are typically text-based
685 return false;
686 }
687 if(is_format<unravel::prefab>(ex))
688 {
689 // .pfb files are typically text-based
690 return false;
691 }
692 if(is_format<unravel::scene_prefab>(ex))
693 {
694 // .spfb files are typically text-based
695 return false;
696 }
697 if(is_format<unravel::animation_controller>(ex))
698 {
699 // .actrl files are text (associative archive)
700 return false;
701 }
702 if(is_format<unravel::physics_material>(ex))
703 {
704 // Physics material files are typically text-based
705 return false;
706 }
707 if(is_format<unravel::script>(ex))
708 {
709 // .cs script files are text-based
710 return false;
711 }
712 if(is_format<unravel::font>(ex))
713 {
714 // Font files (.ttf, .otf) are binary
715 return true;
716 }
717 if(is_format<unravel::ui_tree>(ex))
718 {
719 // .rhtml files are text-based
720 return false;
721 }
722 if(is_format<unravel::style_sheet>(ex))
