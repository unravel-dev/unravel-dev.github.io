208auto internal_m2n_camera_world_to_viewport_point(entt::entity id, const math::vec3& world) -> math::vec3
209{
210 if(auto comp = safe_get_component<camera_component>(id))
211 {
212 const auto& cam = comp->get_camera();
213 const auto& pos = cam.get_viewport_pos();
214 const auto& size = cam.get_viewport_size();
215 if(size.width == 0 || size.height == 0)
216 {
217 return {};
218 }
219 const auto screen = cam.world_to_viewport(world);
220 return {(screen.x - float(pos.x)) / float(size.width),
