61auto find_scenes_outside_build(rtti::context& ctx,
62 const deploy_settings& params,
63 const std::string& startup_scene,
64 const ImGuiTextFilter* search) -> std::vector<asset_handle<scene_prefab>>
65{
66 const std::string project_root = std::string(deployment::CONTENT_PROTOCOL) + ":/";
67 auto scenes = ctx.get_cached<asset_manager>().get_assets_with_predicate<scene_prefab>(
68 [&](const asset_handle<scene_prefab>& scene)
69 {
70 const auto& key = scene.id();
71 const bool is_listed = std::any_of(params.deploy_scenes.begin(),
72 params.deploy_scenes.end(),
73 [&key](const std::string& listed)
74 {
75 return deployment::is_same_key(listed, key);
76 });
77 const bool passes_search = search == nullptr || search->PassFilter(deployment::get_display_path(key).c_str());
78 return deployment::is_key_under(key, project_root) && !is_listed && !deployment::is_same_key(key, startup_scene) &&
79 passes_search;
80 });
81 // The storage puts the empty handle first without asking the predicate.
82 scenes.erase(std::remove_if(scenes.begin(),
83 scenes.end(),
84 [](const asset_handle<scene_prefab>& scene)
85 {
86 return !scene;
87 }),
88 scenes.end());
89 std::sort(scenes.begin(),
90 scenes.end(),
91 [](const asset_handle<scene_prefab>& lhs, const asset_handle<scene_prefab>& rhs)
92 {
93 return lhs.id() < rhs.id();
94 });
95 return scenes;
96}

268{

894void deploy_content_view::draw_status_cell(const deployment::tree_node& node, const deploy_settings& params) const
895{
896 const ImU32 muted = widgets::get_muted_color();
897 const ImU32 text = ImGui::GetColorU32(ImGuiCol_Text);
898 if(node.is_folder())
899 {
900 const auto rule = deployment::has_own_rule(params, node.key) ? deployment::find_rule(params, node.key) : deployment::content_rule::none;
901 if(rule == deployment::content_rule::include)
902 {
903 draw_cell_label(ICON_MDI_PLUS_CIRCLE_OUTLINE, ADDED_COLOR, "Folder added by you", text);
904 return;
905 }
906 if(rule == deployment::content_rule::exclude)
907 {
908 draw_cell_label(ICON_MDI_MINUS_CIRCLE_OUTLINE, widgets::WARNING_COLOR, "Folder left out by you", text);
909 return;
910 }
911 draw_cell_label(nullptr, 0, fmt::format("{} of {} ship", node.shipped_count, node.asset_count), muted);
912 return;
913 }
914 const auto& entry = content_->entries[std::size_t(node.entry)];
915 switch(deployment::get_reason(params, entry))
916 {
917 case deployment::content_reason::scene:
918 draw_cell_label(ICON_MDI_MAP_OUTLINE, widgets::get_accent_u32(), "Scene in the build", text);
919 break;
920 case deployment::content_reason::referenced:
921 draw_cell_label(ICON_MDI_LINK_VARIANT,
922 widgets::get_accent_u32(),
923 fmt::format("Used by {}", deployment::get_key_name(entry.used_by)),
924 text);
925 if(ImGui::BeginItemTooltip())
926 {
927 ImGui::TextUnformatted(get_use_chain(entry).c_str());
928 ImGui::EndTooltip();
929 }
930 break;
931 case deployment::content_reason::named_in_script:
932 draw_cell_label(ICON_MDI_CODE_BRACES,
933 widgets::get_accent_u32(),
934 fmt::format("Named in {}", deployment::get_key_name(entry.used_by)),
935 text);
936 draw_tooltip_text("A script names its path:\n" + deployment::get_display_path(entry.used_by));
937 break;
938 case deployment::content_reason::included:
939 draw_cell_label(ICON_MDI_PLUS_CIRCLE_OUTLINE, ADDED_COLOR, "Added by you", text);
940 draw_tooltip_text("You chose to ship it. Right click > Use Detected to undo.");
941 break;
942 case deployment::content_reason::excluded:
943 draw_cell_label(ICON_MDI_MINUS_CIRCLE_OUTLINE,
944 widgets::WARNING_COLOR,
945 entry.is_used ? "Left out, but used" : "Left out by you",
946 text);
947 if(ImGui::BeginItemTooltip())
948 {
949 ImGui::TextUnformatted(entry.is_used ? (get_use_chain(entry) + "\n\nThe game may miss it.").c_str()
950 : "You chose to leave it out. Right click > Use Detected to undo.");
951 ImGui::EndTooltip();
952 }
953 break;
954 case deployment::content_reason::unused:
955 default:
956 draw_cell_label(nullptr, 0, params.deploy_only_used_assets ? "Not used" : "Not used by the scenes", muted);
957 break;
958 }
959}
