68auto blend(const animation_pose::root_motion_result& r1, const animation_pose::root_motion_result& r2, float factor)
69 -> animation_pose::root_motion_result
70{
71 animation_pose::root_motion_result result;
72 result.root_transform_delta = blend(r1.root_transform_delta, r2.root_transform_delta, factor);
73
74 // Weights of a pose without root motion are meaningless defaults - use the
75 // other pose's weights outright, and lerp only when both have root motion.
76 // Multiplying them could zero BOTH the root and the bone weight of an axis,
77 // leaving it neither animated nor moved.
78 const bool r1_has_position = has_root_position(r1);
79 const bool r2_has_position = has_root_position(r2);
80 if(r1_has_position && r2_has_position)
81 {
82 result.root_position_weights = math::lerp(r1.root_position_weights, r2.root_position_weights, factor);
83 result.bone_position_weights = math::lerp(r1.bone_position_weights, r2.bone_position_weights, factor);
84 }
85 else if(r1_has_position)
86 {
87 result.root_position_weights = r1.root_position_weights;
88 result.bone_position_weights = r1.bone_position_weights;
89 }
90 else if(r2_has_position)
91 {
92 result.root_position_weights = r2.root_position_weights;
93 result.bone_position_weights = r2.bone_position_weights;
94 }
95
96 const bool r1_has_rotation = has_root_rotation(r1);
97 const bool r2_has_rotation = has_root_rotation(r2);
98 if(r1_has_rotation && r2_has_rotation)
99 {
100 result.root_rotation_weight = math::lerp(r1.root_rotation_weight, r2.root_rotation_weight, factor);
101 result.bone_rotation_weight = math::lerp(r1.bone_rotation_weight, r2.bone_rotation_weight, factor);
102 }
103 else if(r1_has_rotation)
104 {
105 result.root_rotation_weight = r1.root_rotation_weight;
106 result.bone_rotation_weight = r1.bone_rotation_weight;
107 }
108 else if(r2_has_rotation)
109 {
110 result.root_rotation_weight = r2.root_rotation_weight;
111 result.bone_rotation_weight = r2.bone_rotation_weight;
112 }
113
114 if(!r1_has_position)
115 {
116 result.root_position_node_index = r2.root_position_node_index;
117 }
118 else if(!r2_has_position)
119 {
120 result.root_position_node_index = r1.root_position_node_index;
121 }
122 else
123 {
124 result.root_position_node_index = factor < 0.5f ? r1.root_position_node_index : r2.root_position_node_index;
125 }
126
127 if(!r1_has_rotation)
128 {
129 result.root_rotation_node_index = r2.root_rotation_node_index;
130 }
131 else if(!r2_has_rotation)
132 {
133 result.root_rotation_node_index = r1.root_rotation_node_index;
134 }
135 else
136 {
137 result.root_rotation_node_index = factor < 0.5f ? r1.root_rotation_node_index : r2.root_rotation_node_index;
138 }
139 return result;

148void blend_poses_by_node_index_sorted(const animation_pose& pose1,
149 const animation_pose& pose2,
150 float factor,
151 animation_pose& result)
152{
153 result.nodes.clear();
154 result.nodes.reserve(pose1.nodes.size() + pose2.nodes.size());
155 result.motion_result = blend(pose1.motion_result, pose2.motion_result, factor);
156 blend_curves(pose1.curves, pose2.curves, factor, result.curves);
157
158 size_t i1 = 0;
159 size_t i2 = 0;
160 while(i1 < pose1.nodes.size() || i2 < pose2.nodes.size())
161 {
162 if(i2 == pose2.nodes.size() || (i1 < pose1.nodes.size() && pose1.nodes[i1].desc.index < pose2.nodes[i2].desc.index))
163 {
164 append_one_sided_node(pose1.nodes[i1++], 1.0f - factor, result);
165 continue;
166 }
167 if(i1 == pose1.nodes.size() || pose2.nodes[i2].desc.index < pose1.nodes[i1].desc.index)
168 {
169 append_one_sided_node(pose2.nodes[i2++], factor, result);
170 continue;
171 }
172 const auto& node1 = pose1.nodes[i1++];
173 const auto& node2 = pose2.nodes[i2++];
174 const float weight1 = node1.coverage * (1.0f - factor);
175 const float weight2 = node2.coverage * factor;
176 auto& node = result.nodes.emplace_back();
177 node.desc = node1.desc;
178 node.transform = blend(node1.transform, node2.transform, get_incremental_factor(weight1, weight2));
179 node.coverage = weight1 + weight2;
180 }

183void blend_poses_by_node_index_sorted_multiway(const std::vector<animation_pose>& poses,
184 const std::vector<float>& weights,
185 animation_pose& result)
186{
187 result.nodes.clear();
188 result.curves.clear();
189 result.motion_result = {};
190
191 if(poses.empty() || weights.size() < poses.size())
192 {
193 return;
194 }
195
196 if(poses.size() == 1)
197 {
198 result = poses[0];
199 return;
200 }
201
202 float weight_sum{0.0f};
203 for(size_t p = 0; p < poses.size(); ++p)
204 {
205 weight_sum += weights[p];
206 }
207
208 // Curves: a weighted sum over every pose, normalized by the total weight, with a curve a pose
209 // lacks counting as 0 there - the same rule the pairwise blend uses.
210 const float inverse_sum = weight_sum > 0.0f ? 1.0f / weight_sum : 0.0f;
211 for(size_t p = 0; p < poses.size(); ++p)
212 {
213 const float w = weights[p] * inverse_sum;
214 for(const auto& curve_value : poses[p].curves)
215 {
216 auto it = std::lower_bound(result.curves.begin(),
217 result.curves.end(),
218 curve_value.id,
219 [](const animation_curve_value& lhs, uint32_t id)
220 {
221 return lhs.id < id;
222 });
223 if(it == result.curves.end() || it->id != curve_value.id)
224 {
225 it = result.curves.insert(it, animation_curve_value{curve_value.id, 0.0f});
226 }
227 it->value += curve_value.value * w;
228 }
229 }
230
231 // Root motion results, with incremental weight normalization.
232 float motion_weight{0.0f};
233 for(size_t p = 0; p < poses.size(); ++p)
234 {
235 const float w = weights[p];
236 if(p == 0)
237 {
238 result.motion_result = poses[p].motion_result;
239 motion_weight = w;
240 continue;
241 }
242 result.motion_result = blend(result.motion_result, poses[p].motion_result, get_incremental_factor(motion_weight, w));
243 motion_weight += w;
244 }
245
246 // Bones: every pose is sorted by node index; a cursor per pose (reused per thread: the pool
247 // blends one character's poses per task, every frame) walks them all in one merge.
248 const size_t k = poses.size();
249 thread_local std::vector<size_t> idx;
250 idx.assign(k, 0);
251 constexpr size_t NO_INDEX = static_cast<size_t>(-1);
252 while(true)
253 {
254 size_t min_index = NO_INDEX;
255 for(size_t p = 0; p < k; ++p)
256 {
257 if(idx[p] < poses[p].nodes.size())
258 {
259 min_index = std::min(min_index, poses[p].nodes[idx[p]].desc.index);
260 }
261 }
262 if(min_index == NO_INDEX)
263 {
264 break;
265 }
266
267 // The poses that animate this bone, each by its weight and its coverage; the share the
268 // others hold is left to what lies below (the bone's coverage).
269 float covered_weight{0.0f};
270 math::transform accum_transform{};
271 const animation_pose::node_desc* first_desc{nullptr};
272 for(size_t p = 0; p < k; ++p)
273 {
274 if(idx[p] >= poses[p].nodes.size() || poses[p].nodes[idx[p]].desc.index != min_index)
275 {
276 continue;
277 }
278 const auto& node = poses[p].nodes[idx[p]++];
279 const float w = weights[p] * node.coverage;
280 if(first_desc == nullptr)
281 {
282 accum_transform = node.transform;
283 covered_weight = w;
284 first_desc = &node.desc;
285 continue;
286 }
287 accum_transform = blend(accum_transform, node.transform, get_incremental_factor(covered_weight, w));
288 covered_weight += w;
289 }
290
291 auto& out = result.nodes.emplace_back();
292 out.desc = *first_desc;
293 out.transform = accum_transform;
294 out.coverage = weight_sum > 0.0f ? std::min(covered_weight / weight_sum, 1.0f) : 1.0f;
295 }
296}

320void blend_curves(const std::vector<animation_curve_value>& lhs,
321 const std::vector<animation_curve_value>& rhs,
322 float factor,
323 std::vector<animation_curve_value>& result)
324{
325 result.clear();
326 result.reserve(lhs.size() + rhs.size());
327 size_t il = 0;
328 size_t ir = 0;
329 while(il < lhs.size() || ir < rhs.size())
330 {
331 if(ir == rhs.size() || (il < lhs.size() && lhs[il].id < rhs[ir].id))
332 {
333 result.push_back({lhs[il].id, lhs[il].value * (1.0f - factor)});
334 ++il;
335 }
336 else if(il == lhs.size() || rhs[ir].id < lhs[il].id)
337 {
338 result.push_back({rhs[ir].id, rhs[ir].value * factor});
339 ++ir;
340 }
341 else
342 {
343 result.push_back({lhs[il].id, math::lerp(lhs[il].value, rhs[ir].value, factor)});
344 ++il;
345 ++ir;
346 }

350void make_pose_additive(animation_pose& pose, const animation_pose& reference)
351{
352 size_t ir = 0;
353 for(auto& node : pose.nodes)
354 {
355 while(ir < reference.nodes.size() && reference.nodes[ir].desc.index < node.desc.index)
356 {
357 ++ir;
358 }
359 math::transform delta = make_no_change();
360 if(ir < reference.nodes.size() && reference.nodes[ir].desc.index == node.desc.index)
361 {
362 const auto& ref = reference.nodes[ir].transform;
363 delta.set_translation(node.transform.get_translation() - ref.get_translation());
364 delta.set_rotation(math::normalize(node.transform.get_rotation() * glm::inverse(ref.get_rotation())));
365 delta.set_scale(node.transform.get_scale() - ref.get_scale());
366 }
367 node.transform = delta;
368 }
369 pose.motion_result.root_transform_delta = {};
370}

386 result.set_translation(base.get_translation() + delta.get_translation() * weight);

405void combine_curves(const std::vector<animation_curve_value>& base,
406 const std::vector<animation_curve_value>& layer,
407 animation_layer_blend_mode mode,
408 float weight,
409 std::vector<animation_curve_value>& result)
410{
411 result.clear();
412 result.reserve(base.size() + layer.size());
413 const bool additive = mode == animation_layer_blend_mode::additive;
414 size_t ib = 0;
415 size_t il = 0;
416 while(ib < base.size() || il < layer.size())
417 {
418 if(il == layer.size() || (ib < base.size() && base[ib].id < layer[il].id))
419 {
420 // A curve the layer does not have keeps the base value, as a bone it does not animate does.
421 result.push_back(base[ib]);
422 ++ib;
423 }
424 else if(ib == base.size() || layer[il].id < base[ib].id)
425 {
426 result.push_back({layer[il].id, layer[il].value * weight});
427 ++il;
428 }
429 else
430 {
431 const float value = additive ? base[ib].value + layer[il].value * weight
432 : math::lerp(base[ib].value, layer[il].value, weight);
433 result.push_back({base[ib].id, value});
434 ++ib;
435 ++il;
436 }
437 }
438}

442void blend_layer(const animation_pose& base,
443 const animation_pose& layer,
444 animation_layer_blend_mode mode,
445 float weight,
446 const std::vector<float>& bone_mask,
447 const animation_pose& ref_pose,
448 animation_pose& result)
449{
450 result.nodes.clear();
451 result.nodes.reserve(std::max(base.nodes.size(), layer.nodes.size()));
452 result.motion_result = base.motion_result;
453 result.motion_state = base.motion_state;
454 combine_curves(base.curves, layer.curves, mode, weight, result.curves);
455
456 size_t ib = 0;
457 size_t il = 0;
458 size_t ir = 0;
459 while(ib < base.nodes.size() || il < layer.nodes.size())
460 {
461 if(il == layer.nodes.size() || (ib < base.nodes.size() && base.nodes[ib].desc.index < layer.nodes[il].desc.index))
462 {
463 const auto& base_node = base.nodes[ib++];
464 auto& out = result.nodes.emplace_back(base_node);
465 out.transform = resolve_coverage(base_node, find_ref_node(ref_pose, base_node.desc.index, ir));
466 out.coverage = 1.0f;
467 continue;
468 }
469 const auto& layer_node = layer.nodes[il++];
470 const float bone_weight =
471 weight * get_mask_weight(bone_mask, layer_node.desc.index) * std::clamp(layer_node.coverage, 0.0f, 1.0f);
472 const auto* rest = find_ref_node(ref_pose, layer_node.desc.index, ir);
473 if(ib == base.nodes.size() || layer_node.desc.index < base.nodes[ib].desc.index)
474 {
475 // A bone only the layer animates: an override layer owns it by its weight, from the
476 // bone's rest pose; an additive change has no base to apply to and is dropped.
477 if(mode == animation_layer_blend_mode::override_pose && bone_weight > 0.0f)
478 {
479 auto& out = result.nodes.emplace_back(layer_node);
480 out.coverage = 1.0f;
481 if(rest != nullptr)
482 {
483 out.transform = combine_node(rest->transform, layer_node.transform, mode, bone_weight);
484 }
485 }
486 continue;
487 }
488 const auto& base_node = base.nodes[ib++];
489 const math::transform base_value = resolve_coverage(base_node, rest);
490 auto& out = result.nodes.emplace_back();
491 out.desc = base_node.desc;
492 out.transform = bone_weight > 0.0f ? combine_node(base_value, layer_node.transform, mode, bone_weight) : base_value;
493 }

515void blend_space_def::compute_blend(const parameters_t& current_params,
516 std::vector<blend_space_weight>& out_clips) const
517{
518 // Clear the output vector
519 out_clips.clear();
520
521 if(points_.empty())
522 {
523 return;
524 }
525
526 if(current_params.size() < parameter_count_)
527 {
528 // Parameters were never (fully) supplied - fall back to the first
529 // point instead of reading out of bounds / silently freezing.
530 out_clips.push_back({points_.front().clip, 1.0f, 0});
531 return;
532 }
533
534 if(parameter_count_ == 1)
535 {
536 compute_blend_1d(current_params, out_clips);
537 return;
538 }
539 if(parameter_count_ == 2)
540 {
541 compute_blend_2d(current_params, out_clips);
542 return;
543 }
544
545 // Blend spaces are 1D or 2D; any other point layout plays its first point (controller
546 // validation reports it).
547 out_clips.push_back({points_.front().clip, 1.0f, 0});
548}
549
550void blend_space_def::compute_blend_1d(const parameters_t& current_params,
551 std::vector<blend_space_weight>& out_clips) const
552{
553 const float param = current_params[0];
554 // The points on either side of the parameter: the highest at or below it and the lowest at or
555 // above it. Of several points on one value the last one counts.
556 int below = -1;
557 int above = -1;
558 for(size_t i = 0; i < points_.size(); ++i)
559 {
560 const float value = points_[i].parameters[0];
561 if(value <= param && (below < 0 || value >= points_[size_t(below)].parameters[0]))
562 {
563 below = int(i);
564 }
565 if(value >= param && (above < 0 || value <= points_[size_t(above)].parameters[0]))
566 {
567 above = int(i);
568 }
569 }
570 if(below < 0 && above < 0)
571 {
572 // Not a number: no point compares; play the first.
573 out_clips.push_back({points_.front().clip, 1.0f, 0});
574 return;
575 }
576 // Outside the range the nearest end plays alone.
577 if(below < 0 || above < 0 || below == above)
578 {
579 const size_t point = size_t(below >= 0 ? below : above);
580 out_clips.push_back({points_[point].clip, 1.0f, point});
581 return;
582 }
583 const float v0 = points_[size_t(below)].parameters[0];
584 const float v1 = points_[size_t(above)].parameters[0];
585 const float t = v1 - v0 > BLEND_1D_EPSILON ? math::clamp((param - v0) / (v1 - v0), 0.0f, 1.0f) : 0.0f;
586 out_clips.push_back({points_[size_t(below)].clip, 1.0f - t, size_t(below)});
587 out_clips.push_back({points_[size_t(above)].clip, t, size_t(above)});
588}

629auto cartesian_band(const vec2_param& pi, const vec2_param& pj, const vec2_param& p) -> float
630{
631 const vec2_param pij{pj.x - pi.x, pj.y - pi.y};
632 const float len2 = dot2(pij, pij);
633 if(len2 < BLEND_POINT_EPSILON * BLEND_POINT_EPSILON)
634 {
635 return 1.0f;
636 }
637 const vec2_param pip{p.x - pi.x, p.y - pi.y};
638 return 1.0f - dot2(pip, pij) / len2;
639}

648auto directional_band(const vec2_param& pi, const vec2_param& pj, const vec2_param& p) -> float
649{
650 const float mi = length2d(pi);
651 const float mj = length2d(pj);
652 const float mp = length2d(p);
653 const float average = (mi + mj) * 0.5f;
654 if(average < BLEND_POINT_EPSILON)
655 {
656 return 1.0f;
657 }
658 const bool i_has_direction = mi > BLEND_POINT_EPSILON;
659 const bool j_has_direction = mj > BLEND_POINT_EPSILON;
660 const bool p_has_direction = mp > BLEND_POINT_EPSILON;
661
662 float angle_ij = 0.0f;
663 float angle_ip = 0.0f;
664 if(i_has_direction && j_has_direction)
665 {
666 angle_ij = signed_angle(pi, pj);
667 angle_ip = p_has_direction ? signed_angle(pi, p) : 0.0f;
668 if(std::fabs(std::fabs(angle_ij) - math::pi<float>()) < OPPOSITE_ANGLE_EPSILON)
669 {
670 // Opposite points have no side: compare unsigned angles, so the result does not hinge
671 // on the sign of a zero cross product.
672 angle_ij = std::fabs(angle_ij);
673 angle_ip = std::fabs(angle_ip);
674 }
675 else
676 {
677 // Measure the point's angle on the branch around the pair, so a point just past the
678 // half turn is not read as a full turn the other way.
679 const float middle = angle_ij * 0.5f;
680 if(angle_ip < middle - math::pi<float>())
681 {
682 angle_ip += math::two_pi<float>();
683 }
684 else if(angle_ip >= middle + math::pi<float>())
685 {
686 angle_ip -= math::two_pi<float>();
687 }
688 }
689 }
690
691 const vec2_param pij{(mj - mi) / average, angle_ij * DIRECTIONAL_ANGLE_SCALE};
692 const vec2_param pip{(mp - mi) / average, angle_ip * DIRECTIONAL_ANGLE_SCALE};
693 const float len2 = dot2(pij, pij);
694 if(len2 < BLEND_POINT_EPSILON * BLEND_POINT_EPSILON)
695 {
696 return 1.0f;
697 }
698 return 1.0f - dot2(pip, pij) / len2;
699}
700} // namespace

711void blend_space_def::compute_blend_2d(const parameters_t& current_params,
712 std::vector<blend_space_weight>& out_clips) const
713{
714 out_clips.clear();
715 if(parameter_count_ != 2 || points_.empty())
716 {
717 return;
718 }
719
720 const vec2_param p{current_params[0], current_params[1]};
721 if(points_.size() == 1)
722 {
723 out_clips.push_back({points_.front().clip, 1.0f, 0});
724 return;
725 }
726
727 const bool directional = mode_2d_ == blend_space_2d_mode::freeform_directional;
728 float total = 0.0f;
729 out_clips.reserve(points_.size());
730 for(size_t i = 0; i < points_.size(); ++i)
731 {
732 const vec2_param pi{points_[i].parameters[0], points_[i].parameters[1]};
733 float influence = 1.0f;
734 for(size_t j = 0; j < points_.size() && influence > 0.0f; ++j)
735 {
736 if(i == j)
737 {
738 continue;
739 }
740 const vec2_param pj{points_[j].parameters[0], points_[j].parameters[1]};
741 const float band = directional ? directional_band(pi, pj, p) : cartesian_band(pi, pj, p);
742 influence = std::min(influence, band);
743 }
744 influence = std::max(influence, 0.0f);
745 if(influence > 0.0f)
746 {
747 out_clips.push_back({points_[i].clip, influence, i});
748 total += influence;
749 }
750 }
751
752 if(total <= 0.0f)
753 {
754 // Every band excluded the input (only possible for degenerate layouts): nearest point.
755 out_clips.clear();
756 size_t best = 0;
757 float best_distance = std::numeric_limits<float>::max();
758 for(size_t i = 0; i < points_.size(); ++i)
759 {
760 const vec2_param offset{points_[i].parameters[0] - p.x, points_[i].parameters[1] - p.y};
761 const float distance = dot2(offset, offset);
762 if(distance < best_distance)
763 {
764 best_distance = distance;
765 best = i;
766 }
767 }
768 out_clips.push_back({points_[best].clip, 1.0f, best});
769 return;
770 }
771
772 for(auto& clip_weight : out_clips)
773 {
774 clip_weight.weight /= total;
775 }
776}
