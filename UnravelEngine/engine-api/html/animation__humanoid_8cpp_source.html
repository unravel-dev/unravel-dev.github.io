90auto build_tree(const std::vector<std::string>& names,
91 const std::vector<int>& parents,
92 const std::vector<math::vec3>& positions,
93 size_t count) -> skeleton_tree
94{
95 skeleton_tree tree{names, parents, positions};
96 tree.children.resize(count);
97 tree.lowest.resize(count);
98 tree.highest.resize(count);
99 tree.size.assign(count, 1);
100 tree.depth.assign(count, 1);
101 tree.floor = std::numeric_limits<float>::max();
102 for(size_t i = 0; i < count; ++i)
103 {
104 tree.lowest[i] = tree.highest[i] = positions[i].y;
105 tree.floor = std::min(tree.floor, positions[i].y);
106 if(parents[i] >= 0 && size_t(parents[i]) < i)
107 {
108 tree.children[size_t(parents[i])].push_back(int(i));
109 }
110 }
111 // Parents come first, so walking backwards finishes every subtree before its parent.
112 for(size_t i = count; i-- > 0;)
113 {
114 for(int child : tree.children[i])
115 {
116 tree.lowest[i] = std::min(tree.lowest[i], tree.lowest[size_t(child)]);
117 tree.highest[i] = std::max(tree.highest[i], tree.highest[size_t(child)]);
118 tree.size[i] += tree.size[size_t(child)];
119 tree.depth[i] = std::max(tree.depth[i], tree.depth[size_t(child)] + 1);
120 }
121 }
122 return tree;
123}

243auto follow_chain(const skeleton_tree& tree, int start, size_t length) -> std::vector<int>
244{
245 std::vector<int> chain;
246 for(int bone = start; bone >= 0 && chain.size() < length;)
247 {
248 const int next = find_main_child(tree, bone);
249 const std::string& name = tree.names[size_t(bone)];
250 const bool is_joint = chain.empty() || (!is_segment_of(name, tree.names[size_t(chain.back())]) &&

252 if(is_joint)

256 bone = next;
257 }
258 return chain;

285auto find_left_axis(const skeleton_tree& tree, const std::vector<int>& legs) -> math::vec3
286{
287 constexpr size_t LEG_JOINTS = 4;
288 math::vec3 forward(0.0f);
289 for(int leg : legs)
290 {
291 const auto chain = follow_chain(tree, leg, LEG_JOINTS);
292 if(chain.size() < LEG_JOINTS)
293 {
294 continue;
295 }
296 math::vec3 toes = tree.positions[size_t(chain[3])] - tree.positions[size_t(chain[2])];
297 toes.y = 0.0f;
298 if(math::length(toes) > MIN_TOE_DIRECTION)
299 {
300 forward += math::normalize(toes);
301 }
302 }
303 if(math::length(forward) <= MIN_TOE_DIRECTION)
304 {
305 return {1.0f, 0.0f, 0.0f};
306 }
307 forward = math::normalize(forward);
308 return {forward.z, 0.0f, -forward.x};
309}

315{

324 const float current_lowest = tree.lowest[size_t(current)];
325 if(std::fabs(lowest - current_lowest) > tie)

335auto find_legs(const skeleton_tree& tree, int hips) -> std::pair<int, int>
336{
337 const math::vec3& position = tree.positions[size_t(hips)];
338 const float hips_height = position.y - tree.floor;
339 const float floor_reach = tree.floor + LEG_REACH * hips_height;
340 const auto reaching_down = [&](int bone)
341 {
342 std::vector<int> down;
343 for(int child : tree.children[size_t(bone)])
344 {
345 if(tree.lowest[size_t(child)] < floor_reach)
346 {
347 down.push_back(child);
348 }
349 }
350 return down;
351 };
352 const auto down = reaching_down(hips);
353 auto legs = pick_legs(tree, down, position, hips_height);
354 if(legs.first >= 0 && legs.second >= 0)
355 {
356 return legs;
357 }
358 for(int pelvis : down)
359 {
360 legs = pick_legs(tree, reaching_down(pelvis), position, hips_height);
361 if(legs.first >= 0 && legs.second >= 0)
362 {
363 return legs;
364 }
365 }
366 return {-1, -1};
367}

388auto find_hips(const skeleton_tree& tree, size_t count) -> int
389{
390 int best = -1;
391 bool best_holds_legs = false;
392 for(size_t i = 0; i < count; ++i)
393 {
394 // The hips stand well above the floor: a root bone near the feet, whose only child holds
395 // the legs, is not them.
396 const float height = tree.positions[i].y - tree.floor;
397 if(height <= MIN_HIPS_HEIGHT * (tree.highest[i] - tree.floor))
398 {
399 continue;
400 }
401 const auto legs = find_legs(tree, int(i));
402 if(legs.first < 0 || legs.second < 0 || find_spine_start(tree, int(i)) < 0)
403 {
404 continue;
405 }
406 const bool holds_legs = tree.parents[size_t(legs.first)] == int(i);
407 const bool is_better = best < 0 || (holds_legs && !best_holds_legs) ||
408 (holds_legs == best_holds_legs && tree.size[i] > tree.size[size_t(best)]);
409 if(is_better)
410 {
411 best = int(i);
412 best_holds_legs = holds_legs;
413 }
414 }
415 return best;
416}

457void assign_arm(const skeleton_tree& tree,
458 int start,
459 const math::vec3& spine,
460 const math::vec3& lateral,
461 humanoid_bone shoulder,
462 humanoid_map& map)
463{
464 const auto chain = follow_chain(tree, start, 5);
465 size_t first = 0;
466 if(chain.size() >= 4)
467 {
468 const auto& p = tree.positions;
469 const float span0 = std::fabs(math::dot(p[size_t(chain[0])] - spine, lateral));
470 const float span1 = std::fabs(math::dot(p[size_t(chain[1])] - spine, lateral));
471 const float length0 = math::length(p[size_t(chain[1])] - p[size_t(chain[0])]);
472 const float length1 = math::length(p[size_t(chain[2])] - p[size_t(chain[1])]);
473 if(span0 < SHOULDER_SPAN * span1 || length0 < SHOULDER_LENGTH * length1)
474 {
475 set(map, shoulder, chain[0]);
476 first = 1;
477 }
478 }
479 for(size_t i = 0; i < 3 && first + i < chain.size(); ++i)
480 {
481 set(map, humanoid_bone(size_t(shoulder) + 1 + i), chain[first + i]);
482 }
483}

495void assign_head(const skeleton_tree& tree, int upper_chest, std::pair<int, int> arms, humanoid_map& map)
496{
497 const float chest_height = tree.positions[size_t(upper_chest)].y;
498 int start = -1;
499 for(int child : tree.children[size_t(upper_chest)])
500 {
501 if(child != arms.first && child != arms.second && tree.highest[size_t(child)] > chest_height &&
502 (start < 0 || tree.highest[size_t(child)] > tree.highest[size_t(start)]))
503 {
504 start = child;
505 }
506 }
507 if(start < 0)
508 {
509 return;
510 }
511 std::vector<int> chain{start};
512 while(chain.size() < MAX_CHAIN && tree.children[size_t(chain.back())].size() == 1)
513 {
514 const int next = tree.children[size_t(chain.back())].front();
515 // A helmet or other attachment hangs from the head; the head chain only climbs.
516 const bool climbs = tree.positions[size_t(next)].y > tree.positions[size_t(chain.back())].y;
517 if(!climbs || (tree.children[size_t(next)].empty() && is_end_bone(tree.names[size_t(next)])))
518 {
519 break;
520 }
521 chain.push_back(next);
522 }
523 set(map, humanoid_bone::head, chain.back());
524 if(chain.size() >= 2)
525 {
526 set(map, humanoid_bone::neck, chain.front());
527 }
528}

531auto find_humanoid_bones(const std::vector<std::string>& names,
532 const std::vector<int>& parents,
533 const std::vector<math::vec3>& rest_positions) -> humanoid_map
534{
535 humanoid_map map;
536 map.fill(-1);
537 const size_t count = std::min({names.size(), parents.size(), rest_positions.size()});
538 if(count == 0)
539 {
540 return map;
541 }
542 const auto tree = build_tree(names, parents, rest_positions, count);
543 const int hips = find_hips(tree, count);
544 if(hips < 0)
545 {
546 return map;
547 }
548 set(map, humanoid_bone::hips, hips);
549 const math::vec3& hips_position = rest_positions[size_t(hips)];
550 const float hips_height = hips_position.y - tree.floor;
551
552 const auto legs = find_legs(tree, hips);
553 assign_leg(tree, legs.first, humanoid_bone::left_upper_leg, map);
554 assign_leg(tree, legs.second, humanoid_bone::right_upper_leg, map);
555 // Left to right, from the thighs: the body's own lateral axis whatever way it faces.
556 math::vec3 lateral = rest_positions[size_t(legs.first)] - rest_positions[size_t(legs.second)];
557 lateral.y = 0.0f;
558 lateral = math::length(lateral) > 0.0f ? math::normalize(lateral) : math::vec3(1.0f, 0.0f, 0.0f);
559
560 // Up the spine to the bone the arms branch from.
561 std::vector<int> spine{find_spine_start(tree, hips)};
562 auto arms = find_arms(tree, spine.back(), hips_position, lateral, hips_height);
563 while((arms.first < 0 || arms.second < 0) && spine.size() < MAX_CHAIN)
564 {
565 int next = -1;
566 for(int child : tree.children[size_t(spine.back())])
567 {
568 if(next < 0 || tree.highest[size_t(child)] > tree.highest[size_t(next)])
569 {
570 next = child;
571 }
572 }
573 if(next < 0)
574 {
575 break;
576 }
577 spine.push_back(next);
578 arms = find_arms(tree, next, hips_position, lateral, hips_height);
579 }
580 // 3ds Max's Biped hangs the arms from the neck: a branch bone named so is the neck, and the
581 // spine bone below it the upper chest.
582 int head_base = spine.back();
583 if(spine.size() >= 2 && to_lower(names[size_t(spine.back())]).find("neck") != std::string::npos)
584 {
585 set(map, humanoid_bone::neck, spine.back());
586 spine.pop_back();
587 }
588 const int upper_chest = spine.back();
589 set(map, humanoid_bone::upper_chest, upper_chest);
590 if(spine.size() >= 2)
591 {
592 set(map, humanoid_bone::spine, spine.front());
593 }
594 if(spine.size() >= 3)
595 {
596 set(map, humanoid_bone::chest, spine[spine.size() - 2]);
597 }
598 if(arms.first >= 0 && arms.second >= 0)
599 {
600 const math::vec3& axis = rest_positions[size_t(upper_chest)];
601 assign_arm(tree, arms.first, axis, lateral, humanoid_bone::left_shoulder, map);
602 assign_arm(tree, arms.second, axis, lateral, humanoid_bone::right_shoulder, map);
603 }
604 assign_head(tree, head_base, arms, map);
605 return map;
606}

655void turn_subtree(const std::vector<int>& parents, int bone, const math::quat& turn, std::vector<math::transform>& globals)
656{
657 const math::vec3 pivot = globals[size_t(bone)].get_position();
658 std::vector<bool> is_moved(globals.size(), false);
659 for(size_t i = size_t(bone); i < globals.size(); ++i)
660 {
661 const int parent = i < parents.size() ? parents[i] : -1;
662 is_moved[i] = i == size_t(bone) || (parent >= 0 && size_t(parent) < i && is_moved[size_t(parent)]);
663 if(!is_moved[i])
664 {
665 continue;
666 }
667 auto& global = globals[i];
668 global.set_position(pivot + turn * (global.get_position() - pivot));
669 global.set_rotation(math::normalize(turn * global.get_rotation()));
670 }

706 if(!is_bone_in(left_bone, globals) || !is_bone_in(right_bone, globals))

709 }

715 }
716 }

720void align_humanoid_to_t_pose(const std::vector<int>& parents, const humanoid_map& roles, std::vector<math::transform>& globals)
721{
722 const math::vec3 side = find_humanoid_side_axis(roles, globals);
723 if(math::length(side) <= MIN_POSE_DIRECTION)
724 {
725 return;
726 }
727 for(const auto& segment : T_POSE_SEGMENTS)
728 {
729 const int from = roles[size_t(segment.from)];
730 const int to = roles[size_t(segment.to)];
731 if(!is_bone_in(from, globals) || !is_bone_in(to, globals))
732 {
733 continue;
734 }
735 const math::vec3 direction = globals[size_t(to)].get_position() - globals[size_t(from)].get_position();
736 if(math::length(direction) <= MIN_POSE_DIRECTION)
737 {
738 continue;
739 }
740 const math::vec3 wanted = side * segment.side + HUMANOID_UP * segment.up;
741 turn_subtree(parents, from, math::from_to_rotation(direction, wanted), globals);
742 }
743}

745void apply_humanoid_pose_offsets(const std::vector<int>& parents,
746 const humanoid_map& roles,
747 const std::vector<humanoid_pose_offset>& offsets,
748 std::vector<math::transform>& globals)
749{
750 const math::vec3 side = offsets.empty() ? math::vec3(0.0f) : find_humanoid_side_axis(roles, globals);
751 if(math::length(side) <= MIN_POSE_DIRECTION)
752 {
753 return;
754 }
755 // Character space to the skeleton's frame: x is the side axis, y up, z forward.
756 const math::mat3 axes(side, HUMANOID_UP, math::cross(side, HUMANOID_UP));
757 const math::quat basis = math::normalize(math::quat_cast(axes));
758 for(const auto& offset : offsets)
759 {
760 const int bone = offset.role < humanoid_bone::count ? roles[size_t(offset.role)] : -1;
761 if(!is_bone_in(bone, globals))
762 {
763 continue;
764 }
765 math::vec3 degrees = offset.rotation;
766 // Mirrored across the character's middle: turns about the side axis keep their sign, turns
767 // about up and forward flip it.
768 if(is_right_side(offset.role))
769 {
770 degrees.y = -degrees.y;
771 degrees.z = -degrees.z;
772 }
773 const math::quat turn = math::normalize(basis * math::quat(math::radians(degrees)) * glm::inverse(basis));
774 turn_subtree(parents, bone, turn, globals);
775 }
776}
